123ReturnValue ImplementCurrentControl::setRefCurrents(const int n_joints, const int *joints, const double *t)
124{
125 std::lock_guard lock(m_imp_mutex);
126 POINTERCHECK(IMPLEMENT_LAYER_COMPONENT,joints)
127 POINTERCHECK(IMPLEMENT_LAYER_COMPONENT,t)
128 JOINTSIDCHECK(IMPLEMENT_LAYER_COMPONENT,n_joints, joints)
129
130 for(int idx=0; idx<n_joints; idx++)
131 {
132 m_buffer_ints[idx] = castToMapper(m_helper)->toHw(joints[idx]);
133 m_buffer_doubles[idx] = castToMapper(m_helper)->ampereA2S(t[idx], joints[idx]);
134 }
135
136 ReturnValue ret = m_iraw->setRefCurrentsRaw(n_joints, m_buffer_ints.data(), m_buffer_doubles.data());

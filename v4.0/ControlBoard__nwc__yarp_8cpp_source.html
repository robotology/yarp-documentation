46inline bool getTimeStamp(Bottle &bot, Stamp &st)
47{
48 if (bot.get(3).asVocab32()==VOCAB_TIMESTAMP)
49 {
50 //yup! we have a timestamp
51 int fr=bot.get(4).asInt32();
52 double ts=bot.get(5).asFloat64();
53 st=Stamp(fr,ts);
54 return true;
55 }
56 return false;
57}

62class DiagnosticThread :
63 public PeriodicThread
64{
65 StateExtendedInputPort *owner{nullptr};
66 std::string ownerName;
67
68public:
69 using PeriodicThread::PeriodicThread;
70
71 void setOwner(StateExtendedInputPort *o)
72 {
73 owner = o;
74 ownerName = owner->getName();
75 }
76
77 void run() override
78 {
79 if (owner != nullptr)
80 {
81 if (owner->getIterations() > 100)
82 {
83 int it;
84 double av;
85 double max;
86 double min;
87 owner->getEstFrequency(it, av, min, max);
88 owner->resetStat();
89 yCDebug(CONTROLBOARD_NWC_YARP,
90 "%s: %d msgs av:%.2lf min:%.2lf max:%.2lf [ms]",
91 ownerName.c_str(),
92 it,
93 av,
94 min,
95 max);
96 }
97
98 }
99 }

120bool ControlBoard_nwc_yarp::open(Searchable& config)
121{
122 if (!parseParams(config)) { return false; }
123
124 extendedIntputStatePort.setTimeout(m_timeout);
125
126 //handle local Qos
127 yarp::os::QosStyle localQos;
128 if (m_local_qos_enable)
129 {
130 localQos.setThreadPriority(m_local_qos_thread_priority);
131 localQos.setThreadPolicy(m_local_qos_thread_policy);
132 localQos.setPacketPriority(m_local_qos_packet_priority);
133 }
134
135 //handle remote Qos
136 yarp::os::QosStyle remoteQos;
137 if (m_remote_qos_enable)
138 {
139 remoteQos.setThreadPriority(m_remote_qos_thread_priority);
140 remoteQos.setThreadPolicy(m_remote_qos_thread_policy);
141 remoteQos.setPacketPriority(m_remote_qos_packet_priority);
142 }
143
144 //handle param writeStrict
145 if (m_writeStrict == "on")
146 {
147 writeStrict_singleJoint = true;
148 writeStrict_moreJoints = true;
149 yCInfo(CONTROLBOARD_NWC_YARP, "RemoteControlBoard is ENABLING the writeStrict option for all commands");
150 }
151 else if(m_writeStrict == "off")
152 {
153 writeStrict_singleJoint = false;
154 writeStrict_moreJoints = false;
155 yCInfo(CONTROLBOARD_NWC_YARP, "RemoteControlBoard is DISABLING the writeStrict option for all commands");
156 }
157 else if (m_writeStrict.empty())
158 {
159 //leave the default values
160 }
161 else
162 {
163 yCError(CONTROLBOARD_NWC_YARP, "Found writeStrict option with wrong value. Accepted options are 'on' or 'off'");
164 return false;
165 }
166
167 //open ports
168 bool portProblem = false;
169 if (m_local != "") {
170 std::string s1;
171 s1 = m_local;
172 s1 += "/command:o";
173 if (!command_p.open(s1)) { portProblem = true; }
174 s1 = m_local;
175 s1 += "/stateExt:i";
176 if (!extendedIntputStatePort.open(s1)) { portProblem = true; }
177 if (!portProblem)
178 {
179 extendedIntputStatePort.useCallback();
180 }
181 }
182
183 bool connectionProblem = false;
184 if (m_remote != "" && !portProblem)
185 {
186 std::string s1 = m_remote;
187 s1 += "/nws/rpc:i";
188 std::string s2 = m_local;
189 s2 += "/rpc:o";
190 bool ok = false;
191 // RPC port needs to be tcp, therefore no carrier option is added here
192 // ok=Network::connect(s2.c_str(), s1.c_str()); //This doesn't take into consideration possible YARP_PORT_PREFIX on local ports
193 // ok=Network::connect(rpc_p.getName(), s1.c_str()); //This should work also with YARP_PORT_PREFIX because getting back the name of the port will return the modified name
194
195 s1 = m_remote;
196 s1 += "/command:i";
197 s2 = m_local;
198 s2 += "/command:o";
199 //ok = Network::connect(s2.c_str(), s1.c_str(), carrier);
200 // ok=Network::connect(command_p.getName(), s1.c_str(), carrier); //doesn't take into consideration possible YARP_PORT_PREFIX on local ports
201 ok = command_p.addOutput(s1, m_carrier_cmd);
202 if (!ok) {
203 yCError(CONTROLBOARD_NWC_YARP, "Problem connecting to %s, is the remote device available?", s1.c_str());
204 connectionProblem = true;
205 }
206 // set the QoS preferences for the 'command' port
207 if (m_local_qos_enable || m_remote_qos_enable) {
208 NetworkBase::setConnectionQos(command_p.getName(), s1, localQos, remoteQos, false);
209 }
210
211 s1 = m_remote;
212 s1 += "/stateExt:o";
213 s2 = m_local;
214 s2 += "/stateExt:i";
215 // not checking return value for now since it is wip (different machines can have different compilation flags
216 ok = Network::connect(s1, extendedIntputStatePort.getName(), m_carrier);
217 if (ok)
218 {
219 // set the QoS preferences for the 'state' port
220 if (m_local_qos_enable || m_remote_qos_enable) {
221 NetworkBase::setConnectionQos(s1, extendedIntputStatePort.getName(), remoteQos, localQos, false);
222 }
223 }
224 else
225 {
226 yCError(CONTROLBOARD_NWC_YARP, "Problem connecting to %s, is the remote device available?", s1.c_str());
227 connectionProblem = true;
228 }
229 }
230
231 if (connectionProblem||portProblem)
232 {
233 command_p.close();
234 extendedIntputStatePort.close();
235 return false;
236 }
237
238 command_buffer.attach(command_p);
239
240 // open rpc port
241 std::string local_rpc_portname = m_local + "/nwc/rpc:o";
242 if (!m_rpcPort.open(local_rpc_portname))
243 {
244 yCError(CONTROLBOARD_NWC_YARP, "open() error could not open rpc port %s, check network\n", local_rpc_portname.c_str());
245 command_p.close();
246 extendedIntputStatePort.close();
247 return false;
248 }
249
250 //Attach the ControlBoardMsgs to the port
251 if (!m_RPC.yarp().attachAsClient(m_rpcPort))
252 {
253 yCError(CONTROLBOARD_NWC_YARP, "Error! Cannot attach the port as a client");
254 command_p.close();
255 extendedIntputStatePort.close();
256 m_rpcPort.close();
257 return false;
258 }
259
260 // connection for the rpc port
261 std::string nws_rpc_portname = m_remote + "/nws/rpc:i";
262 if (!Network::connect(local_rpc_portname, nws_rpc_portname))
263 {
264 yCError(CONTROLBOARD_NWC_YARP, "open() error could not connect to %s\n", nws_rpc_portname.c_str());
265 command_p.close();
266 extendedIntputStatePort.close();
267 m_rpcPort.close();
268 return false;
269 }
270
271 // Check the protocol version
272 if (!m_RPC.checkProtocolVersion())
273 {
274 command_p.close();
275 extendedIntputStatePort.close();
276 m_rpcPort.close();
277 return false;
278 }
279
280 if (!isLive())
281 {
282 if (m_remote!="")
283 {
284 yCError(CONTROLBOARD_NWC_YARP, "Problems with obtaining the number of controlled axes");
285 command_p.close();
286 extendedIntputStatePort.close();
287 m_rpcPort.close();
288 return false;
289 }
290 }
291
292 if (m_diagnostic)
293 {
294 diagnosticThread = new DiagnosticThread(DIAGNOSTIC_THREAD_PERIOD);
295 diagnosticThread->setOwner(&extendedIntputStatePort);
296 diagnosticThread->start();
297 }
298
299 // allocate memory for helper struct
300 // single joint
301 last_singleJoint.jointPosition.resize(1);
302 last_singleJoint.jointVelocity.resize(1);
303 last_singleJoint.jointAcceleration.resize(1);
304 last_singleJoint.motorPosition.resize(1);
305 last_singleJoint.motorVelocity.resize(1);
306 last_singleJoint.motorAcceleration.resize(1);
307 last_singleJoint.torque.resize(1);
308 last_singleJoint.pwmDutycycle.resize(1);
309 last_singleJoint.current.resize(1);
310 last_singleJoint.controlMode.resize(1);
311 last_singleJoint.interactionMode.resize(1);
312
313 // whole part (safe here because we already got the m_nj
314 last_wholePart.jointPosition.resize(m_nj);
315 last_wholePart.jointVelocity.resize(m_nj);
316 last_wholePart.jointAcceleration.resize(m_nj);
317 last_wholePart.motorPosition.resize(m_nj);
318 last_wholePart.motorVelocity.resize(m_nj);
319 last_wholePart.motorAcceleration.resize(m_nj);
320 last_wholePart.torque.resize(m_nj);
321 last_wholePart.current.resize(m_nj);
322 last_wholePart.pwmDutycycle.resize(m_nj);
323 last_wholePart.controlMode.resize(m_nj);
324 last_wholePart.interactionMode.resize(m_nj);
325
326 //check device capabilities
327 auto ret = m_RPC.getDeviceInterfacesRPC();
328 if (!ret.ret)
329 {
330 yCError(CONTROLBOARD_NWC_YARP, "Unable to getDeviceInterfacesRPC");
331 command_p.close();
332 extendedIntputStatePort.close();
333 m_rpcPort.close();
334 return false;
335 }
336 m_device_has_interfaces = ret.interfaces;
337
338 return true;
339}

341bool ControlBoard_nwc_yarp::close()
342{
343 if (diagnosticThread!=nullptr)
344 {
345 diagnosticThread->stop();
346 delete diagnosticThread;
347 }
348
349 command_buffer.detach(); //TBC
350
351 m_rpcPort.interrupt(); // TBC
352 command_p.interrupt(); // TBC
353 extendedIntputStatePort.interrupt(); // TBC
354
355 m_rpcPort.close();
356 command_p.close();
357 extendedIntputStatePort.close();
358 return true;
359}

30bool ImageRotation::create(const yarp::os::Property& options)
31{
32 if (options.check("options_rotate"))
33 {
34 const std::string val = options.find("options_rotate").asString();
35 if (val == "rotate_cw") { m_rotate_option = yarp::sig::utils::RotateOption::rotate_cw; }
36 else if (val == "rotate_ccw") { m_rotate_option = yarp::sig::utils::RotateOption::rotate_ccw; }
37 else if (val == "rotate_180") { m_rotate_option = yarp::sig::utils::RotateOption::rotate_180; }
38 else if (val == "rotate_none"){ m_rotate_option = yarp::sig::utils::RotateOption::rotate_none; }
39 else
40 {
41 yCDebug(IMAGEROTATION) << "Invalid value of `options_rotate` parameter:" << val;
42 return false;
43 }
44 }
45
46 if (options.check("options_flip"))
47 {
48 const std::string val = options.find("options_flip").asString();
49 if (val == "flip_x") { m_flip_option = yarp::sig::utils::FlipOption::flip_x; }
50 else if (val == "flip_y") { m_flip_option = yarp::sig::utils::FlipOption::flip_y; }
51 else if (val == "flip_xy") { m_flip_option = yarp::sig::utils::FlipOption::flip_xy; }
52 else if (val == "flip_none"){ m_flip_option = yarp::sig::utils::FlipOption::flip_none; }
53 else
54 {
55 yCDebug(IMAGEROTATION) << "Invalid value of `options_flip` parameter:" << val;
56 return false;
57 }
58 }
59
60 return true;
61}

77bool ImageRotation::accept(yarp::os::Things& thing)
78{
79 auto* img = thing.cast_as<yarp::sig::Image>();
80 if (img == nullptr) {
81 yCError(IMAGEROTATION, "Expected type Image, but got wrong data type!");
82 return false;

84 if (img->getPixelCode() != VOCAB_PIXEL_RGB && img->getPixelCode() != VOCAB_PIXEL_MONO_FLOAT) {
85 yCError(IMAGEROTATION, "Received image with invalid/unsupported pixelCode!");

91yarp::os::Things& ImageRotation::update(yarp::os::Things& thing)
92{
93 yarp::sig::Image* yarpimg = thing.cast_as<yarp::sig::Image>();
94 if (yarpimg->getPixelCode() == VOCAB_PIXEL_RGB)
95 {
96 yarp::sig::utils::rotate(m_outImgRgb, *yarpimg, m_rotate_option, m_flip_option);
97 m_th.setPortWriter(&m_outImgRgb);
98 }
99 else if (yarpimg->getPixelCode() == VOCAB_PIXEL_MONO_FLOAT)
100 {
101 yarp::sig::utils::rotate(m_outImgFloat, *yarpimg, m_rotate_option, m_flip_option);
102 m_th.setPortWriter(&m_outImgFloat);
103 }
104 else {
105 yCError(IMAGEROTATION, "Invalid Image type!");
106 }
107 return m_th;

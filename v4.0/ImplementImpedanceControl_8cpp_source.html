78{
79 std::lock_guard lock(m_imp_mutex);
80 JOINTIDCHECK(IMPLEMENT_LAYER_COMPONENT,j)
81 POINTERCHECK(IMPLEMENT_LAYER_COMPONENT,stiffness)
82 POINTERCHECK(IMPLEMENT_LAYER_COMPONENT,damping)
83
84 int k;
85 k=castToMapper(m_helper)->toHw(j);
86 ReturnValue ret=m_iraw->getImpedanceRaw(k, stiffness, damping);
87 *stiffness = (castToMapper(m_helper)->impS2N(*stiffness, k));
88 *damping = (castToMapper(m_helper)->impS2N(*damping, k));
89 //prevent negative stiffness
90 *stiffness = fabs (*stiffness);

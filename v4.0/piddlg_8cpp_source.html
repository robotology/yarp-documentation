120PidDlg::PidDlg(QString partname, int jointIndex, QString jointName, int cmode, QWidget *parent) :
121 QDialog(parent),
122 ui(new Ui::PidDlg)
123{
124 ui->setupUi(this);
125
126 //pos, posd, vel, veld, trq, cur, pwm
127 switch (cmode)
128 {
129 case 1:
130 ui->tabMain->setCurrentIndex(getCurrentTabIndexByName(ui->tabMain, "Position PID"));
131 break;
132 case 2:
133 ui->tabMain->setCurrentIndex(getCurrentTabIndexByName(ui->tabMain, "PositionDirect PID"));
134 break;
135 case 3:
136 ui->tabMain->setCurrentIndex(getCurrentTabIndexByName(ui->tabMain, "Mixed PID"));
137 break;
138 case 4:
139 ui->tabMain->setCurrentIndex(getCurrentTabIndexByName(ui->tabMain, "Velocity PID"));
140 break;
141 case 5:
142 ui->tabMain->setCurrentIndex(getCurrentTabIndexByName(ui->tabMain, "VelocityDirect PID"));
143 break;
144 case 6:
145 ui->tabMain->setCurrentIndex(getCurrentTabIndexByName(ui->tabMain, "Torque PID"));
146 break;
147 case 7:
148 ui->tabMain->setCurrentIndex(getCurrentTabIndexByName(ui->tabMain, "PWM Params"));
149 break;
150 case 8:
151 ui->tabMain->setCurrentIndex(getCurrentTabIndexByName(ui->tabMain, "Current PID"));
152 break;
153 default:
154 ui->tabMain->setCurrentIndex(0);
155 break;
156 }
157
158 this->jointIndex = jointIndex;
159
160 QString title = QString("Pid Control %1 JNT:%2 (%3)").arg(partname).arg(jointIndex).arg(jointName);
161 setWindowTitle(title);
162
163 connect(ui->btnRefresh, SIGNAL(clicked()), this, SLOT(onRefresh()));
164 connect(ui->btnSend,SIGNAL(clicked()),this,SLOT(onSend()));
165 connect(ui->btnCancel,SIGNAL(clicked()),this,SLOT(onCancel()));
166 connect(ui->btnDump, SIGNAL(clicked()), this, SLOT(onDumpRemoteVariables()));
167 connect(ui->selectPosition, SIGNAL(currentIndexChanged(int)), this, SLOT(onComboBoxIndexChangedPos(int)));
168 connect(ui->selectVelocity, SIGNAL(currentIndexChanged(int)), this, SLOT(onComboBoxIndexChangedVel(int)));
169 connect(ui->selectPositionDir, SIGNAL(currentIndexChanged(int)), this, SLOT(onComboBoxIndexChangedPosDir(int)));
170 connect(ui->selectVelocityDir, SIGNAL(currentIndexChanged(int)), this, SLOT(onComboBoxIndexChangedVelDir(int)));
171 connect(ui->selectTorque , SIGNAL(currentIndexChanged(int)), this, SLOT(onComboBoxIndexChangedTrq(int)));
172 connect(ui->selectCurrent, SIGNAL(currentIndexChanged(int)), this, SLOT(onComboBoxIndexChangedCur(int)));
173
174 connect(ui->tableStiffness, &QTableWidget::itemChanged, this, [this](QTableWidgetItem* item)
175 {
176 if (item == ui->tableStiffness->item(2, 3))
177 {
178 onForceOffsetChanged(item->text());
179 }
180 });
181
182 ui->tablePosition->setItemDelegate(new TableDoubleDelegate);
183 ui->tableVelocity->setItemDelegate(new TableDoubleDelegate);
184 ui->tablePositionDir->setItemDelegate(new TableDoubleDelegate);

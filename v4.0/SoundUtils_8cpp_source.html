38bool utils::makeSawTooth(Sound& outSound,double duration_s, size_t channels, size_t sampleRate, double frequency, size_t amplitude)
39{
40 if (!prepareSound(outSound, duration_s, channels, sampleRate, frequency, amplitude)) { return false; }
41
42 size_t totalSamples = outSound.getSamples();
43
44 for (size_t i = 0; i < totalSamples; i++)
45 {
46 double t = static_cast<double>(i) / sampleRate;
47 double value = 2.0 * (t * frequency - floor(t * frequency + 0.5));
48 short sample = static_cast<short>(amplitude * value);
49 for (size_t ch = 0; ch < channels; ch++)
50 {
51 outSound.set(sample, i, ch);
52 }
53 }
54

99Sound utils::mix(const Sound& A, const Sound& B, double percentage)
100{
101 Sound result;
102
103 // Ensure percentage is clamped between 0 and 1
104 if (percentage < 0.0) percentage = 0.0;
105 if (percentage > 1.0) percentage = 1.0;
106
107 // Check for matching sample rates and number of channels
108 if (A.getFrequency() != B.getFrequency() || A.getChannels() != B.getChannels())
109 {
110 yError() << "Cannot mix sounds with different sample rates or channel counts.";
111 return result;
112 }
113
114 // Determine the length of the resulting sound
115 size_t maxSamples = std::max(A.getSamples(), B.getSamples());
116 size_t channels = A.getChannels();
117 double alpha = percentage;
118 double beta = 1.0 - percentage;
119
120 // Create the resulting sound
121 result.resize(maxSamples, channels);
122 result.setFrequency(A.getFrequency());
123
124 // Mix samples
125 for (size_t ch = 0; ch < channels; ++ch)
126 {
127 for (size_t i = 0; i < maxSamples; ++i)
128 {
129 double sampleA = (i < A.getSamples()) ? A.getSafe(i, ch) : 0.0;
130 double sampleB = (i < B.getSamples()) ? B.getSafe(i, ch) : 0.0;
131 yarp::sig::Sound::audio_sample sampleO = static_cast<yarp::sig::Sound::audio_sample>(alpha * sampleA + beta * sampleB);
132 result.setSafe(sampleO, i, ch);
133 }
134 }
135
136 return result;
137}

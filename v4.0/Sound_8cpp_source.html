83void Sound::overwrite(const Sound& alt, size_t offset, size_t len)
84{
85 if (alt.m_channels != m_channels)
86 {
87 yCError(SOUND, "unable to overwrite sounds with different number of channels!");
88 return;
89 }
90 if (alt.m_frequency != m_frequency)
91 {
92 yCError(SOUND, "unable to overwrite sounds with different sample rate!");
93 return;
94 }
95
96 size_t current_size = (*this).getSamples();
97 size_t alt_size = alt.getSamples();
98 if (len == 0) len = alt_size;
99 if ( offset + len > current_size)
100 {
101 len = current_size - offset;
102 }
103
104 for (size_t ch = 0; ch < m_channels; ch++)
105 {
106 size_t pdst = ch * this->getBytesPerSample() * this->m_samples + offset* this->getBytesPerSample();
107 unsigned char* dst = &this->getRawData()[pdst];
108 size_t psrc = ch * this->getBytesPerSample() * alt.m_samples;
109 unsigned char* src = &alt.getRawData()[psrc];
110 memcpy((void*) dst, (void*) src, len * this->m_bytesPerSample);
111 }
112
113 //invalidate all the markers
114 m_markers.clear();
115
116}

118Sound& Sound::operator += (const Sound& alt)
119{
120 if (alt.m_channels!= m_channels)
121 {
122 yCError(SOUND) << "unable to concatenate sounds with different number of channels!" <<
123 alt.m_channels << "!=" << m_channels;
124 return *this;
125 }
126 if (alt.m_frequency!= m_frequency)
127 {
128 yCError(SOUND) << "unable to concatenate sounds with different sample rate!" <<
129 alt.m_frequency << "!=" << m_frequency;
130 return *this;
131 }
132
133 Sound orig= *this;
134 this->resize(this->m_samples+alt.m_samples,m_channels);
135
136 unsigned char* orig_start = orig.getRawData();
137 unsigned char* alt_start = alt.getRawData();
138 unsigned char* pout = this->getRawData();
139 size_t orig_singlechannel_size = orig.getBytesPerSample() * orig.m_samples;
140 size_t alt_singlechannel_size = alt.getBytesPerSample() * alt.m_samples;
141
142 for (size_t ch=0; ch<m_channels; ch++)
143 {
144 size_t out1 = ch* this->getBytesPerSample() * this->m_samples;
145 size_t out2 = ch* this->getBytesPerSample() * this->m_samples + this->getBytesPerSample() * orig.m_samples;
146
147 size_t orig_pointer = ch * orig.getBytesPerSample() * orig.m_samples;
148 size_t alt_pointer = ch * orig.getBytesPerSample() * alt.m_samples;
149
150 memcpy((void *) &pout[out1], (void *) (orig_start + orig_pointer), orig_singlechannel_size);
151 memcpy((void *) &pout[out2], (void *) (alt_start + alt_pointer), alt_singlechannel_size);
152 }
153
154 //invalidate all the markers
155 m_markers.clear();
156
157 return *this;
158}

186Sound Sound::subSound(size_t first_sample, size_t len)
187{
188 Sound s;
189
190 // Nothing to copy
191 if (len == 0 || first_sample >= m_samples) {
192 s.setFrequency(m_frequency);
193 s.resize(0, m_channels);
194 return s;
195 }
196
197 size_t last_sample = std::min(first_sample + len, m_samples);
198 size_t out_len = last_sample - first_sample;
199
200 s.resize(out_len, this->m_channels);
201 s.setFrequency(this->m_frequency);
202
203 //safe implementation
204 size_t j=0;
205 for (size_t i=first_sample; i<last_sample; i++)
206 {
207 for (size_t c = 0; c < this->m_channels; c++) {
208 s.set(this->get(i, c), j, c);
209 }
210 j++;
211 }
212 return s;
213}

374bool Sound::read(ConnectionReader& connection)
375{
376 connection.convertTextMode();
377 SoundHeader header;
378 bool ok = connection.expectBlock((char*)&header, sizeof(header));
379 if (!ok) {
380 return false;
381 }
382
383 init(header.bytesPerSample);
384 m_frequency = header.frequency;
385 m_channels = header.channelsSize;
386 m_samples = header.samplesSize;
387
388 connection.expectInt32(); //dummy list
389 size_t data_size = connection.expectInt32(); //data size
390 if (data_size != m_channels * m_samples)
391 {
392 return false;
393 }
394
395 //sound data
396 auto* pp = ((std::vector<NetUint16>*)(implementation));
397 pp->resize(data_size);
398 for (size_t l = 0; l < data_size; l++)
399 {
400 pp->at(l) = connection.expectInt16();
401 }
402
403 connection.expectInt32(); //dummy list
404 size_t markersListLen = connection.expectInt32(); //markers size
405
406 //markers data
407 m_markers.clear();
408 for (size_t i = 0; i < markersListLen; i++)
409 {
410 int innerTag = connection.expectInt32();
411 if (innerTag != BOTTLE_TAG_LIST) {
412 return false;
413 }
414
415 int innerSize = connection.expectInt32();
416 if (innerSize != 4) {
417 return false;
418 }
419
420 std::string key = connection.expectString();
421
422 yarp::sig::SoundMarker marker;
423 marker.label = connection.expectString();
424 marker.channel = connection.expectInt32();
425 marker.sample_id = static_cast<size_t>(connection.expectInt64());
426
427 m_markers[key] = marker;
428 }
429
430
431 return true;
432}

435bool Sound::write(yarp::os::ConnectionWriter& connection) const
436{
437 SoundHeader header;
438 header.outerListTag = BOTTLE_TAG_LIST;
439 header.outerListLen = 6;
440 header.samplesSizeTag = BOTTLE_TAG_INT32; //1
441 header.samplesSize = m_samples;
442 header.bytesPerSampleTag = BOTTLE_TAG_INT32; //2
443 header.bytesPerSample = m_bytesPerSample;
444 header.channelsSizeTag = BOTTLE_TAG_INT32; //3
445 header.channelsSize = m_channels;
446 header.frequencyTag = BOTTLE_TAG_INT32; //4
447 header.frequency = m_frequency;
448
449 connection.appendBlock((char*)&header, sizeof(header));
450
451 auto* pp = ((std::vector<NetUint16>*)(implementation));
452 size_t siz = pp->size();
453 connection.appendInt32(BOTTLE_TAG_LIST+BOTTLE_TAG_INT16);
454 connection.appendInt32(siz);
455
456 //add the sound data
457 for (size_t l = 0; l < pp->size(); l++) {
458 connection.appendInt16(pp->at(l));
459 }
460
461 connection.appendInt32(BOTTLE_TAG_LIST);
462 connection.appendInt32(static_cast<int32_t>(m_markers.size()));
463
464 //add the markers data
465 for (const auto& it : m_markers) {
466 const std::string& key = it.first;
467 const yarp::sig::SoundMarker& marker = it.second;
468
469 connection.appendInt32(BOTTLE_TAG_LIST);
470 connection.appendInt32(4); // key + 3 fields of the struct
471
472 connection.appendString(key);
473 connection.appendString(marker.label);
474 connection.appendInt32(marker.channel);
475 connection.appendInt64(static_cast<int64_t>(marker.sample_id));
476 }
477
478 connection.convertTextMode();
479 return true;
480}

519Sound Sound::extractChannelAsSound(size_t channel_id) const
520{
521 Sound news(this->m_bytesPerSample);
522 news.setFrequency(this->m_frequency);
523 news.resize(this->m_samples, 1);
524
525 unsigned char* p_src = this->getRawData();
526 unsigned char* p_dst = news.getRawData();
527 size_t j = 0;
528 //pointer to the first element of the row of the matrix we need to copy
529 size_t first_sample = 0 + (this->m_samples * this->m_bytesPerSample)*channel_id;
530 //pointer to the last element of the row of the matrix we need to copy
531 size_t last_sample = first_sample + (this->m_samples * this->m_bytesPerSample);
532 for (auto i = first_sample; i < last_sample; i++)
533 {
534 p_dst[j++] = p_src[i];
535 }
536 return news;

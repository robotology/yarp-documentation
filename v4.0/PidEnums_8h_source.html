46inline size_t PidControlTypeEnum2Index(PidControlTypeEnum type)
47{
48 int32_t vocab = static_cast<yarp::conf::vocab32_t>(type);
49 std::string dc = yarp::os::Vocab32::decode(vocab);
50 if (dc[3] >= '0' && dc[3] <= '9')
51 return dc[3] - '1'; // returns 0 for '1', 1 for '2', etc.
52 return 0;
53}

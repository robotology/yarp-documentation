73bool FakeMotionControlMicro::extractGroup(Bottle &input, Bottle &out, const std::string &key1, const std::string &txt, int size)
74{
75 size++;
76 Bottle &tmp=input.findGroup(key1, txt);
77
78 if (tmp.isNull())
79 {
80 yCError(FAKEMOTIONCONTROLMICRO) << key1.c_str() << "parameter not found";
81 return false;
82 }
83
84 if(tmp.size()!=(size_t) size)
85 {
86 yCError(FAKEMOTIONCONTROLMICRO) << key1.c_str() << "incorrect number of entries";
87 return false;
88 }
89
90 out=tmp;
91 return true;
92}

229 //start the rateThread

231 if(!init)

248bool FakeMotionControlMicro::fromConfig(yarp::os::Searchable &config)
249{
250 size_t i;
251
252 // AxisMap
253 if (!m_GENERAL_AxisMap.empty())
254 {
255 for (i = 0; i < m_GENERAL_AxisMap.size(); i++) {
256 _axisMap[i] = m_GENERAL_AxisMap[i];
257 }
258 }
259 else
260 {
261 for (i = 0; i < _njoints; i++) {
262 _axisMap[i] = i;
263 }
264 }
265
266 // AxisName
267 if (!m_GENERAL_AxisName.empty())
268 {
269 for (i = 0; i < m_GENERAL_AxisName.size(); i++) {
270 _axisName[_axisMap[i]] = m_GENERAL_AxisName[i];
271 }
272 }
273 else
274 {
275 for (i = 0; i < _njoints; i++) {
276 _axisName[_axisMap[i]] = "joint" + toString(i);
277 }
278 }
279
280 // Axis Type
281 if (!m_GENERAL_AxisType.empty())
282 {
283 //beware: axis type has to be remapped here because they are not set using the toHw() helper function
284 for (i = 0; i < m_GENERAL_AxisType.size(); i++)
285 {
286 std::string typeString = m_GENERAL_AxisType[i];
287 if (typeString == "revolute") {
288 _jointType[_axisMap[i]] = JointTypeEnum::VOCAB_JOINTTYPE_REVOLUTE;
289 }
290 else if (typeString == "prismatic") {
291 _jointType[_axisMap[i]] = JointTypeEnum::VOCAB_JOINTTYPE_PRISMATIC;
292 }
293 else {
294 yCError(FAKEMOTIONCONTROLMICRO, "Unknown AxisType value %s!", typeString.c_str());
295 _jointType[_axisMap[i]] = JointTypeEnum::VOCAB_JOINTTYPE_UNKNOWN;
296 return false;
297 }
298 }
299 }
300 else
301 {
302 yCInfo(FAKEMOTIONCONTROLMICRO) << "Using default AxisType (revolute)";
303 for (i = 0; i < _njoints; i++) {
304 _jointType[_axisMap[i]] = JointTypeEnum::VOCAB_JOINTTYPE_REVOLUTE;
305 }
306 }
307
308
309
310
311
312
313// double tmp_A2E;
314 // Encoder scales
315 if(!m_GENERAL_Encoder.empty())
316 {
317 for (i = 0; i < m_GENERAL_Encoder.size(); i++) {
318 _angleToEncoder[i] = m_GENERAL_Encoder[i]; }
319 }
320 else
321 {
322 yCInfo(FAKEMOTIONCONTROLMICRO) << "Using default Encoder=1";
323 for (i = 0; i < _njoints; i++) {
324 _angleToEncoder[i] = 1; }
325 }
326
327 return true;
328}

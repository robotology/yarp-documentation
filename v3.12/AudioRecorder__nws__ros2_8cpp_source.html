63bool AudioRecorder_nws_ros2::open(yarp::os::Searchable &config)
64{
65 parseParams(config);
66 if (m_node_name[0] == '/') {
67 yCError(AUDIORECORDER_NWS_ROS2) << "node_name parameter cannot begin with '/'";
68 return false;
69 }
70 if (m_topic_name[0] != '/') {
71 yCError(AUDIORECORDER_NWS_ROS2) << "missing initial / in topic_name parameter";
72 return false;
73 }
74
75 rclcpp::NodeOptions node_options;
76 node_options.allow_undeclared_parameters(true);
77 node_options.automatically_declare_parameters_from_overrides(true);
78 if(m_namespace.empty()) {
79 m_node = NodeCreator::createNode(m_node_name, node_options);
80 } else {
81 m_node = NodeCreator::createNode(m_node_name, m_namespace, node_options);
82 }
83 if (m_node == nullptr) {
84 yCError(AUDIORECORDER_NWS_ROS2) << " opening " << m_node_name << " Node, check your yarp-ROS2 network configuration\n";
85 return false;
86 }
87 m_ros2Publisher_status = m_node->create_publisher<std_msgs::msg::Int16MultiArray>(m_topic_name, 10);
88
89 yCInfo(AUDIORECORDER_NWS_ROS2) << "Waiting for device to attach";
90 return true;

97void AudioRecorder_nws_ros2::run()
98{
99
100 if (m_audioRecorder_interface!=nullptr && m_ros2Publisher_status)
101 {
102 if(m_ros2Publisher_status->get_subscription_count()>0)
103 {
104 double synchronized_timestamp = 0;
105
106 yarp::sig::AudioBufferSize device_buffer_current_size;
107 yarp::sig::AudioBufferSize device_buffer_max_size;
108 bool isRecording=0;
109 m_audioRecorder_interface->getRecordingAudioBufferCurrentSize(device_buffer_current_size);
110 m_audioRecorder_interface->getRecordingAudioBufferMaxSize(device_buffer_max_size);
111 m_audioRecorder_interface->isRecording(isRecording);
112
113 if (std::isnan(synchronized_timestamp) == false)
114 {
115 m_timeStamp.update(synchronized_timestamp);
116 }
117 else
118 {
119 m_timeStamp.update(yarp::os::Time::now());
120 }
121
122 std_msgs::msg::Int16MultiArray statusMsg;
123 statusMsg.data.resize(3);
124 statusMsg.data[0] = static_cast<int>(isRecording);
125 statusMsg.data[1] = static_cast<int>(device_buffer_current_size.getBufferElements());
126 statusMsg.data[2] = static_cast<int>(device_buffer_max_size.getBufferElements());
127 m_ros2Publisher_status->publish(statusMsg);
128 }
129
130 } else{
131 yCError(AUDIORECORDER_NWS_ROS2) << "the interface is not valid";
132 }
133}

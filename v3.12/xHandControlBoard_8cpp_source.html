22bool yarp::dev::xHandControlBoard::open(yarp::os::Searchable& config)
23{
24 if (!parseParams(config))
25 {
26 yCError(CB) << "Failed to parse parameters for xHandControlBoard.";
27 return false;
28 }
29
30 // Params check
31 if(m_GENERAL_AxisName.size() != EV_HAND_JOINT_NUM)
32 {
33 yCError(CB) << "Number of Axis names provided (" << m_GENERAL_AxisName.size() << ") does not match the number of joints supported by the hand (" << EV_HAND_JOINT_NUM << ").";
34 return false;
35 }
36 if(m_GENERAL_AxisMap.size() != EV_HAND_JOINT_NUM)
37 {
38 yCError(CB) << "Number of Axis map entries provided (" << m_GENERAL_AxisMap.size() << ") does not match the number of joints supported by the hand (" << EV_HAND_JOINT_NUM << ").";
39 return false;
40 }
41 if(m_GENERAL_AxisType.size() != EV_HAND_JOINT_NUM)
42 {
43 yCError(CB) << "Number of Axis types provided (" << m_GENERAL_AxisType.size() << ") does not match the number of joints supported by the hand (" << EV_HAND_JOINT_NUM << ").";
44 return false;
45 }
46 if(m_LIMITS_jntPosMax.size() != EV_HAND_JOINT_NUM)
47 {
48 yCError(CB) << "Number of joint position max limits provided (" << m_LIMITS_jntPosMax.size() << ") does not match the number of joints supported by the hand (" << EV_HAND_JOINT_NUM << ").";
49 return false;
50 }
51 if(m_LIMITS_jntPosMin.size() != EV_HAND_JOINT_NUM)
52 {
53 yCError(CB) << "Number of joint position min limits provided (" << m_LIMITS_jntPosMin.size() << ") does not match the number of joints supported by the hand (" << EV_HAND_JOINT_NUM << ").";
54 return false;
55 }
56 if(m_LIMITS_jntVelMin.size() != EV_HAND_JOINT_NUM)
57 {
58 yCError(CB) << "Number of joint velocity min limits provided (" << m_LIMITS_jntVelMin.size() << ") does not match the number of joints supported by the hand (" << EV_HAND_JOINT_NUM << ").";
59 return false;
60 }
61 if(m_LIMITS_jntVelMax.size() != EV_HAND_JOINT_NUM)
62 {
63 yCError(CB) << "Number of joint velocity max limits provided (" << m_LIMITS_jntVelMax.size() << ") does not match the number of joints supported by the hand (" << EV_HAND_JOINT_NUM << ").";
64 return false;
65 }
66 for(size_t i=0; i<EV_HAND_JOINT_NUM; i++)
67 {
68 if(m_LIMITS_jntPosMin[i] >= m_LIMITS_jntPosMax[i])
69 {
70 yCError(CB) << "Joint position limits for joint " << i << " are invalid: min limit (" << m_LIMITS_jntPosMin[i] << ") >= max limit (" << m_LIMITS_jntPosMax[i] << ").";
71 return false;
72 }
73 }
74 for(size_t i=0; i<EV_HAND_JOINT_NUM; i++){
75 if(m_LIMITS_jntVelMin[i] >= m_LIMITS_jntVelMax[i]){
76 yCError(CB) << "Joint velocity limits for joint " << i << " are invalid: min limit (" << m_LIMITS_jntVelMin[i] << ") >= max limit (" << m_LIMITS_jntVelMax[i] << ").";
77 return false;
78 }
79 }
80
81 // Looking for available devices
82 std::vector<std::string> ifnames = m_XHCtrl.enumerate_devices(m_connection_type);
83 if (ifnames.empty()) {
84 yCError(CB) << "No devices found for connection type: " << m_connection_type;
85 return false;
86 }
87
88 std::string debugMsg{};
89
90 xhand_control::ErrorStruct ret;
91 std::vector<std::string>::iterator ifnames_it{};
92 if (m_connection_type == "RS485"){
93
94 ifnames_it = std::find(ifnames.begin(), ifnames.end(), m_RS485_port);
95
96 if (ifnames_it == ifnames.end()){
97 for (const auto& ifname : ifnames) {debugMsg+=ifname + " ";}
98 yCError(CB) << "Specified RS485 port" << m_RS485_port << "not found among available devices: " << debugMsg;
99 return false;
100 }
101
102 ret = m_XHCtrl.open_serial(m_RS485_port, m_RS485_baudrate);
103 }
104 else if (m_connection_type == "EtherCAT"){
105 ifnames_it = std::find(ifnames.begin(), ifnames.end(), m_ETHERCAT_eth_ifname);
106
107 if (ifnames_it == ifnames.end()){
108 for (const auto& ifname : ifnames) {debugMsg+=ifname + " ";}
109 yCError(CB) << "Specified EtherCAT interface" << m_ETHERCAT_eth_ifname << "not found among available devices: " << debugMsg;
110 return false;
111 }
112
113 ret = m_XHCtrl.open_ethercat(m_ETHERCAT_eth_ifname);
114 }
115 else{
116 yCError(CB) << "Unknown connection type: " << m_connection_type;
117 return false;
118 }
119
120 if (!ret){
121 yCError(CB) << "Failed to open " << m_connection_type;
122 printErrorStruct(ret);
123 return false;
124 }
125
126 yDebug() << "Successfully connected to " << (m_connection_type == "RS485" ? m_RS485_port : m_ETHERCAT_eth_ifname);
127
128 std::vector<uint8_t> hand_list = m_XHCtrl.list_hands_id();
129 if (hand_list.empty()){
130 yCError(CB) << "No hands found on the network.";
131 return false;
132 }
133
134 debugMsg.clear();
135 for (const auto& hand : hand_list) {debugMsg+=std::to_string(hand) + " ";}
136 yCDebug(CB) << "Hand IDs:" << debugMsg;
137 m_id = hand_list[0]; //<---------------------------------------------------------------- For now, we support only one hand !!!
138 yCDebug(CB) << "Using hand with ID:" << static_cast<int>(m_id);
139
140 //start the rateThread
141 bool init = this->yarp::os::PeriodicThread::start();
142 if(!init)
143 {
144 yCError(CB) << "Unable to start xHandControlBoard periodic thread"; ;
145 return false;
146 }
147
148 // Give some time to the hand to be ready
149 double delay = 1.0;
150 yCInfo(CB) << "Waiting " << delay << " s for the hand to be ready...";
151 yarp::os::Time::delay(delay);
152
153 m_ppids = new yarp::dev::Pid[EV_HAND_JOINT_NUM];
154 m_vpids = new yarp::dev::Pid[EV_HAND_JOINT_NUM];
155 m_cpids = new yarp::dev::Pid[EV_HAND_JOINT_NUM];
156 m_tpids = new yarp::dev::Pid[EV_HAND_JOINT_NUM];
157 m_controlModes = new int [EV_HAND_JOINT_NUM];
158
159 // Initialize hand command structure
160 memset(&m_handCommand, 0, sizeof(m_handCommand));
161 for(uint16_t i = 0; i < EV_HAND_JOINT_NUM; i++){
162 m_handCommand.finger_command[i].id = i;
163 double enc = 0.0;
164 getEncoder(i, &enc);
165 m_handCommand.finger_command[i].position = static_cast<float>(deg2rad(enc));
166 m_handCommand.finger_command[i].kp = m_PID_kp;
167 m_handCommand.finger_command[i].ki = m_PID_ki;
168 m_handCommand.finger_command[i].kd = m_PID_kd;
169 m_handCommand.finger_command[i].mode = m_MODE_mode;
170 m_handCommand.finger_command[i].tor_max = m_TORQUE_tor_max;
171 }
172
173 sendCommandToHand(m_id, m_handCommand);
174
175 for (size_t i=0; i< EV_HAND_JOINT_NUM; i++)setControlMode(i,VOCAB_CM_POSITION);
176 yarp::dev::Pid positionPidInit(m_PID_kp, m_PID_kd, m_PID_ki, 0.0, 1.0, 1.0);
177 std::vector<yarp::dev::Pid> positionPids(EV_HAND_JOINT_NUM, positionPidInit);
178 setPids(VOCAB_PIDTYPE_POSITION, positionPids.data());
179
180 return true;
181}

183bool yarp::dev::xHandControlBoard::close()
184{
185 if (this->isRunning()){
186 std::lock_guard lock(m_mutex);
187 this->yarp::os::PeriodicThread::askToStop();
188 }
189
190 m_XHCtrl.close_device();
191
192 delete [] m_ppids;
193 delete [] m_vpids;
194 delete [] m_cpids;
195 delete [] m_tpids;
196 delete[] m_controlModes;
197
198 return true;
199}

209bool yarp::dev::xHandControlBoard::setPid(const yarp::dev::PidControlTypeEnum& pidtype, int j, const yarp::dev::Pid& p)
210{
211 switch (pidtype)
212 {
213 case VOCAB_PIDTYPE_POSITION:
214 m_ppids[j] = p;
215 m_handCommand.finger_command[j].kp = p.kp;
216 m_handCommand.finger_command[j].ki = p.ki;
217 m_handCommand.finger_command[j].kd = p.kd;
218 sendCommandToHand(m_id, m_handCommand);
219 break;
220 case VOCAB_PIDTYPE_VELOCITY:
221 m_vpids[j] = p;
222 break;
223 case VOCAB_PIDTYPE_CURRENT:
224 m_cpids[j] = p;
225 break;
226 case VOCAB_PIDTYPE_TORQUE:
227 m_tpids[j] = p;
228 break;
229 default:
230 return false;
231 break;
232 }
233 return true;
234}

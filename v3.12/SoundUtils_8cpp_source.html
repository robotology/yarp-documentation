19{
20 if (duration_s <= 0 || channels == 0 || sampleRate == 0)
21 {
22 yError() << "yarp::sig::utils::makeTone: invalid parameters";
23 return false;
24 }
25
26 // Sound parameters
27 size_t totalSamples = sampleRate * duration_s;
28
29 // Generate sine wave parameters
30 double frequency = 440.0; // A4 standard tone
31 double amplitude = 30000; // max ~32767 for 16-bit
32 double twoPiF = 2.0 * M_PI * frequency;
33
34 // Create the sound object
35 outSound = yarp::sig::Sound();
36 outSound.resize(totalSamples, channels);
37 outSound.setFrequency(sampleRate);
38
39 // Fill the sound buffer with a sine wave
40 for (size_t i = 0; i < totalSamples; i++)
41 {
42 double t = static_cast<double>(i) / sampleRate;
43 short sample = static_cast<short>(amplitude * sin(twoPiF * t));
44 outSound.set(sample, i, 0);
45 }
46 return true;
47}

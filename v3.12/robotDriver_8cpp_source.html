52bool robotDriver::init()
53{
54 if (!drv_ll)
55 {
56 yError() << "Invalid pointer to Polydriver!";
57 return false;
58 }
59
60 bool ok = true;
61 if (drv_ll->isValid())
62 {
63 ok = drv_ll->view(ipos_ll) && drv_ll->view(iposdir_ll) && drv_ll->view(ienc_ll) &&
64 drv_ll->view(ipid_ll) && drv_ll->view(imotenc_ll) && drv_ll->view(icmd_ll);
65 }
66 else
67 {
68 ok = false;
69 }
70
71 if (!ok)
72 {
73 yError() << "Unable to view some interface!";
74 return false;
75 }
76
77 //get the number of the joints
78 ok &= ienc_ll->getAxes(&n_joints);
79
80 //set the trajectory duration for the first movement
81 m_trajectoryTime.resize(n_joints,4.0);
82
83 //set the initial reference speeds to 20
84 for (int i = 0; i < n_joints; i++)
85 {
86 m_stored_speed.push_back(20.0);
87 }
88 ok &= loadRefVelocities();
89
90 return ok;
91}

121bool robotDriver::positionMove(int j, double v)
122{
123 if (!ipos_ll) return false;
124 bool ret = true;
125
126 // set ref speed to each joint so that the motion is completed in
127 // the requested amount of time
128 if (1)
129 {
130 double enc = 0;
131 ret &= ienc_ll->getEncoder(j, &enc);
132 double diff = fabs(v - enc);
133 double speed = diff/m_trajectoryTime[j];
134 ipos_ll->setRefSpeed(j, speed);
135 }
136
137 ret &= ipos_ll->positionMove(j, v);
138
139 return ret;
140}

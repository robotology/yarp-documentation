25inline yarp::sig::Pose6D toYarpPose(const simulated_world_nws_ros2_msgs::msg::Pose6D& msg)
26{
27 yarp::sig::Pose6D pose;
28 pose.x = msg.x;
29 pose.y = msg.y;
30 pose.z = msg.z;
31 pose.roll = msg.roll;
32 pose.pitch = msg.pitch;
33 pose.yaw = msg.yaw;
34 return pose;
35}

61bool SimulatedWorld_nws_ros2::attach(yarp::dev::PolyDriver* driver)
62{
63 yCInfo(SIMULATEDWORLD_NWS_ROS2, "Attaching to driver");
64 if (driver && driver->isValid())
65 {
66 driver->view(m_iSim);
67 }
68
69 if (m_iSim == nullptr)
70 {
71 yCError(SIMULATEDWORLD_NWS_ROS2, "Subdevice passed to attach method is invalid");
72 return false;
73 }
74
75 return true;
76}

84bool SimulatedWorld_nws_ros2::open(yarp::os::Searchable& config)
85{
86 yCInfo(SIMULATEDWORLD_NWS_ROS2, "Opening SimulatedWorld_nws_ros2");
87 parseParams(config);
88
89 if (!m_node_name.empty() && m_node_name[0] == '/')
90 {
91 yCError(SIMULATEDWORLD_NWS_ROS2) << "node_name cannot begin with an initial /";
92 return false;
93 }
94
95 rclcpp::NodeOptions node_options;
96 node_options.allow_undeclared_parameters(true);
97 node_options.automatically_declare_parameters_from_overrides(true);
98 if (m_namespace.empty()) {
99 m_node = NodeCreator::createNode(m_node_name, node_options);
100 } else {
101 m_node = NodeCreator::createNode(m_node_name, m_namespace, node_options);
102 }
103 if (m_node == nullptr) {
104 yCError(SIMULATEDWORLD_NWS_ROS2) << " opening " << m_node_name << " Node, check your yarp-ROS2 network configuration\n";
105 return false;
106 }
107
108 m_srv_makeSphere = m_node->create_service<simulated_world_nws_ros2_msgs::srv::MakeSphere>(
109 m_makesphere_servicename, std::bind(&SimulatedWorld_nws_ros2::makeSphereCallback, this, _1, _2, _3));
110 m_srv_makeBox = m_node->create_service<simulated_world_nws_ros2_msgs::srv::MakeBox>(
111 m_makebox_servicename, std::bind(&SimulatedWorld_nws_ros2::makeBoxCallback, this, _1, _2, _3));
112 m_srv_makeCylinder = m_node->create_service<simulated_world_nws_ros2_msgs::srv::MakeCylinder>(
113 m_makecylinder_servicename, std::bind(&SimulatedWorld_nws_ros2::makeCylinderCallback, this, _1, _2, _3));
114 m_srv_makeFrame = m_node->create_service<simulated_world_nws_ros2_msgs::srv::MakeFrame>(
115 m_makeframe_servicename, std::bind(&SimulatedWorld_nws_ros2::makeFrameCallback, this, _1, _2, _3));
116 m_srv_makeModel = m_node->create_service<simulated_world_nws_ros2_msgs::srv::MakeModel>(
117 m_makemodel_servicename, std::bind(&SimulatedWorld_nws_ros2::makeModelCallback, this, _1, _2, _3));
118 m_srv_makeActor = m_node->create_service<simulated_world_nws_ros2_msgs::srv::MakeActor>(
119 m_makeactor_servicename, std::bind(&SimulatedWorld_nws_ros2::makeActorCallback, this, _1, _2, _3));
120 m_srv_changeColor = m_node->create_service<simulated_world_nws_ros2_msgs::srv::ChangeColor>(
121 m_changecolor_servicename, std::bind(&SimulatedWorld_nws_ros2::changeColorCallback, this, _1, _2, _3));
122 m_srv_setPose = m_node->create_service<simulated_world_nws_ros2_msgs::srv::SetPose>(
123 m_setpose_servicename, std::bind(&SimulatedWorld_nws_ros2::setPoseCallback, this, _1, _2, _3));
124 m_srv_enableGravity = m_node->create_service<simulated_world_nws_ros2_msgs::srv::EnableGravity>(
125 m_enablegravity_servicename, std::bind(&SimulatedWorld_nws_ros2::enableGravityCallback, this, _1, _2, _3));
126 m_srv_enableCollision = m_node->create_service<simulated_world_nws_ros2_msgs::srv::EnableCollision>(
127 m_enablecollision_servicename, std::bind(&SimulatedWorld_nws_ros2::enableCollisionCallback, this, _1, _2, _3));
128 m_srv_getPose = m_node->create_service<simulated_world_nws_ros2_msgs::srv::GetPose>(
129 m_getpose_servicename, std::bind(&SimulatedWorld_nws_ros2::getPoseCallback, this, _1, _2, _3));
130 m_srv_deleteObject = m_node->create_service<simulated_world_nws_ros2_msgs::srv::DeleteObject>(
131 m_deleteobject_servicename, std::bind(&SimulatedWorld_nws_ros2::deleteObjectCallback, this, _1, _2, _3));
132 m_srv_deleteAll = m_node->create_service<simulated_world_nws_ros2_msgs::srv::DeleteAll>(
133 m_deleteall_servicename, std::bind(&SimulatedWorld_nws_ros2::deleteAllCallback, this, _1, _2, _3));
134 m_srv_getList = m_node->create_service<simulated_world_nws_ros2_msgs::srv::GetList>(
135 m_getlist_servicename, std::bind(&SimulatedWorld_nws_ros2::getListCallback, this, _1, _2, _3));
136 m_srv_attach = m_node->create_service<simulated_world_nws_ros2_msgs::srv::Attach>(
137 m_attach_servicename, std::bind(&SimulatedWorld_nws_ros2::attachCallback, this, _1, _2, _3));
138 m_srv_detach = m_node->create_service<simulated_world_nws_ros2_msgs::srv::Detach>(
139 m_detach_servicename, std::bind(&SimulatedWorld_nws_ros2::detachCallback, this, _1, _2, _3));
140 m_srv_rename = m_node->create_service<simulated_world_nws_ros2_msgs::srv::Rename>(
141 m_rename_servicename, std::bind(&SimulatedWorld_nws_ros2::renameCallback, this, _1, _2, _3));
142
143 // Start the spinner
144 m_spinner = new Ros2Spinner(m_node);
145 m_spinner->start();
146
147 yCInfo(SIMULATEDWORLD_NWS_ROS2) << "Waiting for device to attach";
148 return true;
149}

151bool SimulatedWorld_nws_ros2::close()
152{
153 yCTrace(SIMULATEDWORLD_NWS_ROS2, "Close");
154
155 // Stop and delete the spinner
156 rclcpp::shutdown();
157 if (m_spinner) {
158 m_spinner->stop();
159 delete m_spinner;
160 m_spinner = nullptr;
161 }
162
163 detach();
164 return true;
165}

167void SimulatedWorld_nws_ros2::makeSphereCallback(
168 const std::shared_ptr<rmw_request_id_t> request_header,
169 const std::shared_ptr<simulated_world_nws_ros2_msgs::srv::MakeSphere::Request> request,
170 std::shared_ptr<simulated_world_nws_ros2_msgs::srv::MakeSphere::Response> response)
171{
172 (void)request_header;
173 std::lock_guard<std::mutex> lock(m_mutex);
174 if (!m_iSim)
175 {
176 response->success = false;
177 return;
178 }
179 if (request->frame_name == "None")
180 {
181 request->frame_name = "";
182 yCWarning(SIMULATEDWORLD_NWS_ROS2) << "Frame name was 'None', setting to empty string";
183 }
184
185 if (request->id == "None" || request->id.empty())
186 {
187 yCError(SIMULATEDWORLD_NWS_ROS2) << "Invalid request ID";
188 response->success = false;
189 return;
190 }
191
192 auto ret = m_iSim->makeSphere(request->id, request->radius, toYarpPose(request->pose),
193 toYarpColor(request->color), request->frame_name,
194 request->gravity_enable, request->collision_enable);
195 response->success = static_cast<bool>(ret);
196}

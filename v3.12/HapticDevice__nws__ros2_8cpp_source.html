173void HapticDevice_nws_ros2::getTransformationCallback(
174 const std::shared_ptr<yarp_control_msgs::srv::GetTransformation::Request> request,
175 std::shared_ptr<yarp_control_msgs::srv::GetTransformation::Response> response)
176{
177 if (!iHapticDevice)
178 {
179 yCError(HAPTICDEVICE_NWS_ROS2) << "IHapticDevice interface not available";
180 response->success = false;
181 response->response = "IHapticDevice interface not available";
182 return;
183 }
184 // Transform
185 yarp::sig::Matrix trans;
186 if (!iHapticDevice->getTransformation(trans) || trans.rows() < 3 || trans.cols() < 4)
187 {
188 yCError(HAPTICDEVICE_NWS_ROS2) << "Failed to get transformation";
189 response->success = false;
190 response->response = "Failed to get transformation";
191 return;
192 }
193
194 geometry_msgs::msg::Transform trans_msg;
195 trans_msg.translation.x = trans(0, 3);
196 trans_msg.translation.y = trans(1, 3);
197 trans_msg.translation.z = trans(2, 3);
198
199 yarp::math::Quaternion trans_q;
200 yarp::sig::Matrix R = trans.submatrix(0,2,0,2);
201 trans_q.fromRotationMatrix(R);
202 trans_msg.rotation.x = trans_q.x();
203 trans_msg.rotation.y = trans_q.y();
204 trans_msg.rotation.z = trans_q.z();
205 trans_msg.rotation.w = trans_q.w();
206
207 response->transform = trans_msg;
208 response->success = true;
209}

211void HapticDevice_nws_ros2::setTransformationCallback(
212 const yarp_control_msgs::srv::SetTransformation::Request::SharedPtr request,
213 std::shared_ptr<yarp_control_msgs::srv::SetTransformation::Response> response)
214{
215 if (!request)
216 {
217 yCError(HAPTICDEVICE_NWS_ROS2) << "Received null request in set_transformation callback";
218 response->success = false;
219 response->response = "Received null request";
220 return;
221 }
222
223 if (iHapticDevice == nullptr)
224 {
225 yCError(HAPTICDEVICE_NWS_ROS2) << "IHapticDevice interface not available in set_transformation callback";
226 return;
227 }
228
229 yarp::sig::Matrix T(4, 4);
230 T.eye();
231
232 const auto& q = request->transform.rotation;
233 const auto& t = request->transform.translation;
234
235 // Quaternion to rotation matrix
236 yarp::math::Quaternion yarp_q;
237 yarp_q.x() = q.x;
238 yarp_q.y() = q.y;
239 yarp_q.z() = q.z;
240 yarp_q.w() = q.w;
241 T.setSubmatrix(yarp_q.toRotationMatrix3x3(), 0, 0);
242 T(0, 3) = t.x;
243 T(1, 3) = t.y;
244 T(2, 3) = t.z;
245
246 if (!iHapticDevice->setTransformation(T))
247 {
248 yCError(HAPTICDEVICE_NWS_ROS2) << "setTransformation() failed";
249 response->response = "setTransformation failed";
250 response->success = false;
251 return;
252 }
253
254 response->success = true;
255}

313rcl_interfaces::msg::SetParametersResult HapticDevice_nws_ros2::parameter_callback(
314 const std::vector<rclcpp::Parameter> &parameters)
315{
316 rcl_interfaces::msg::SetParametersResult result;
317 result.successful = true;
318
319 for (const auto & param : parameters)
320 {
321 if (param.get_name() == "mode_feedback")
322 {
323 std::string new_mode = param.value_to_string();
324 if (mode_feedback != new_mode)
325 {
326 bool success = forceModeCallback(new_mode);
327 if (success)
328 {
329 mode_feedback = new_mode;
330 }
331 else
332 {
333 result.successful = false;
334 result.reason = "Failed to set mode_feedback to " + new_mode;
335 yCError(HAPTICDEVICE_NWS_ROS2) << result.reason;
336 }
337 }
338 }
339 else
340 {
341 yCWarning(HAPTICDEVICE_NWS_ROS2, "Parameter '%s' is not recognized. Ignoring.", param.get_name().c_str());
342 }
343 }
344
345 return result;
346}

348bool HapticDevice_nws_ros2::forceModeCallback(const std::string& mode)
349{
350 if (!iHapticDevice)
351 {
352 yCError(HAPTICDEVICE_NWS_ROS2) << "IHapticDevice interface not available";
353 return false;
354 }
355
356 if (mode != "cartesian" && mode != "joint")
357 {
358 yCError(HAPTICDEVICE_NWS_ROS2, "Invalid value for mode_feedback: %s. Allowed values are 'cartesian' or 'joint'", mode.c_str());
359 return false;
360 }
361
362 bool success = (mode == "cartesian")
363 ? iHapticDevice->setCartesianForceMode()
364 : iHapticDevice->setJointTorqueMode();
365
366 if (!success)
367 {
368 yCError(HAPTICDEVICE_NWS_ROS2) << "Failed to set haptic mode";
369 return false;
370 }
371
372 bool is_force_mode;
373 if (!iHapticDevice->isCartesianForceModeEnabled(is_force_mode))
374 {
375 yCError(HAPTICDEVICE_NWS_ROS2) << "Failed to get haptic mode";
376 return false;
377 }
378
379 bool expected_force_mode = (mode == "cartesian");
380 if (is_force_mode != expected_force_mode)
381 {
382 yCError(HAPTICDEVICE_NWS_ROS2) << "Switched to" << mode << "mode failed";
383 return false;
384 }
385
386 yCInfo(HAPTICDEVICE_NWS_ROS2) << "Switched to" << mode << "mode successfully";
387 return true;
388}

391bool HapticDevice_nws_ros2::attach(yarp::dev::PolyDriver * poly)
392{
393 if (poly == nullptr)
394 {
395 yCError(HAPTICDEVICE_NWS_ROS2) << "attach() received nullptr";
396 return false;
397 }
398
399 if (!poly->isValid())
400 {
401 yCError(HAPTICDEVICE_NWS_ROS2) << "attach() received invalid PolyDriver";
402 return false;
403 }
404
405 if (!poly->view(iHapticDevice))
406 {
407 yCError(HAPTICDEVICE_NWS_ROS2) << "attach() failed to obtain iHapticDevice interface";
408 return false;
409 }
410
411 if (!publisherConfigureRosHandlers() || !subscriberConfigureRosHandlers()
412 || !servicesConfigureRosHandlers() || !configureRosParameters())
413 {
414 yCError(HAPTICDEVICE_NWS_ROS2) << "Failed to configure ROS handlers";
415 destroyRosHandlers();
416 return false;
417 }
418
419 yarp::os::PeriodicThread::setPeriod(m_period);
420 if (!yarp::os::PeriodicThread::start())
421 {
422 yCError(HAPTICDEVICE_NWS_ROS2) << "Error starting thread";
423 return false;
424 }
425
426 if (m_spinner)
427 {
428 if (!m_spinner->start())
429 {
430 yCError(HAPTICDEVICE_NWS_ROS2) << "Error starting the spinner";
431 return false;
432 }
433 }
434
435 return true;
436}

510void HapticDevice_nws_ros2::run()
511{
512 if (iHapticDevice != nullptr)
513 {
514 // Pose
515 yarp::sig::Vector pos;
516 yarp::sig::Vector rpy;
517 if (iHapticDevice->getPosition(pos) && pos.size() >= 3 &&
518 iHapticDevice->getOrientation(rpy) && rpy.size() >= 3)
519 {
520 geometry_msgs::msg::Pose pose_msg;
521 pose_msg.position.x = pos[0];
522 pose_msg.position.y = pos[1];
523 pose_msg.position.z = pos[2];
524
525 yarp::sig::Matrix dcm = yarp::math::rpy2dcm(rpy);
526 yarp::math::Quaternion pose_q;
527 pose_q.fromRotationMatrix(dcm);
528 pose_msg.orientation.x = pose_q.x();
529 pose_msg.orientation.y = pose_q.y();
530 pose_msg.orientation.z = pose_q.z();
531 pose_msg.orientation.w = pose_q.w();
532 m_stat->publish(pose_msg);
533 }
534
535 // Buttons
536 yarp::sig::Vector buttons;
537 if (iHapticDevice->getButtons(buttons) && buttons.size() >= 1)
538 {
539 std_msgs::msg::Int32MultiArray btn_msg;
540 for (size_t i = 0; i < buttons.size(); ++i)
541 {
542 btn_msg.data.push_back(static_cast<int>(buttons[i]));
543 }
544 m_buttons->publish(btn_msg);
545 }
546 }

36bool Llama2Device::open(yarp::os::Searchable &config)
37{
38 if (!parseParams(config)) { return false; }
39 // initialize LLM
40 std::string model_path;
41 bool has_model_context{m_model_context != ""};
42 yarp::os::ResourceFinder resource_finder;
43 if(has_model_context)
44 {
45 resource_finder.setDefaultContext(m_model_context);
46 model_path = resource_finder.findFile(m_model_name);
47 }
48 else
49 {
50 model_path = m_model_name;
51 }
52 init_LLM(model_path);
53 bool has_prompt_file{m_prompt_file != ""};
54 bool has_prompt_context{m_prompt_context != ""};
55 if(has_prompt_context)
56 {
57 resource_finder.setDefaultContext(m_prompt_context);
58 }
59
60 if(has_prompt_file)
61 {
62 std::string prompt_file_fullpath = resource_finder.findFile(m_prompt_file);
63 auto stream = std::ifstream(prompt_file_fullpath);
64 if (!stream)
65 {
66 yCWarning(LLAMA2DEVICE) << "File:" << prompt_file_fullpath << "does not exist or path is invalid";
67 }
68 else
69 {
70 std::ostringstream sstr;
71 sstr << stream.rdbuf(); //Reads the entire file into the stringstream
72 if(!setPrompt(sstr.str()))
73 {
74 return false;
75 }
76 }
77 }
78
79 return true;
80}

91yarp::dev::ReturnValue Llama2Device::ask(const std::string &question, yarp::dev::LLM_Message &oAnswer)
92{
93 nlohmann::json request_json;
94 request_json["model"] = "default";
95 request_json["max_tokens"] = m_npredict;
96
97 // Pass the conversation to the model
98 nlohmann::json messages_array = nlohmann::json::array();
99 for (const auto& msg : m_conversation) {
100 messages_array.push_back({{"role", msg.type}, {"content", msg.content}});
101 }
102
103 // Add new question to the conversation
104 messages_array.push_back({{"role", "user"}, {"content", question}});
105
106 // Insert array messages into the request
107 request_json["messages"] = messages_array;
108
109 // Convert JSON to string and send the request to the model
110 std::string risposta = sendPostRequest(request_json.dump());
111
112 // extract model's answer
113 std::string final_output = extractContent(risposta);
114
115 // add question to the conversation
116 yarp::dev::LLM_Message message;
117 message.type = "user";
118 message.content = question;
119 m_conversation.push_back(message);
120
121 // variable used to store the complete answer generated by the model
122 final_output = extractContent(risposta);
123
124 // add model answer to the conversation
125 message.type = "assitant";
126 message.content = final_output;
127 m_conversation.push_back(message);
128
129 // write model answer inside oAnswer
130 oAnswer.type = "assistant";
131 oAnswer.content = final_output;
132
133 return yarp::dev::ReturnValue::return_code::return_value_ok;
134}

137yarp::dev::ReturnValue Llama2Device::setPrompt(const std::string &prompt)
138{
139 // check if a prompt is already set
140 if(prompt_set == true){
141 yCError(LLAMA2DEVICE)<< "A prompt is already set. You must delete conversation first";
142 return yarp::dev::ReturnValue::return_code::return_value_error_generic;
143 }
144 // if the prompt is not already set, set the prompt
145 prompt_set = true;
146 first_prompt_set = true;
147
148 m_prompt.type = "system";
149 m_prompt.content = prompt;
150
151 // add prompt to the conversation
152 yarp::dev::LLM_Message message;
153 message.type = "system";
154 message.content = prompt;
155 m_conversation.push_back(message);
156
157 return yarp::dev::ReturnValue::return_code::return_value_ok;

175yarp::dev::ReturnValue Llama2Device::getConversation(std::vector<yarp::dev::LLM_Message> &oConversation)
176{
177 // check if the conversation is empty
178 if(m_conversation.empty()){
179 yCInfo(LLAMA2DEVICE) << "The conversation is empty";
180 return yarp::dev::ReturnValue::return_code::return_value_error_generic;
181 }
182
183 // if the conversation is not empty, print it
184 for (const auto& msg : oConversation) {
185 yCInfo(LLAMA2DEVICE) << "Role: " << msg.type;
186 yCInfo(LLAMA2DEVICE) << "Content: " << msg.content;
187 yCInfo(LLAMA2DEVICE) << "-------------------";
188 }
189
190 // set the value of the conversation
191 oConversation = m_conversation;
192 return yarp::dev::ReturnValue::return_code::return_value_ok;
193}

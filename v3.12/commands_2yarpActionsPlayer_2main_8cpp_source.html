175 bool loadConfiguration(std::string filename, double resample_period)
176 {
177 yarp::os::Property p;
178 if (!p.fromConfigFile(filename))
179 {
180 yError() << "Unable to read configuration file!";
181 return false;
182 }
183
184 //load the clients
185 yarp::os::Bottle& bot_clients = p.findGroup("CLIENTS");
186 if (bot_clients.size() == 0)
187 {
188 yError() << "Unable to read CLIENTS section";
189 return false;
190 }
191 size_t num_of_clients = bot_clients.size();
192 for (size_t i = 1; i < num_of_clients; i++)
193 {
194 yDebug() << bot_clients.get(i).toString();
195 yarp::os::Bottle* bot_client_elem = bot_clients.get(i).asList();
196 if (bot_client_elem == nullptr)
197 {
198 yError() << "Invalid entry in CLIENTS section";
199 return false;
200 }
201 size_t num_of_client_elems = bot_client_elem->size();
202 if (num_of_client_elems == 2)
203 {
204 //parse a line of the CLIENTS section
205 std::string client_name = bot_client_elem->get(0).toString();
206 std::string remoteControlBoards = bot_client_elem->get(1).toString();
207 if (m_robotClients[client_name] != nullptr)
208 {
209 yError() << "Duplicate client entry: " << client_name;
210 return false;
211 }
212 yarp::dev::PolyDriver* pd_client = new yarp::dev::PolyDriver;
213 yarp::os::Property clientoptions;
214 clientoptions.put("device", "remote_controlboard");
215 clientoptions.put("remote", remoteControlBoards);
216 clientoptions.put("local", m_name + "/client/" + client_name);
217 clientoptions.put("carrier", "fast_tcp");
218
219 //configure the client
220 bool rob_ok = pd_client->open(clientoptions);
221 if (!rob_ok)
222 {
223 yError() << "Unable to initialize client: " << client_name;
224 delete pd_client;
225 return false;
226 }
227 m_robotClients[client_name] = pd_client;
228 }
229 else
230 {
231 yError() << "Invalid number of entries in CLIENTS section";
232 return false;
233 }
234 }
235
236 //load the remappers
237 yarp::os::Bottle& bot_cont = p.findGroup("REMAPPERS");
238 if (bot_cont.size() == 0)
239 {
240 yError() << "Unable to read REMAPPERS section";
241 return false;
242 }
243 size_t num_of_controllers = bot_cont.size();
244 for (size_t i = 1; i < num_of_controllers; i++)
245 {
246 yDebug() << bot_cont.get(i).toString();
247 yarp::os::Bottle* bot_cont_elem = bot_cont.get(i).asList();
248 if (bot_cont_elem == nullptr)
249 {
250 yError() << "Invalid entry in REMAPPERS section";
251 return false;
252 }
253 size_t num_of_celems = bot_cont_elem->size();
254 if (num_of_celems == 3)
255 {
256 //parse a line of the remappers section
257 std::string remapper_name = bot_cont_elem->get(0).toString();
258 yarp::os::Bottle* clients_name = bot_cont_elem->get(1).asList();
259 std::string axes_names = bot_cont_elem->get(2).toString();
260 if (clients_name==nullptr || clients_name->size() == 0)
261 {
262 yError() << "error in clients_name";
263 return false;
264 }
265 if (axes_names.empty())
266 {
267 yError() << "error in axes_names";
268 return false;
269 }
270
271 //configure the remapper
272 if (m_robotRemappers.count(remapper_name))
273 {
274 yError() << "Duplicate remapper entry: " << remapper_name;
275 return false;
276 }
277 yarp::dev::PolyDriver* pd_remapper = new yarp::dev::PolyDriver;
278 yarp::os::Property rmoptions;
279 rmoptions.put("device", "controlboardremapper");
280 yarp::os::Value* jlist = yarp::os::Value::makeList(axes_names.c_str());
281 rmoptions.put("axesNames",jlist);
282 bool rob_ok = pd_remapper->open(rmoptions);
283 if (!rob_ok)
284 {
285 yError() << "Unable to initialize remapper: " << remapper_name;
286 delete pd_remapper;
287 return false;
288 }
289 m_robotRemappers[remapper_name] = pd_remapper;
290
291 //attach the remapper to one or more clients
292 {
293 yarp::dev::IMultipleWrapper* ww_rem=nullptr;
294 m_robotRemappers[remapper_name]->view(ww_rem);
295 yarp::dev::PolyDriverList pdlist;
296 for (size_t j=0; j<clients_name->size(); j++)
297 {
298 std::string cnn = clients_name->get(j).toString();
299 auto it = m_robotClients.find(cnn);
300 if (it == m_robotClients.end() || it->second == nullptr)
301 {
302 yError() << "Configuration error: client '" << cnn << "' not found or is invalid.";
303 return false;
304 }
305 yarp::dev::PolyDriver* ddnwc = it->second;
306 std::string boardname = "nwcboard" + std::to_string(j);
307 pdlist.push(ddnwc, boardname.c_str());
308 }
309 bool result_att = ww_rem->attachAll(pdlist);
310 if (!result_att)
311 {
312 yError() << "Configuration error while trying to attach the remapper: "
313 << remapper_name << " with the control_boards: " << clients_name->toString();
314 return false;
315 }
316 }
317
318 //put the controller in the list
319 if (m_robotControllers.count(remapper_name))
320 {
321 yError() << "Duplicate controller entry: " << remapper_name;
322 return false;
323 }
324 robotDriver* robDrv = new robotDriver;
325 rob_ok = robDrv->configure(m_robotRemappers[remapper_name]);
326 rob_ok &= robDrv->init();
327 if (!rob_ok)
328 {
329 yError() << "Unable to initialize controller: " << remapper_name;
330 delete robDrv;
331 return false;
332 }
333 m_robotControllers[remapper_name] = robDrv;
334 }
335 else
336 {
337 yError() << "Invalid number of entries in REMAPPERS section";
338 return false;
339 }
340 }
341
342 yarp::os::Bottle& bot_action = p.findGroup("ACTIONS");
343 if (bot_action.size() == 0)
344 {
345 yError() << "Unable to read ACTIONS section";
346 return false;
347 }
348 for (size_t i = 1; i < bot_action.size(); i++)
349 {
350 std::string str = bot_action.toString();
351 yDebug() << bot_action.get(i).toString();
352 yarp::os::Bottle* bot_act_elem = bot_action.get(i).asList();
353 if (bot_act_elem == nullptr)
354 {
355 yError() << "Invalid entry in ACTIONS section";
356 return false;
357 }
358 size_t num_of_aelems = bot_act_elem->size();
359 if (num_of_aelems==3 || num_of_aelems == 4)
360 {
361 //parse a line of the ACTIONS section
362 std::string action_name = bot_act_elem->get(0).toString();
363 std::string controller_name = bot_act_elem->get(1).toString();
364 std::string action_file_name = bot_act_elem->get(2).toString();
365 yarp::os::Bottle* tolerances_bot = bot_act_elem->get(3).asList();
366
367 //check if the controller name exists
368 if (m_robotControllers.find(controller_name) == m_robotControllers.end())
369 {
370 yError() << controller_name << "in action" << action_name << "does not exists";
371 return false;
372 }
373
374 //load the action file
375 action_class tmpAction;
376 tmpAction.action_name = action_name;
377 tmpAction.controller_name = controller_name;
378 size_t njoints = m_robotControllers[controller_name]->getNJoints();
379
380 if (!tmpAction.openFile(action_file_name, njoints, 0.010))
381 {
382 yError() << "Unable to parse file";
383 return false;
384 }
385 if (resample_period!=0.0)
386 {
387 tmpAction.interpolate_action_frames(resample_period);
388 }
389
390 tmpAction.m_tolerances.resize(njoints);
391 for (size_t i=0; i<njoints; i++)
392 {
393 tmpAction.m_tolerances[i] = 3.0; //deg
394 }
395 if (tolerances_bot)
396 {
397 if (tolerances_bot->size() != njoints)
398 {
399 yError() << "Invalid size of tolerances vector, should be"<<njoints << "got " << tolerances_bot->size();
400 return false;
401 }
402 for (size_t i=0; i<njoints; i++)
403 {
404 tmpAction.m_tolerances[i] = tolerances_bot->get(i).asFloat64();
405 }
406 }
407 yDebug() << "Tolerances for action:" << action_name << vectorToString(tmpAction.m_tolerances);
408
409 //put the action in the list
410 m_actions[action_name] = tmpAction;
411 }
412 else
413 {
414 yError() << "Invalid entry in ACTIONS section, size should be 3 or 4, got " <<num_of_aelems;
415 return false;
416 }
417 }
418
419 yInfo() << "configuration file successfully loaded";
420 return true;
421 }

442 virtual bool configure(yarp::os::ResourceFinder& rf)
443 {
444 std::string test_string = rf.toString();
445
446 // generic configuration
447 if (rf.check("name"))
448 m_name = std::string("/") + rf.find("name").asString().c_str();
449 else
450 m_name = "/yarpActionsPlayer";
451
452 // rpc port
453 m_rpcPort.open((m_name + "/rpc").c_str());
454 this->yarp().attachAsServer(m_rpcPort);
455
456 // get the configuration for parameter period
457 double period = 0.005;
458 if (rf.check("period") == true)
459 {
460 period = rf.find("period").asFloat64();
461 }
462
463 // Instantiate the thread
464 m_wthread = new ControlThread(m_name,period);
465
466 //set the configuration for parameter execute
467 if (rf.check("execute")==true)
468 {
469 yInfo() << "Enabling iPid->setReference() controller";
470 m_wthread->m_enable_execute_joint_command = true;
471 }
472 else
473 {
474 yInfo() << "Not using iPid->setReference() controller";
475 m_wthread->m_enable_execute_joint_command = false;
476 }
477
478 //set the configuration for parameter verbose
479 if (rf.check("verbose")==true)
480 {
481 yInfo() << "yarpActionPlayer running in verbose mode";
482 m_wthread->m_verbose = true;
483 }
484 else
485 {
486 yInfo() << "yarpActionPlayer running in non-verbose mode";
487 m_wthread->m_verbose = false;
488 }
489
490 //set the position tolerance
491 if (rf.check("help") == true)
492 {
493 print_help();
494 return false;
495 }
496
497 if (rf.check("initial_move_time") == true)
498 {
499 double imt = rf.find("initial_move_time").asFloat64();
500 m_wthread->setInitialMoveTime(imt);
501 }
502
503 //set the position tolerance
504 if (rf.check("pos_tolerance") == true)
505 {
506 double tol = rf.find("tolerance").asFloat64();
507 yInfo() << "Position tolerance set to " << tol << "degrees";
508 m_wthread->setPositionTolerance(tol);
509 }
510
511 //set the position timeout
512 if (rf.check("pos_timeout") == true)
513 {
514 double timeout = rf.find("pos_timeout").asFloat64();
515 yInfo() << "Position timeout set to " << timeout << "seconds";
516 m_wthread->setPositionTimeout(timeout);
517 }
518
519 //set the position timeout
520 if (rf.check("pos_strict_check") == true)
521 {
522 bool enable = rf.find("pos_strict_check").asBool();
523 m_wthread->setPositionStrictCheck(enable);
524 }
525
526 double resample_period = 0;
527 if (rf.check("resample") == true)
528 {
529 resample_period = rf.find("resample_period").asFloat64();
530 yInfo() << "Set resample period equal to:" << resample_period << "s";
531 }
532
533 //open the configuration file
534 if (rf.check("filename")==true)
535 {
536 if (rf.find("filename").isString())
537 {
538 std::string filename = rf.find("filename").asString();
539 bool b = loadConfiguration(filename, resample_period);
540 if (!b)
541 {
542 yError() << "Configuration error!";
543 return false;
544 }
545 }
546 else
547 {
548 yError() << "`filename` option syntax error.";
549 print_help();
550 return false;
551 }
552 }
553 else
554 {
555 yWarning() << "`filename` option not found. No sequence files loaded.";
556 }
557
558 //check if actions are valid
559 if (m_actions.empty())
560 {
561 yInfo() << "There are no actions!";
562 return false;
563 }
564
565 //select the first action
566 yInfo() << "automatically selecting the first action";
567 std::string first_action_name;
568 first_action_name = this->m_actions.begin()->first;
569 this->chooseActionByName(first_action_name);
570
571 //start the thread
572 if (!m_wthread->start())
573 {
574 yError() << "Working thread did not start, queue will not work";
575 }
576 else
577 {
578 yInfo() << "Working thread started";
579 }
580
581 yInfo() << "module successfully configured. ready.";
582 return true;
583 }

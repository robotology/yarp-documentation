149bool PlayerModule::send()
150{
151 bool ret;
152 double st= yarp::os::Time::now();
153 yInfo() << "Sending sound...";
154 if (m_audioPort.getOutputCount() > 0)
155 {
156 if (m_split_size==0)
157 {
158 //send the whole file
159 ret = m_audioPort.write(m_audioFile);
160 }
161 else
162 {
163 //split the file in blocks of fixed size
164 size_t totalSamples = m_audioFile.getSamples();
165
166 //this is a ceiling division on integers so that: 10 / 3 = 4
167 size_t iterations = (totalSamples + m_split_size - 1) / m_split_size;
168
169 size_t block_counter =0;
170 for (size_t first = 0; first < totalSamples; first += m_split_size)
171 {
172 Sound block = m_audioFile.subSound(first, m_split_size);
173 double st_block = yarp::os::Time::now();
174 ret = m_audioPort.write(block);
175 double en_block = yarp::os::Time::now();
176 yInfo() << "Sent block" << block_counter << "/" << iterations << " with " << block.getSamples() << " samples in" << en_block - st_block << "s";
177 block_counter++;
178 }
179 }
180 }
181 else
182 {
183 yWarning() << "No receivers are connected";
184 }
185 double en= yarp::os::Time::now();
186 yInfo() << "Sound sent in: " << en - st << "s";
187
188 return true;
189}

59ReturnValue ImplementControlLimits::setPosLimits(int axis, double min, double max)
60{
61 std::lock_guard lock(m_imp_mutex);
62 JOINTIDCHECK(IMPLEMENT_LAYER_COMPONENT,axis);
63
64 double minEnc=0;
65 double maxEnc=0;
66
67 int k=0;
68 castToMapper(m_helper)->posA2E(min, axis, minEnc, k);
69 castToMapper(m_helper)->posA2E(max, axis, maxEnc, k);
70
71 if( (max > min) && (minEnc > maxEnc)) //angle to encoder conversion factor is negative
72 {
73 double temp; // exchange max and min limits
74 temp = minEnc;
75 minEnc = maxEnc;
76 maxEnc = temp;
77 }
78
79 return m_iraw->setPosLimitsRaw(k, minEnc, maxEnc);
80}

83ReturnValue ImplementControlLimits::getPosLimits(int axis, double *min, double *max)
84{
85 std::lock_guard lock(m_imp_mutex);
86 JOINTIDCHECK(IMPLEMENT_LAYER_COMPONENT,axis)
87 POINTERCHECK(IMPLEMENT_LAYER_COMPONENT,min)
88 POINTERCHECK(IMPLEMENT_LAYER_COMPONENT,max)
89
90 double minEnc=0;
91 double maxEnc=0;
92
93 int k=castToMapper(m_helper)->toHw(axis);
94 ReturnValue ret=m_iraw->getPosLimitsRaw(k, &minEnc, &maxEnc);
95
96 *min=castToMapper(m_helper)->posE2A(minEnc, k);
97 *max=castToMapper(m_helper)->posE2A(maxEnc, k);
98
99 if( (*max < *min) && (minEnc < maxEnc)) //angle to encoder conversion factor is negative
100 {
101 double temp; // exchange max and min limits
102 temp = *min;
103 *min = *max;
104 *max = temp;
105 }
106 return ret;
107}

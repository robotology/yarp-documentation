55bool PortCoreClient::AdminPortLink::read(ConnectionReader& connection)
56{
57 Bottle cmd;
58 Bottle reply;
59 cmd.read(connection);
60 //yCDebug(PORTCORECLIENT, "asking %s: %s", m_contact.toString().c_str(), cmd.toString().c_str());
61 if (!NetworkBase::write(m_contact, cmd, reply, m_style)) {
62 if (!m_style.quiet) {
63 yCError(PORTCORECLIENT, "Cannot write to %s", m_contact.getName().c_str());
64 }
65 return false;
66 }
67 ConnectionWriter* writer = connection.getWriter();
68 if (writer != nullptr) {
69 reply.write(*writer);
70 }
71 return true;
72}

48bool StreamingMessagesParser::initialize()
49{
50 stream_nJoints = 0;
51 if (stream_IPosCtrl) { stream_IPosCtrl->getAxes(stream_nJoints); return true; }
52 if (stream_IVel) { stream_IVel->getAxes(stream_nJoints); return true; }
53 if (stream_ITorque) { stream_ITorque->getAxes(stream_nJoints); return true; }
54 if (stream_IAxis) { stream_IAxis->getAxes(stream_nJoints); return true; }
55
56 yCError(CONTROLBOARD, "Unable to get number of joints");
57 return false;
58}

60// streaming port callback
61void StreamingMessagesParser::onRead(yarp::dev::CommandMessageData& v)
62{
63 Bottle& b = v.head;
64 Vector& cmdVector = v.body;
65
66 //Use the following only for debug, since it can heavily slow down the system
67 yCTrace(CONTROLBOARD, "Received command %s, %s\n", b.toString().c_str(), cmdVector.toString().c_str());
68
69 // some consistency checks
70 if (static_cast<int>(cmdVector.size()) > stream_nJoints) {
71 std::string str = yarp::os::Vocab32::decode(b.get(0).asVocab32());
72 yCError(CONTROLBOARD, "Received command vector with number of elements bigger than axis controlled by this wrapper (cmd: %s requested jnts: %d received jnts: %d)\n", str.c_str(), stream_nJoints, (int)cmdVector.size());
73 return;
74 }
75 if (cmdVector.data() == nullptr) {
76 yCError(CONTROLBOARD, "Received null command vector");
77 return;
78 }
79
80 switch (b.get(0).asVocab32()) {
81 // manage commands with interface name as first...
82 // PWM control
83 case VOCAB_PWMCONTROL_INTERFACE: {
84 switch (b.get(1).asVocab32()) {
85 case VOCAB_PWMCONTROL_REF_PWM: {
86 if (stream_IPWM) {
87 int joint = b.get(2).asInt32();
88 bool ok = stream_IPWM->setRefDutyCycle(joint, cmdVector[0]);
89 if (!ok) {
90 yCError(CONTROLBOARD, "Errors while trying to command an pwm message");
91 }
92 } else {
93 yCError(CONTROLBOARD, "PWM interface not valid");
94 }
95 } break;
96 case VOCAB_PWMCONTROL_REF_PWMS: {
97 if (stream_IPWM) {
98 bool ok = stream_IPWM->setRefDutyCycles(cmdVector.data());
99 if (!ok) {
100 yCError(CONTROLBOARD, "Errors while trying to command an pwm message");
101 }
102 } else {
103 yCError(CONTROLBOARD, "PWM interface not valid");
104 }
105 } break;
106 default:
107 {
108 std::string str = yarp::os::Vocab32::decode(b.get(0).asVocab32());
109 yCError(CONTROLBOARD, "Unrecognized message while receiving on command port (%s)\n", str.c_str());
110 } break;
111 }
112 } break;
113
114 // PID control
115 case VOCAB_PIDCONTROL_INTERFACE: {
116 switch (b.get(1).asVocab32()) {
117 case VOCAB_PIDCONTROL_SET_OFFSET: {
118 if (stream_IPid) {
119 PidControlTypeEnum pidtype = PidControlTypeEnum(b.get(2).asVocab32());
120 int joint = b.get(3).asInt32();
121 double off = cmdVector[0];
122 bool ok = stream_IPid->setPidOffset(pidtype, joint, off);
123 if (!ok) {
124 yCError(CONTROLBOARD, "Errors while trying to execute streaming command setPidOffset()\n");
125 }
126 }
127 } break;
128 case VOCAB_PIDCONTROL_SET_FEEDFORWARD: {
129 if (stream_IPid) {
130 PidControlTypeEnum pidtype = PidControlTypeEnum(b.get(2).asVocab32());
131 int joint = b.get(3).asInt32();
132 double ffd = cmdVector[0];
133 bool ok = stream_IPid->setPidFeedforward(pidtype, joint, ffd);
134 if (!ok) {
135 yCError(CONTROLBOARD, "Errors while trying to execute streaming command setPidFeedforward()\n");
136 }
137 }
138 } break;
139 default:
140 {
141 std::string str = yarp::os::Vocab32::decode(b.get(0).asVocab32());
142 yCError(CONTROLBOARD, "Unrecognized message while receiving on command port (%s)\n", str.c_str());
143 } break;
144 }
145 } break;
146
147 // Current control
148 case VOCAB_CURRENTCONTROL_INTERFACE: {
149 switch (b.get(1).asVocab32()) {
150 case VOCAB_CURRENT_REF: {
151 if (stream_ICurrent) {
152 int joint = b.get(2).asInt32();
153 bool ok = stream_ICurrent->setRefCurrent(joint, cmdVector[0]);
154 if (!ok) {
155 yCError(CONTROLBOARD, "Errors while trying to execute streaming command setRefCurrent()\n");
156 }
157 }
158 } break;
159 case VOCAB_CURRENT_REFS: {
160 if (stream_ICurrent) {
161 bool ok = stream_ICurrent->setRefCurrents(cmdVector.data());
162 if (!ok) {
163 yCError(CONTROLBOARD, "Errors while trying to execute streaming command setRefCurrents()\n");
164 }
165 }
166 } break;
167 case VOCAB_CURRENT_REF_GROUP: {
168 if (stream_ICurrent) {
169 int n_joints = b.get(2).asInt32();
170 Bottle* jlut = b.get(3).asList();
171 if ((static_cast<int>(jlut->size()) != n_joints) && (static_cast<int>(cmdVector.size()) != n_joints)) {
172 yCError(CONTROLBOARD, "Received VOCAB_CURRENT_REF_GROUP size of joints vector or currents vector does not match the selected joint number\n");
173 }
174
175 int* joint_list = new int[n_joints];
176 for (int i = 0; i < n_joints; i++) {
177 joint_list[i] = jlut->get(i).asInt32();
178 }
179
180
181 bool ok = stream_ICurrent->setRefCurrents(n_joints, joint_list, cmdVector.data());
182 if (!ok) {
183 yCError(CONTROLBOARD, "Errors while trying to execute streaming command setRefCurrents() group\n");
184 }
185
186 delete[] joint_list;
187 }
188 } break;
189 default:
190 {
191 std::string str = yarp::os::Vocab32::decode(b.get(0).asVocab32());
192 yCError(CONTROLBOARD, "Unrecognized message while receiving on command port (%s)\n", str.c_str());
193 } break;
194 }
195 } break;
196
197 // Torque control
198 case VOCAB_TORQUECONTROL_INTERFACE: {
199 switch (b.get(1).asVocab32()) {
200 case VOCAB_TORQUES_DIRECT: {

65inline std::string get_string(const std::string& key, bool* found = nullptr)
66{
67 const char* result = std::getenv(key.c_str());
68 if (found != nullptr) {
69 *found = (result != nullptr);
70 }
71 if (result == nullptr) {
72 return {};
73 }
74 return std::string(result);
75}

142inline std::vector<std::string> get_path(const std::string& key, const std::string& altKey, const std::string& altDefaultValue, const std::string& altAppend = {})
143{
144 bool found;
145 std::vector<std::string> path = get_path(key, &found);
146 if (!found) {
147 path = get_path(altKey, altDefaultValue);
148 if (!altAppend.empty()) {
149 for (auto& dir : path) {
150 dir.append(altAppend);
151 }
152 }
153 }
154 return path;
155}

231inline bool set_bool(const std::string& key, bool value)
232{
233 // Boolean environment variables are usually set as "1" or "0", and not

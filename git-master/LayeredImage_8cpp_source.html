63bool LayeredImage::operator==(const LayeredImage& alt) const
64{
65 size_t l1 = this->layers.size();
66 size_t l2 = alt.layers.size();
67
68 if (l1 != l2)
69 {
70 return false;
71 }
72
73 if (background != alt.background)
74 {
75 return false;
76 }
77
78 for (size_t i = 0; i < l1; i++)
79 {
80 if ((this->layers[i].enable != alt.layers[i].enable) ||
81 (this->layers[i].colorkey.enable != alt.layers[i].colorkey.enable) ||
82 (this->layers[i].colorkey.value != alt.layers[i].colorkey.value) ||
83 (this->layers[i].alpha.enable != alt.layers[i].alpha.enable) ||
84 (fabs(this->layers[i].alpha.value - alt.layers[i].alpha.value) > 0.001) ||
85 (this->layers[i].layer != alt.layers[i].layer) ||
86 (this->layers[i].can_be_compressed != alt.layers[i].can_be_compressed) ||
87 (this->layers[i].offset_x != alt.layers[i].offset_x) ||
88 (this->layers[i].offset_y != alt.layers[i].offset_y))

74ReturnValue ImplementVelocityControl::velocityMove(const int n_joints, const int *joints, const double *spds)
75{
76 std::lock_guard lock(m_imp_mutex);
77 POINTERCHECK(IMPLEMENT_LAYER_COMPONENT,joints);
78 POINTERCHECK(IMPLEMENT_LAYER_COMPONENT,spds);
79 JOINTSIDCHECK(IMPLEMENT_LAYER_COMPONENT,n_joints, joints)
80
81 std::vector<int> vectorInt_tmp(n_joints);
82 std::vector<double> vectorDouble_tmp(n_joints);
83
84 for(int idx=0; idx<n_joints; idx++)
85 {
86 vectorInt_tmp[idx] = castToMapper(m_helper)->toHw(joints[idx]);
87 vectorDouble_tmp[idx] = castToMapper(m_helper)->velA2E(spds[idx], joints[idx]);
88 }
89 ReturnValue ret = m_iraw->velocityMoveRaw(n_joints, vectorInt_tmp.data(), vectorDouble_tmp.data());
90
91 return ret;

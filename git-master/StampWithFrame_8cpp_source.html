45 void update(yarp::conf::float64_t time)
46 {
47 sequenceNumber++;
48 if (sequenceNumber == StampWithFrame::npos) {
49 // npos is not used to store a valid StampWithFrame, just restart the
50 // counter
51 sequenceNumber = 0;
52 }
53 timeStamp = time;
54 }

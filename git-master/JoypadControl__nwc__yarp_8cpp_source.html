37bool JoypadControl_nwc_yarp::open(yarp::os::Searchable& config)
38{
39 if (!parseParams(config)) { return false; }
40
41 //open the rpc port
42 if(!m_rpcPort.open(m_local + "/rpc:o"))
43 {
44 yCError(JOYPADCONTROLCLIENT) << "Unable to open rpc port.";
45 return false;
46 }
47 yCInfo(JOYPADCONTROLCLIENT) << "rpc port opened. starting the handshake";
48
49 //connection for the rpc port
50 if(!yarp::os::NetworkBase::connect(m_local + "/rpc:o", m_remote + "/rpc:i"))
51 {
52 yCError(JOYPADCONTROLCLIENT) << "Handshake failed. unable to connect to remote port" << m_remote + "/rpc:i";
53 return false;
54 }
55
56 if (!m_rpcMsgs.yarp().attachAsClient(m_rpcPort))
57 {
58 yCError(JOYPADCONTROLCLIENT, "Error! Cannot attach the port as a client");
59 return false;
60 }
61
62 // Check the protocol version
63 if (!m_rpcMsgs.checkProtocolVersion()) {
64 return false;
65 }
66
67 //Get basic infos from the joypad
68 if(!getJoypadInfo())
69 {
70 yCError(JOYPADCONTROLCLIENT) << "Unable to get joypad info.";
71 return false;
72 }
73
74 if (m_use_streaming==false) {return true;}
75
76 //The code below is valid only for streaming mode
77 if (m_use_all_port)
78 {
79 bool b = true;
80 b &= m_allJoyDataPort.open(m_local + "/allJoyData:i");
81 if (!b) {
82 yCError(JOYPADCONTROLCLIENT) << "Unable to open local streaming ports";
83 close();
84 return false;
85 }
86
87 b&= yarp::os::NetworkBase::connect(m_remote+"/allJoyData:o", m_local+"/allJoyData:i");
88 if (!b) {
89 yCError(JOYPADCONTROLCLIENT) << "Unable to connect to remote streaming port";
90 close();
91 return false;
92 }
93
94 m_allJoyDataPort.useCallback();
95 }
96 else
97 {
98 bool b = true;
99 b&= m_buttonsPort.open(m_local+"/buttons:i");
100 b&= m_axisPort.open(m_local+"/axis:i");
101 b&= m_trackballPort.open(m_local+"/trackball:i");
102 b&= m_touchPort.open(m_local+"/touch:i");
103 b&= m_hatsPort.open(m_local+"/hat:i");
104 b&= m_stickPort.open(m_local+"/stick:i");
105 if (!b) {
106 yCError(JOYPADCONTROLCLIENT) << "Unable to open local streaming ports";
107 close();
108 return false;
109 }
110
111 b&= yarp::os::NetworkBase::connect(m_remote+"/buttons:o",m_local+"/buttons:i");
112 b&= yarp::os::NetworkBase::connect(m_remote+"/axis:o",m_local+"/axis:i");
113 b&= yarp::os::NetworkBase::connect(m_remote+"/trackball:o",m_local+"/trackball:i");
114 b&= yarp::os::NetworkBase::connect(m_remote+"/touch:o",m_local+"/touch:i");
115 b&= yarp::os::NetworkBase::connect(m_remote+"/hat:o",m_local+"/hat:i");
116 b&= yarp::os::NetworkBase::connect(m_remote+"/stick:o",m_local+"/stick:i");
117 if (!b) {
118 yCError(JOYPADCONTROLCLIENT) << "Unable to connect to remote streaming port";
119 close();
120 return false;
121 }
122
123 m_buttonsPort.useCallback();
124 m_axisPort.useCallback();
125 m_trackballPort.useCallback();
126 m_touchPort.useCallback();
127 m_hatsPort.useCallback();
128 m_stickPort.useCallback();
129 }
130
131 return true;
132}

178{

180 {
181 auto ret = m_rpcMsgs.getButton(button_id);
182 value = ret.value;
183 return ret.ret;
184 }
185 else if (m_use_all_port)
186 {

191 return ReturnValue_ok;

196 return ReturnValue_error_method_failed;

408ReturnValue JoypadControl_nwc_yarp::getTouch(size_t touch_id, std::vector<yarp::dev::TouchData>& value)
409{
410 value.clear();
411 if(!m_use_streaming)
412 {
413 auto ret = m_rpcMsgs.getTouch(touch_id);
414 value = ret.value;
415 return ret.ret;
416 }
417 else if (m_use_all_port)
418 {
419 auto tmp = m_allJoyDataPort.getData();
420 if(touch_id < tmp.TouchDataVal.size())
421 {
422 value = tmp.TouchDataVal[touch_id].touches;
423 return ReturnValue_ok;
424 }
425 else
426 {
427 yCError(JOYPADCONTROLCLIENT) << "GetTouch() error. touch_id out of bound";
428 return ReturnValue_error_method_failed;
429 }
430 }
431 else
432 {
433 auto temp = m_touchPort.getData();
434 if(touch_id < temp.value.size())
435 {
436 value.resize(temp.value[touch_id].touches.size());
437 for (size_t i = 0; i < value.size(); i++)
438 {
439 value[i] = temp.value[touch_id].touches[i];
440 }
441 return ReturnValue_ok;
442 }
443 else
444 {
445 return ReturnValue_error_method_failed;
446 }

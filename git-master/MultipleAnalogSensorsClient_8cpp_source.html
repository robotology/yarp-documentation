19void SensorStreamingDataInputPort::onRead(yarp::dev::SensorStreamingData& v)
20{
21 std::lock_guard<std::mutex> guard(dataMutex);
22 receivedData = v;
23 lastTimeStampReadInSeconds = yarp::os::Time::now();
24 status = yarp::dev::MAS_OK;
25}
26
27void SensorStreamingDataInputPort::updateTimeoutStatus() const
28{
29 if (status == yarp::dev::MAS_OK)
30 {
31 double now = yarp::os::Time::now();
32 if ((now-lastTimeStampReadInSeconds) > this->timeoutInSeconds)
33 {
34 yCError(MULTIPLEANALOGSENSORSCLIENT,
35 "No data received in the last %lf seconds, timeout enabled.",
36 (now-lastTimeStampReadInSeconds));
37 status = yarp::dev::MAS_TIMEOUT;
38 }
39 }
40}

42bool MultipleAnalogSensorsClient::open(yarp::os::Searchable& config)
43{
44 if (!parseParams(config)) { return false; }
45
46 m_streamingPort.timeoutInSeconds = m_timeout;
47
48 std::string localRPCPortName = m_local + "/rpc:i";
49 std::string localStreamingPortName = m_local + "/measures:i";
50 std::string remoteRPCPortName = m_remote + "/rpc:o";
51 std::string remoteStreamingPortName = m_remote + "/measures:o";
52
53 // TODO(traversaro) : as soon as the method for checking port names validity
54 // are available in YARP ( https://github.com/robotology/yarp/pull/1508 ) add some checks
55
56 // Open ports
57 bool ok = m_rpcPort.open(localRPCPortName);
58 if (!ok)
59 {
60 yCError(MULTIPLEANALOGSENSORSCLIENT,
61 "Failure to open the port %s.",
62 localRPCPortName.c_str());
63 close();
64 return false;
65 }
66
67 ok = m_streamingPort.open(localStreamingPortName);
68 m_streamingPort.useCallback();
69 if (!ok)
70 {
71 yCError(MULTIPLEANALOGSENSORSCLIENT,
72 "Failure to open the port %s.",
73 localStreamingPortName.c_str());
74 close();
75 return false;
76 }
77
78 // Connect ports
79 if (!m_externalConnection) {
80 // RPC port needs to be tcp, therefore no carrier option is added here
81 ok = yarp::os::Network::connect(localRPCPortName, remoteRPCPortName);
82 if (!ok) {
83 yCError(MULTIPLEANALOGSENSORSCLIENT,
84 "Failure connecting port %s to %s.",
85 localRPCPortName.c_str(),
86 remoteRPCPortName.c_str());
87 yCError(MULTIPLEANALOGSENSORSCLIENT, "Check that the specified MultipleAnalogSensorsServer is up.");
88 close();
89 return false;
90 }
91
92 ok = yarp::os::Network::connect(remoteStreamingPortName, localStreamingPortName, m_carrier);
93 if (!ok) {
94 yCError(MULTIPLEANALOGSENSORSCLIENT,
95 "Failure connecting port %s to %s.",
96 remoteStreamingPortName.c_str(),
97 localStreamingPortName.c_str());
98 yCError(MULTIPLEANALOGSENSORSCLIENT, "Check that the specified MultipleAnalogSensorsServer is up.");
99 close();
100 return false;
101 }
102
103 // Once the connection is active, we just the metadata only once
104 ok = m_RPCInterface.yarp().attachAsClient(m_rpcPort);
105 if (!ok) {
106 yCError(MULTIPLEANALOGSENSORSCLIENT, "Failure opening Thrift-based RPC interface.");
107 return false;
108 }
109
110 // TODO(traversaro): there is a limitation on the thrift-generated
111 // YARP structures related to how to get connection errors during the call
112 // If this is ever solved at YARP level, we should properly handle errors
113 // here
114 m_sensorsMetadata = m_RPCInterface.getMetadata();
115
116 }
117
118 yCDebug(MULTIPLEANALOGSENSORSCLIENT,"Open complete");
119 return true;
120}

131size_t MultipleAnalogSensorsClient::genericGetNrOfSensors(const std::vector<SensorMetadata>& metadataVector,
132 const yarp::dev::SensorMeasurements& measurementsVector) const
133{
134 if (!m_externalConnection) {
135 return metadataVector.size();
136 } else {
137 std::lock_guard<std::mutex> guard(m_streamingPort.dataMutex);
138 m_streamingPort.updateTimeoutStatus();
139 if (m_streamingPort.status == yarp::dev::MAS_OK) {
140 return measurementsVector.measurements.size();
141 } else {
142 return 0;
143 }
144 }
145}

150 m_streamingPort.updateTimeoutStatus();
151 return m_streamingPort.status;

154ReturnValue MultipleAnalogSensorsClient::genericGetName(const std::vector<SensorMetadata>& metadataVector, const std::string& tag,
155 size_t sens_index, std::string& name) const
156{
157 if (m_externalConnection) {
158 yCError(MULTIPLEANALOGSENSORSCLIENT,
159 "Missing metadata, the device has been configured with the option"
160 "externalConnection set to true.");
161 return ReturnValue_error_method_failed;
162 }
163 if (sens_index >= metadataVector.size())
164 {
165 yCError(MULTIPLEANALOGSENSORSCLIENT,
166 "No sensor of type %s with index %lu (nr of sensors: %lu).",
167 tag.c_str(),
168 sens_index,
169 metadataVector.size());
170 return ReturnValue_error_input_out_of_bounds;
171 }
172
173 name = metadataVector[sens_index].name;
174 return ReturnValue_ok;
175}

200ReturnValue MultipleAnalogSensorsClient::genericGetMeasure(const std::vector<SensorMetadata>& metadataVector, const std::string& tag,
201 const yarp::dev::SensorMeasurements& measurementsVector,
202 size_t sens_index, yarp::sig::Vector& out, double& timestamp) const
203{
204
205 std::lock_guard<std::mutex> guard(m_streamingPort.dataMutex);
206 m_streamingPort.updateTimeoutStatus();
207 if (m_streamingPort.status != yarp::dev::MAS_OK)
208 {
209 yCError(MULTIPLEANALOGSENSORSCLIENT,
210 "Sensor of type %s with index %lu has non-MAS_OK status.",
211 tag.c_str(),
212 sens_index);
213 return ReturnValue_error_not_ready;
214 }
215
216 if (m_streamingPort.status != (sens_index >= measurementsVector.measurements.size()))
217 {
218 yCError(MULTIPLEANALOGSENSORSCLIENT,
219 "No sensor of type %s with index %lu (nr of sensors: %lu).",
220 tag.c_str(),
221 sens_index,
222 metadataVector.size());
223 return ReturnValue_error_input_out_of_bounds;
224 }
225
226 if (!m_externalConnection) {
227 assert(metadataVector.size() == measurementsVector.measurements.size());
228 }
229
230 timestamp = measurementsVector.measurements[sens_index].timestamp;
231 out = measurementsVector.measurements[sens_index].measurement;
232
233 return ReturnValue_ok;
234}

236size_t MultipleAnalogSensorsClient::genericGetSize(const std::vector<SensorMetadata>& metadataVector,
237 const std::string& tag, const yarp::dev::SensorMeasurements& measurementsVector, size_t sens_index) const
238{
239 std::lock_guard<std::mutex> guard(m_streamingPort.dataMutex);
240 if (m_streamingPort.status != yarp::dev::MAS_OK)
241 {
242 yCError(MULTIPLEANALOGSENSORSCLIENT, "No data received, no information on the size of the specified sensor.");
243 return 0;
244 }
245
246
247 if (sens_index >= measurementsVector.measurements.size())
248 {
249 yCError(MULTIPLEANALOGSENSORSCLIENT,
250 "No sensor of type %s with index %lu (nr of sensors: %lu).",
251 tag.c_str(),
252 sens_index,
253 metadataVector.size());
254 return 0;
255 }
256
257 return measurementsVector.measurements[sens_index].measurement.size();
258}

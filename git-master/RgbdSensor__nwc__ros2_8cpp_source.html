36bool RgbdSensor_nwc_ros2::open(yarp::os::Searchable& config)
37{
38 parseParams(config);
39 m_topic_depth_camera_info = m_depth_topic_name.substr(0, m_depth_topic_name.rfind('/')) + "/camera_info";
40 m_topic_rgb_camera_info = m_color_topic_name.substr(0, m_color_topic_name.rfind('/')) + "/camera_info";
41
42 m_verbose = m_verbose_on == 1;
43
44 if(m_namespace.empty())
45 {
46 m_node = NodeCreator::createNode(m_node_name);
47 }
48 else
49 {
50 m_node = NodeCreator::createNode(m_node_name, m_namespace);
51 }
52 if(!m_node)
53 {
54 yCError(RGBDSENSOR_NWC_ROS2) << "Cannot create node";
55 return false;
56 }
57 m_sub1= new Ros2Subscriber<RgbdSensor_nwc_ros2, sensor_msgs::msg::CameraInfo>(m_node, this);
58 m_sub1->subscribe_to_topic(m_topic_rgb_camera_info);
59 m_sub1->subscribe_to_topic(m_topic_depth_camera_info);
60 m_sub2= new Ros2Subscriber<RgbdSensor_nwc_ros2, sensor_msgs::msg::Image>(m_node, this);
61 m_sub2->subscribe_to_topic(m_color_topic_name);
62 m_sub2->subscribe_to_topic(m_depth_topic_name);
63
64 m_spinner = new Ros2Spinner(m_node);
65 m_spinner->start();
66
67 yCInfo(RGBDSENSOR_NWC_ROS2) << "opened";
68
69 return true;
70}

139void RgbdSensor_nwc_ros2::saveIntrinsics(sensor_msgs::msg::CameraInfo::SharedPtr msg, yarp::sig::IntrinsicParams& params)
140{
141 params.focalLengthX = msg->k[0];
142 params.focalLengthY = msg->k[4];
143 params.principalPointX = msg->k[2];
144 params.principalPointY = msg->k[5];
145 // distortion model
146 if (msg->distortion_model=="plumb_bob")
147 {
148 if (params.distortionModel.k1 == 0 &&
149 params.distortionModel.k2 == 0 &&
150 params.distortionModel.k3 == 0 &&
151 params.distortionModel.t1 == 0 &&
152 params.distortionModel.t2 == 0)
153 {
154 params.distortionModel.type = yarp::sig::YarpDistortion::YARP_DISTORTION_NONE;
155 }
156 params.distortionModel.type = yarp::sig::YarpDistortion::YARP_PLUMB_BOB;
157 params.distortionModel.k1 = msg->d[0];
158 params.distortionModel.k2 = msg->d[1];
159 params.distortionModel.t1 = msg->d[2];
160 params.distortionModel.t2 = msg->d[3];
161 params.distortionModel.k3 = msg->d[4];
162 }
163 else
164 {
165 yCError(RGBDSENSOR_NWC_ROS2) << "Unsupported distortion model";
166 params.distortionModel.type = yarp::sig::YarpDistortion::YARP_UNSUPPORTED;
167 }
168}

363{

365 std::lock_guard<std::mutex> guard_rgb_camera_info(m_rgb_camera_info_mutex);

368 {
369 if (m_rgb_stamp_valid)
370 {
371 if (rgb_image_stamp == nullptr)
372 {
373 yarp::os::Stamp stamp(m_current_rgb_stamp.getCount(),m_current_rgb_stamp.getTime());
374 rgb_image_stamp = &stamp;
375 } else {
376 rgb_image_stamp->update(m_current_rgb_stamp.getTime());
377 }
378 yarp::dev::Ros2RGBDConversionUtils::deepCopyFlexImage(m_current_rgb_image, rgb_image);
379 rgb_ok = true;
380 }
381 else
382 {
383 yCError(RGBDSENSOR_NWC_ROS2, "missing rgb camera info");
384 }
385 }
386 else
387 {
388 yCError(RGBDSENSOR_NWC_ROS2, "missing rgb image");
389 }
390 if (rgb_ok) return ReturnValue_ok;
391 return ReturnValue::return_code::return_value_error_not_ready;

399/*
400 Compare two type names skipping the space characters, therefore
401 "char*" == "char *" and "Class<int>" == "Class<int >", etc.
402
403 Return 0 when the two name types are equivalent, as in
404 strncmp, but skipping ' '.
405*/
406SWIGRUNTIME int
407SWIG_TypeNameComp(const char *f1, const char *l1,
408 const char *f2, const char *l2) {
409 for (;(f1 != l1) && (f2 != l2); ++f1, ++f2) {
410 while ((*f1 == ' ') && (f1 != l1)) ++f1;
411 while ((*f2 == ' ') && (f2 != l2)) ++f2;
412 if (*f1 != *f2) return (*f1 > *f2) ? 1 : -1;
413 }
414 return (int)((l1 - f1) - (l2 - f2));
415}

417/*
418 Check type equivalence in a name list like <name1>|<name2>|...
419 Return 0 if equal, -1 if nb < tb, 1 if nb > tb
420*/
421SWIGRUNTIME int
422SWIG_TypeCmp(const char *nb, const char *tb) {
423 int equiv = 1;
424 const char* te = tb + strlen(tb);
425 const char* ne = nb;
426 while (equiv != 0 && *ne) {
427 for (nb = ne; *ne; ++ne) {
428 if (*ne == '|') break;
429 }
430 equiv = SWIG_TypeNameComp(nb, ne, tb, te);
431 if (*ne) ++ne;
432 }
433 return equiv;
434}

446 Check the typename

448SWIGRUNTIME swig_cast_info *
449SWIG_TypeCheck(const char *c, swig_type_info *ty) {
450 if (ty) {
451 swig_cast_info *iter = ty->cast;
452 while (iter) {
453 if (strcmp(iter->type->name, c) == 0) {
454 if (iter == ty->cast)
455 return iter;
456 /* Move iter to the top of the linked list */
457 iter->prev->next = iter->next;
458 if (iter->next)
459 iter->next->prev = iter->prev;
460 iter->next = ty->cast;
461 iter->prev = 0;
462 if (ty->cast) ty->cast->prev = iter;
463 ty->cast = iter;
464 return iter;
465 }
466 iter = iter->next;
467 }
468 }
469 return 0;
470}

576/*
577 Search for a swig_type_info structure only by mangled name
578 Search is a O(log #types)
579
580 We start searching at module start, and finish searching when start == end.
581 Note: if start == end at the beginning of the function, we go all the way around
582 the circular list.
583*/
584SWIGRUNTIME swig_type_info *
585SWIG_MangledTypeQueryModule(swig_module_info *start,
586 swig_module_info *end,
587 const char *name) {
588 swig_module_info *iter = start;
589 do {
590 if (iter->size) {
591 size_t l = 0;
592 size_t r = iter->size - 1;
593 do {
594 /* since l+r >= 0, we can (>> 1) instead (/ 2) */
595 size_t i = (l + r) >> 1;
596 const char *iname = iter->types[i]->name;
597 if (iname) {
598 int compare = strcmp(name, iname);
599 if (compare == 0) {
600 return iter->types[i];
601 } else if (compare < 0) {
602 if (i) {
603 r = i - 1;
604 } else {
605 break;
606 }
607 } else if (compare > 0) {
608 l = i + 1;
609 }
610 } else {
611 break; /* should never happen */
612 }
613 } while (l <= r);
614 }
615 iter = iter->next;
616 } while (iter != end);
617 return 0;
618}

620/*
621 Search for a swig_type_info structure for either a mangled name or a human readable name.
622 It first searches the mangled names of the types, which is a O(log #types)
623 If a type is not found it then searches the human readable names, which is O(#types).
624
625 We start searching at module start, and finish searching when start == end.
626 Note: if start == end at the beginning of the function, we go all the way around
627 the circular list.
628*/
629SWIGRUNTIME swig_type_info *
630SWIG_TypeQueryModule(swig_module_info *start,
631 swig_module_info *end,
632 const char *name) {
633 /* STEP 1: Search the name field using binary search */
634 swig_type_info *ret = SWIG_MangledTypeQueryModule(start, end, name);
635 if (ret) {
636 return ret;
637 } else {
638 /* STEP 2: If the type hasn't been found, do a complete search
639 of the str field (the human readable name) */
640 swig_module_info *iter = start;
641 do {
642 size_t i = 0;
643 for (; i < iter->size; ++i) {
644 if (iter->types[i]->str && (SWIG_TypeEquiv(iter->types[i]->str, name)))
645 return iter->types[i];
646 }
647 iter = iter->next;
648 } while (iter != end);
649 }
650
651 /* neither found a match */
652 return 0;
653}

672 Unpack binary data from a string
673*/
674SWIGRUNTIME const char *
675SWIG_UnpackData(const char *c, void *ptr, size_t sz) {
676 unsigned char *u = (unsigned char *) ptr;
677 const unsigned char *eu = u + sz;
678 for (; u != eu; ++u) {
679 char d = *(c++);
680 unsigned char uu;
681 if ((d >= '0') && (d <= '9'))
682 uu = (unsigned char)((d - '0') << 4);
683 else if ((d >= 'a') && (d <= 'f'))
684 uu = (unsigned char)((d - ('a'-10)) << 4);
685 else
686 return (char *) 0;
687 d = *(c++);
688 if ((d >= '0') && (d <= '9'))
689 uu |= (unsigned char)(d - '0');
690 else if ((d >= 'a') && (d <= 'f'))
691 uu |= (unsigned char)(d - ('a'-10));

1203/* Unpack the argument tuple */
1204
1205SWIGINTERN Py_ssize_t
1206SWIG_Python_UnpackTuple(PyObject *args, const char *name, Py_ssize_t min, Py_ssize_t max, PyObject **objs)
1207{
1208 if (!args) {
1209 if (!min && !max) {
1210 return 1;
1211 } else {
1212 PyErr_Format(PyExc_TypeError, "%s expected %s%d arguments, got none",
1213 name, (min == max ? "" : "at least "), (int)min);
1214 return 0;
1215 }
1216 }
1217 if (!PyTuple_Check(args)) {
1218 if (min <= 1 && max >= 1) {
1219 Py_ssize_t i;
1220 objs[0] = args;
1221 for (i = 1; i < max; ++i) {
1222 objs[i] = 0;
1223 }
1224 return 2;
1225 }
1226 PyErr_SetString(PyExc_SystemError, "UnpackTuple() argument list is not a tuple");
1227 return 0;
1228 } else {
1229 Py_ssize_t l = PyTuple_GET_SIZE(args);
1230 if (l < min) {
1231 PyErr_Format(PyExc_TypeError, "%s expected %s%d arguments, got %d",
1232 name, (min == max ? "" : "at least "), (int)min, (int)l);
1233 return 0;
1234 } else if (l > max) {
1235 PyErr_Format(PyExc_TypeError, "%s expected %s%d arguments, got %d",
1236 name, (min == max ? "" : "at most "), (int)max, (int)l);
1237 return 0;
1238 } else {
1239 Py_ssize_t i;
1240 for (i = 0; i < l; ++i) {
1241 objs[i] = PyTuple_GET_ITEM(args, i);
1242 }
1243 for (; l < max; ++l) {
1244 objs[l] = 0;
1245 }
1246 return i + 1;
1247 }
1248 }

1829SWIGRUNTIME void
1830SwigPyObject_dealloc(PyObject *v)
1831{
1832 SwigPyObject *sobj = (SwigPyObject *) v;
1833 PyObject *next = sobj->next;
1834 if (sobj->own == SWIG_POINTER_OWN) {
1835 swig_type_info *ty = sobj->ty;
1836 SwigPyClientData *data = ty ? (SwigPyClientData *) ty->clientdata : 0;
1837 PyObject *destroy = data ? data->destroy : 0;
1838 if (destroy) {
1839 /* destroy is always a VARARGS method */
1840 PyObject *res;
1841
1842 /* PyObject_CallFunction() has the potential to silently drop
1843 the active exception. In cases of unnamed temporary
1844 variable or where we just finished iterating over a generator
1845 StopIteration will be active right now, and this needs to
1846 remain true upon return from SwigPyObject_dealloc. So save
1847 and restore. */
1848
1849 PyObject *type = NULL, *value = NULL, *traceback = NULL;
1850 PyErr_Fetch(&type, &value, &traceback);
1851
1852 if (data->delargs) {
1853 /* we need to create a temporary object to carry the destroy operation */
1854 PyObject *tmp = SwigPyObject_New(sobj->ptr, ty, 0);
1855 if (tmp) {
1856 res = SWIG_Python_CallFunctor(destroy, tmp);
1857 } else {
1858 res = 0;
1859 }
1860 Py_XDECREF(tmp);
1861 } else {
1862 PyCFunction meth = PyCFunction_GET_FUNCTION(destroy);
1863 PyObject *mself = PyCFunction_GET_SELF(destroy);
1864 res = ((*meth)(mself, v));
1865 }
1866 if (!res)
1867 PyErr_WriteUnraisable(destroy);
1868
1869 PyErr_Restore(type, value, traceback);
1870
1871 Py_XDECREF(res);
1872 }
1873#if !defined(SWIG_PYTHON_SILENT_MEMLEAK)
1874 else {
1875 const char *name = SWIG_TypePrettyName(ty);
1876 printf("swig/python detected a memory leak of type '%s', no destructor found.\n", (name ? name : "unknown"));
1877 }
1878#endif
1879 Py_XDECREF(Swig_Capsule_global);
1880 }
1881 Py_XDECREF(next);
1882#ifdef SWIGPYTHON_BUILTIN
1883 Py_XDECREF(sobj->dict);
1884#endif
1885 PyObject_DEL(v);
1886}

2469SWIGRUNTIME int
2470SWIG_Python_ConvertPtrAndOwn(PyObject *obj, void **ptr, swig_type_info *ty, int flags, int *own) {
2471 int res;
2472 SwigPyObject *sobj;
2473 int implicit_conv = (flags & SWIG_POINTER_IMPLICIT_CONV) != 0;
2474
2475 if (!obj)
2476 return SWIG_ERROR;
2477 if (obj == Py_None && !implicit_conv) {
2478 if (ptr)
2479 *ptr = 0;
2480 return (flags & SWIG_POINTER_NO_NULL) ? SWIG_NullReferenceError : SWIG_OK;
2481 }
2482
2483 res = SWIG_ERROR;
2484
2485 sobj = SWIG_Python_GetSwigThis(obj);
2486 if (own)
2487 *own = 0;
2488 while (sobj) {
2489 void *vptr = sobj->ptr;
2490 if (ty) {
2491 swig_type_info *to = sobj->ty;
2492 if (to == ty) {
2493 /* no type cast needed */
2494 if (ptr) *ptr = vptr;
2495 break;
2496 } else {
2497 swig_cast_info *tc = SWIG_TypeCheck(to->name,ty);
2498 if (!tc) {
2499 sobj = (SwigPyObject *)sobj->next;
2500 } else {
2501 if (ptr) {
2502 int newmemory = 0;
2503 *ptr = SWIG_TypeCast(tc,vptr,&newmemory);
2504 if (newmemory == SWIG_CAST_NEW_MEMORY) {
2505 assert(own); /* badly formed typemap which will lead to a memory leak - it must set and use own to delete *ptr */
2506 if (own)
2507 *own = *own | SWIG_CAST_NEW_MEMORY;
2508 }
2509 }
2510 break;
2511 }
2512 }
2513 } else {
2514 if (ptr) *ptr = vptr;
2515 break;
2516 }
2517 }
2518 if (sobj) {
2519 if (((flags & SWIG_POINTER_RELEASE) == SWIG_POINTER_RELEASE) && !sobj->own) {
2520 res = SWIG_ERROR_RELEASE_NOT_OWNED;
2521 } else {
2522 if (own)
2523 *own = *own | sobj->own;
2524 if (flags & SWIG_POINTER_DISOWN) {
2525 sobj->own = 0;
2526 }
2527 if (flags & SWIG_POINTER_CLEAR) {
2528 sobj->ptr = 0;
2529 }
2530 res = SWIG_OK;
2531 }
2532 } else {
2533 if (implicit_conv) {
2534 SwigPyClientData *data = ty ? (SwigPyClientData *) ty->clientdata : 0;
2535 if (data && !data->implicitconv) {
2536 PyObject *klass = data->klass;
2537 if (klass) {
2538 PyObject *impconv;
2539 data->implicitconv = 1; /* avoid recursion and call 'explicit' constructors*/
2540 impconv = SWIG_Python_CallFunctor(klass, obj);
2541 data->implicitconv = 0;
2542 if (PyErr_Occurred()) {
2543 PyErr_Clear();
2544 impconv = 0;
2545 }
2546 if (impconv) {
2547 SwigPyObject *iobj = SWIG_Python_GetSwigThis(impconv);
2548 if (iobj) {
2549 void *vptr;
2550 res = SWIG_Python_ConvertPtrAndOwn((PyObject*)iobj, &vptr, ty, 0, 0);
2551 if (SWIG_IsOK(res)) {
2552 if (ptr) {
2553 *ptr = vptr;
2554 /* transfer the ownership to 'ptr' */
2555 iobj->own = 0;
2556 res = SWIG_AddCast(res);
2557 res = SWIG_AddNewMask(res);
2558 } else {
2559 res = SWIG_AddCast(res);
2560 }
2561 }
2562 }
2563 Py_DECREF(impconv);
2564 }
2565 }
2566 }
2567 if (!SWIG_IsOK(res) && obj == Py_None) {
2568 if (ptr)
2569 *ptr = 0;
2570 if (PyErr_Occurred())
2571 PyErr_Clear();

2573 }
2574 }
2575 }

2577}

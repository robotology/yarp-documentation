94void Quaternion::fromRotationMatrix(const yarp::sig::Matrix &R)
95{
96 if ((R.rows()<3) || (R.cols()<3))
97 {
98 yCError(QUATERNION, "fromRotationMatrix() failed, matrix should be >= 3x3");
99 yCAssert(QUATERNION, R.rows() >= 3 && R.cols() >= 3);
100 }
101
102 double tr = R(0, 0) + R(1, 1) + R(2, 2);
103
104 if (tr>0.0)
105 {
106 double sqtrp1 = sqrt(tr + 1.0);
107 double sqtrp12 = 2.0*sqtrp1;
108 q_w = 0.5*sqtrp1;
109 q_x = (R(2, 1) - R(1, 2)) / sqtrp12;
110 q_y = (R(0, 2) - R(2, 0)) / sqtrp12;
111 q_z = (R(1, 0) - R(0, 1)) / sqtrp12;
112 }
113 else if ((R(1, 1)>R(0, 0)) && (R(1, 1)>R(2, 2)))
114 {
115 double sqdip1 = sqrt(R(1, 1) - R(0, 0) - R(2, 2) + 1.0);
116 q_y = 0.5*sqdip1;
117
118 if (sqdip1 > 0.0) {
119 sqdip1 = 0.5 / sqdip1;
120 }
121
122 q_w = (R(0, 2) - R(2, 0))*sqdip1;
123 q_x = (R(1, 0) + R(0, 1))*sqdip1;
124 q_z = (R(2, 1) + R(1, 2))*sqdip1;
125 }
126 else if (R(2, 2)>R(0, 0))
127 {
128 double sqdip1 = sqrt(R(2, 2) - R(0, 0) - R(1, 1) + 1.0);
129 q_z = 0.5*sqdip1;
130
131 if (sqdip1 > 0.0) {
132 sqdip1 = 0.5 / sqdip1;
133 }
134
135 q_w = (R(1, 0) - R(0, 1))*sqdip1;
136 q_x = (R(0, 2) + R(2, 0))*sqdip1;
137 q_y = (R(2, 1) + R(1, 2))*sqdip1;
138 }
139 else
140 {
141 double sqdip1 = sqrt(R(0, 0) - R(1, 1) - R(2, 2) + 1.0);
142 q_x = 0.5*sqdip1;
143
144 if (sqdip1 > 0.0) {
145 sqdip1 = 0.5 / sqdip1;
146 }
147
148 q_w = (R(2, 1) - R(1, 2))*sqdip1;
149 q_y = (R(1, 0) + R(0, 1))*sqdip1;
150 q_z = (R(0, 2) + R(2, 0))*sqdip1;
151 }
152}

154yarp::sig::Matrix Quaternion::toRotationMatrix4x4() const
155{
156 yarp::sig::Vector q = this->toVector();
157 yarp::sig::Vector qin = (1.0 / yarp::math::norm(q))*q;
158
159 yarp::sig::Matrix R = yarp::math::eye(4, 4);
160 R(0, 0) = qin[0] * qin[0] + qin[1] * qin[1] - qin[2] * qin[2] - qin[3] * qin[3];
161 R(1, 0) = 2.0*(qin[1] * qin[2] + qin[0] * qin[3]);
162 R(2, 0) = 2.0*(qin[1] * qin[3] - qin[0] * qin[2]);
163 R(0, 1) = 2.0*(qin[1] * qin[2] - qin[0] * qin[3]);
164 R(1, 1) = qin[0] * qin[0] - qin[1] * qin[1] + qin[2] * qin[2] - qin[3] * qin[3];
165 R(2, 1) = 2.0*(qin[2] * qin[3] + qin[0] * qin[1]);
166 R(0, 2) = 2.0*(qin[1] * qin[3] + qin[0] * qin[2]);
167 R(1, 2) = 2.0*(qin[2] * qin[3] - qin[0] * qin[1]);
168 R(2, 2) = qin[0] * qin[0] - qin[1] * qin[1] - qin[2] * qin[2] + qin[3] * qin[3];
169
170 return R;
171}

173yarp::sig::Matrix Quaternion::toRotationMatrix3x3() const
174{
175 yarp::sig::Vector q = this->toVector();
176 yarp::sig::Vector qin = (1.0 / yarp::math::norm(q))*q;
177
178 yarp::sig::Matrix R = yarp::math::zeros(3,3);
179 R(0, 0) = qin[0] * qin[0] + qin[1] * qin[1] - qin[2] * qin[2] - qin[3] * qin[3];
180 R(1, 0) = 2.0*(qin[1] * qin[2] + qin[0] * qin[3]);
181 R(2, 0) = 2.0*(qin[1] * qin[3] - qin[0] * qin[2]);
182 R(0, 1) = 2.0*(qin[1] * qin[2] - qin[0] * qin[3]);
183 R(1, 1) = qin[0] * qin[0] - qin[1] * qin[1] + qin[2] * qin[2] - qin[3] * qin[3];
184 R(2, 1) = 2.0*(qin[2] * qin[3] + qin[0] * qin[1]);
185 R(0, 2) = 2.0*(qin[1] * qin[3] + qin[0] * qin[2]);
186 R(1, 2) = 2.0*(qin[2] * qin[3] - qin[0] * qin[1]);
187 R(2, 2) = qin[0] * qin[0] - qin[1] * qin[1] - qin[2] * qin[2] + qin[3] * qin[3];
188
189 return R;
190}

192std::string Quaternion::toString(int precision, int width) const
193{
194 std::string ret;
195 char tmp[350];
196 if (width<0)
197 {
198 sprintf(tmp, "w=% .*lf\t", precision, q_w); ret += tmp;
199 sprintf(tmp, "x=% .*lf\t", precision, q_x); ret += tmp;
200 sprintf(tmp, "y=% .*lf\t", precision, q_y); ret += tmp;
201 sprintf(tmp, "z=% .*lf\t", precision, q_z); ret += tmp;
202 }
203 else
204 {
205 sprintf(tmp, "w=% *.*lf ", width, precision, q_w); ret += tmp;
206 sprintf(tmp, "x=% *.*lf ", width, precision, q_x); ret += tmp;
207 sprintf(tmp, "y=% *.*lf ", width, precision, q_y); ret += tmp;
208 sprintf(tmp, "z=% *.*lf ", width, precision, q_z); ret += tmp;
209 }
210
211 return ret.substr(0, ret.length() - 1);
212}

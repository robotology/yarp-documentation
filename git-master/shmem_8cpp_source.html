30// Replies to each incoming Bottle with the same Bottle, prefixed by "echo"
31class EchoReader : public PortReader
32{
33public:
34 bool read(ConnectionReader& connection) override
35 {
36 Bottle in;
37 if (!in.read(connection)) {
38 return false;
39 }
40 Bottle out;
41 out.addString("echo");
42 out.append(in);
43 ConnectionWriter* writer = connection.getWriter();
44 if (writer != nullptr) {
45 out.write(*writer);
46 }
47 return true;
48 }
49};

51// Wait until the connection is reported (or give up after a timeout)
52bool waitConnection(const std::string& src, const std::string& dest, bool connected)
53{
54 for (int i = 0; i < 50; ++i) {
55 if (Network::isConnected(src, dest, "shmem", true) == connected) {
56 return true;
57 }
58 Time::delay(0.1);
59 }
60 return false;
61}

63// Run the shmem_peer helper in a different process, asynchronously.
64// The result is the exit code of the helper.
65std::future<int> runPeer(const std::string& mode, const Contact& rpc, const Contact& stream)
66{
67 std::string cmd = std::string("\"") + SHMEM_PEER_PATH + "\" " + mode + " " + rpc.getHost() + " "
68 + std::to_string(rpc.getPort()) + " " + std::to_string(stream.getPort());
69#if defined(_WIN32)
70 // cmd.exe strips the outer quotes
71 cmd = "\"" + cmd + "\"";
72#endif
73 return std::async(std::launch::async, [cmd]() {
74 int ret = std::system(cmd.c_str());
75#if !defined(_WIN32)
76 if (ret != -1 && WIFEXITED(ret)) {
77 ret = WEXITSTATUS(ret);
78 }
79#endif
80 return ret;
81 });
82}

92 SECTION("test small messages")
93 {
94 BufferedPort<Bottle> in;

105 b.clear();

107 b.addString("hello");
108 b.addFloat64(i * 0.5);
109 out.writeStrict();
110
111 Bottle* r = in.read();
112 REQUIRE(r != nullptr);
113 REQUIRE(r->size() == 3);
114 CHECK(r->get(0).asInt32() == i);
115 CHECK(r->get(1).asString() == "hello");

120 in.interrupt();
121 out.close();
122 in.close();
123 }

125 SECTION("test increasing message size (shared memory resize)")
126 {
127 BufferedPort<Bottle> in;
128 BufferedPort<Bottle> out;
129 in.setStrict();
130
131 REQUIRE(in.open("/shmem/in"));
132 REQUIRE(out.open("/shmem/out"));
133 REQUIRE(Network::connect(out.getName(), in.getName(), "shmem"));
134 REQUIRE(waitConnection(out.getName(), in.getName(), true));
135
136 // The default buffer is SHMEM_DEFAULT_SIZE (4096) bytes, larger
137 // messages force the shared memory to be reallocated
138 for (size_t len : {10, 1000, 4000, 5000, 20000, 100000, 1000000}) {
139 std::string payload(len, ' ');
140 for (size_t j = 0; j < len; ++j) {
141 payload[j] = static_cast<char>('a' + (j % 26));
142 }
143
144 Bottle& b = out.prepare();
145 b.clear();
146 b.addInt32(static_cast<int>(len));
147 b.addString(payload);
148 out.writeStrict();
149
150 Bottle* r = in.read();
151 REQUIRE(r != nullptr);
152 REQUIRE(r->size() == 2);
153 CHECK(r->get(0).asInt32() == static_cast<int>(len));
154 CHECK(r->get(1).asString() == payload);
155 }
156
157 // Small messages must still work after the resize
158 Bottle& b = out.prepare();
159 b.clear();
160 b.addString("after resize");
161 out.writeStrict();
162 Bottle* r = in.read();
163 REQUIRE(r != nullptr);
164 CHECK(r->get(0).asString() == "after resize");
165
166 out.interrupt();
167 in.interrupt();
168 out.close();
169 in.close();
170 }

241// The tests above run both sides of the connection in the same process. These
242// ones run the other side in a different process (see shmem_peer.cpp), so
243// that the inter-process objects (shared memory, semaphores) are really shared
244// between two processes.
245TEST_CASE("carriers::shmem::multiprocess", "[carriers]")
246{
247 YARP_REQUIRE_PLUGIN("shmem", "carrier");
248
249 // The peer addresses the ports directly by host and port number, it does
250 // not need to access the name server
251 Network::setLocalMode(true);
252
253 SECTION("rpc and streaming with another process")
254 {
255 Port server;
256 EchoReader echo;
257 server.setReader(echo);
258 BufferedPort<Bottle> in;
259 in.setStrict();
260
261 REQUIRE(server.open("/shmem/xproc/server"));
262 REQUIRE(in.open("/shmem/xproc/in"));
263
264 auto peer = runPeer("normal", server.where(), in.where());
265
266 // The peer performs the rpc calls (checking the replies) and then
267 // streams 200 messages
268 int received = 0;
269 size_t bytes = 0;
270 bool order_ok = true;
271 double start = Time::now();
272 while (received < 200 && Time::now() - start < 60.0) {
273 Bottle* b = in.read(false);
274 if (b == nullptr) {
275 if (peer.wait_for(std::chrono::seconds(0)) == std::future_status::ready && in.getPendingReads() == 0) {
276 break;
277 }
278 Time::delay(0.01);
279 continue;
280 }
281 if (b->get(0).asInt32() != received) {
282 order_ok = false;
283 }
284 bytes += b->get(1).asString().size();
285 ++received;
286 }
287
288 CHECK(peer.get() == 0);
289 CHECK(received == 200);
290 CHECK(order_ok);
291 CHECK(bytes == 196 * 100 + 4 * 500000);
292
293 in.close();
294 server.close();
295 }
296
297 SECTION("the other process crashes")
298 {
299 BufferedPort<Bottle> in;
300 in.setStrict();
301 REQUIRE(in.open("/shmem/xproc/in"));
302
303 auto peer = runPeer("crash", in.where(), in.where());
304
305 Bottle* b = in.read();
306 REQUIRE(b != nullptr);
307 CHECK(b->get(0).asString() == "bye");
308 CHECK(peer.get() == 0);
309
310 // The peer exited without closing the connection, this must be
311 // detected and the connection removed
312 bool removed = false;
313 for (int i = 0; i < 100 && !removed; ++i) {
314 removed = (in.getInputCount() == 0);
315 Time::delay(0.1);
316 }
317 CHECK(removed);
318
319 in.close();
320 }
321
322 Network::setLocalMode(false);

108 {
109 yarp::dev::IEncoders* ienc = nullptr;
110 dd_fake.view(ienc);
111 REQUIRE(ienc);
112 REQUIRE(ienc->getAxes(nAxes));

116 //talk to the RPC port, like `yarp rpc` does
117 yarp::os::Port rpc;
118 REQUIRE(rpc.open("/test/controlboard/rpc"));
119 REQUIRE(yarp::os::Network::connect(rpc.getName(), "/controlboard/nws/rpc:i"));
120
121 ControlBoardMsgs rpc_client;
122 REQUIRE(rpc_client.yarp().attachAsClient(rpc));
123
124 //set a known mode on axis 0 through the existing RPC setter
125 CHECK(rpc_client.setControlModeOneRPC(0, yarp::dev::SelectableControlModeEnum::VOCAB_CM_POSITION_DIRECT) == ReturnValue_ok);
126
127 //read it back through the new RPC getters
128 {
129 return_getControlModeOne reply = rpc_client.getControlModeOneRPC(0);
130 CHECK(reply.ret == ReturnValue_ok);
131 CHECK(reply.mode == yarp::dev::ControlModeEnum::VOCAB_CM_POSITION_DIRECT);
132 }
133
134 {
135 return_getControlModeAll reply = rpc_client.getControlModeAllRPC();
136 CHECK(reply.ret == ReturnValue_ok);
137 CHECK(reply.modes.size() == nAxes);
138 CHECK(reply.modes[0] == yarp::dev::ControlModeEnum::VOCAB_CM_POSITION_DIRECT);
139 }
140
141 {
142 return_getControlModeGroup reply = rpc_client.getControlModeGroupRPC({0, 1});
143 CHECK(reply.ret == ReturnValue_ok);
144 CHECK(reply.modes.size() == 2);
145 CHECK(reply.modes[0] == yarp::dev::ControlModeEnum::VOCAB_CM_POSITION_DIRECT);
146 }
147
148 rpc.close();
149

162bool SDLJoypad::parseStickInfo(const yarp::os::Searchable& cfg)
163{
164 if(!cfg.check("sticks") || !cfg.find("sticks").isInt32())
165 {
166 yCError(SDLJOYPAD) << "Missing 'sticks' parameter or not an integer";
167 return false;
168 }
169
170 for(size_t i = 0; i < m_axisCount; i++)
171 {
172 m_axes.push_back(true);
173 }
174
175 m_stickCount = cfg.find("sticks").asInt32();
176 for(size_t i = 0; i < m_stickCount; i++)
177 {
178 std::string stickName;
179 int axesCount;
180 stick currentStick;
181
182 stickName = "STICK"+std::to_string(i);
183
184 if(!cfg.check(stickName))
185 {
186 yCError(SDLJOYPAD) << "Missing" << stickName << "group in configuration";
187 return false;
188 }
189
190 Bottle& stickParams = cfg.findGroup(stickName);
191
192 if(0 == stickParams.size())
193 {
194 yCError(SDLJOYPAD) << "Group" << stickName << "is empty";
195 return false;
196 }
197
198 if(!stickParams.check("axes") || !stickParams.find("axes").isInt32())
199 {
200 yCError(SDLJOYPAD) << "Missing 'axes' count in" << stickName << "group or not an integer";
201 return false;
202 }
203
204 axesCount = stickParams.find("axes").asInt32();
205
206 for(int j = 0; j < axesCount; j++)
207 {
208 std::string axisName, invertName;
209 size_t axis_id;
210 axisName = "axis" + std::to_string(j) + "_id";
211 invertName = "invert_axis_" + std::to_string(j);
212
213 if(!stickParams.check(axisName) || !stickParams.find(axisName).isInt32())
214 {
215 yCError(SDLJOYPAD) << "Missing" << axisName << "param in" << stickName << "group or not an integer.";
216 return false;
217 }
218
219 axis_id = (size_t)stickParams.find(axisName).asInt32();
220 if(axis_id > m_axisCount - 1)
221 {
222 yCError(SDLJOYPAD) << "Axis id out of bound";
223 return false;
224 }
225
226 if(!stickParams.check(invertName) || !stickParams.find(invertName).isBool())
227 {
228 yCError(SDLJOYPAD) << "Missing" << invertName << "param in" << stickName << "group or not an bool.";
229 return false;
230 }
231
232 currentStick.axes_ids.push_back(axis_id);
233 currentStick.direction.push_back(stickParams.find(invertName).asBool() ? -1 : 1);
234 m_axes[axis_id] = false;
235 }
236
237 if(!stickParams.check("deadZone") || !stickParams.find("deadZone").isFloat64())
238 {
239 yCError(SDLJOYPAD) << "Missing deadZone param in" << stickName << "group or not an double.";
240 return false;
241 }
242
243 currentStick.deadZone = stickParams.find("deadZone").asFloat64();
244 m_sticks.push_back(currentStick);
245 }
246 return true;
247}

290ReturnValue SDLJoypad::getButton(size_t button_id, double& value)
291{
292 if(button_id > m_buttonCount - 1)
293 {
294 yCError(SDLJOYPAD) << "Button id out of bound!";
295 return ReturnValue_error_input_out_of_bounds;
296 }
297 updateJoypad();
298 size_t i;
299 for(i = 0; i < m_device.size(); ++i)
300 {
301 size_t localCount = SDL_JoystickNumButtons(m_device[i]);
302 if(button_id > localCount - 1)
303 {
304 button_id -= localCount;
305 }
306 else
307 {
308 break;
309 }
310 }
311
312 value = float(SDL_JoystickGetButton(m_device[i], button_id));
313 if(value > 0.5 && m_actionsHandler.actionDetected(button_id) &&
314 yarp::os::Time::now() - m_actionTimestamp > actionsExecutionTime)
315 {
316 m_actionsHandler.executeAction(button_id);
317 m_actionTimestamp = yarp::os::Time::now();
318 }
319 return ReturnValue_ok;
320}

369ReturnValue SDLJoypad::getStick(size_t stick_id, yarp::dev::StickData& value, JoypadCtrl_coordinateMode coordinate_mode)
370{
371 if (stick_id > m_stickCount - 1)
372 {
373 yCError(SDLJOYPAD) << "Stick id out of bound!";
374 return ReturnValue_error_input_out_of_bounds;
375 }
376 updateJoypad();
377 stick& stk_config = m_sticks[stick_id];
378 double val;
379
380 if (stk_config.axes_ids.size() > 2)
381 {
382 yCError(SDLJOYPAD) << "Only for 2-dimensional sticks are supported at the moment";
383 return ReturnValue_error_not_implemented_by_device;
384 }
385
386 for(size_t i = 0; i < stk_config.axes_ids.size(); i++)
387 {
388 if(!getAxis(stk_config.axes_ids[i], val)) {return ReturnValue_error_method_failed;}
389 value.s1=(val * stk_config.direction[i] * (fabs(val) > stk_config.deadZone));
390 value.s2=(val * stk_config.direction[i] * (fabs(val) > stk_config.deadZone));
391 }
392
393 if (coordinate_mode == JoypadCtrl_coordinateMode::JypCtrlcoord_POLAR)
394 {
395 double v0= value.s1;
396 double v1= value.s2;
397 value.s1 = sqrt(v0 * v0 + v1 * v1);
398 value.s2 = atan2(v0, v1);
399 }
400 return ReturnValue_ok;
401}

490bool JoystickActionsHandler::parseActions(const yarp::os::Searchable& cfg, size_t maxButtons, int* count)
491{
492 int dummy = 0;
493 size_t i;
494 int& actCount = count ? *count : dummy;
495 if(!cfg.check(buttActionGroupName))
496 {
497 yCInfo(SDLJOYPAD) << "no actions found in the configuration file (no" << buttActionGroupName << "group found)";
498 actCount = 0;
499 return true;
500 }
501 Bottle& actionsGroup = cfg.findGroup(buttActionGroupName);
502
503 if(!actionsGroup.size())
504 {
505 yCError(SDLJOYPAD) << "no action found under" << buttActionGroupName << "group";
506 actCount = 0;
507 return false;
508 }
509
510 for(i = 1; i < actionsGroup.size(); i++)
511 {
512 if(!actionsGroup.get(i).isList())
513 {
514 yDebug() << "error parsing cfg";
515 return false;
516 }
517
518 Bottle& keyvalue = *actionsGroup.get(i).asList();
519 yDebug() << keyvalue.toString();
520 if(maxButtons==0)
521 {
522 yCError(SDLJOYPAD) << "invalid button count";
523 actCount = 0;
524 return false;
525 }
526 if(!keyvalue.get(0).isInt32() ||
527 keyvalue.get(0).asInt32() < 0 ||
528 (unsigned int) keyvalue.get(0).asInt32() > maxButtons-1 ||
529 !keyvalue.get(1).isString())
530 {
531 yCError(SDLJOYPAD) << "Button's actions parameters must be in the format 'unsigned int string' and the button id must be in range";
532 actCount = 0;
533 return false;
534 }
535 yCInfo(SDLJOYPAD) << "assigning actions" << keyvalue.get(1).asString() << "to button" << keyvalue.get(0).asInt32();
536 m_actions[keyvalue.get(0).asInt32()] = keyvalue.get(1).asString();
537 }
538
539 actCount = i;
540 yCInfo(SDLJOYPAD) << actCount << "action parsed successfully";
541 return true;
542}

278bool JoypadControl_nws_yarp::attach(PolyDriver* poly)
279{
280 if (poly) {
281 poly->view(m_IJoypad);
282 }
283
284 if(m_IJoypad == nullptr)
285 {
286 yCError(JOYPADCONTROLSERVER) << "Attached device has no valid IJoypadController interface.";
287 return false;
288 }
289
290 if (!openPorts())
291 {
292 yCError(JOYPADCONTROLSERVER) << "Unable to open ports";
293 return false;
294 }
295
296 m_RPCMsgs = std::make_unique<IJoypadControlRPCd>(this->m_IJoypad);
297
298 PeriodicThread::setPeriod(m_period);
299 if (!PeriodicThread::start()) {
300 return false;
301 }
302
303 return true;

348bool JoypadControl_nws_yarp::openPorts()
349{
350 if(!m_IJoypad)
351 {
352 return false;
353 }
354
355 if (!m_rpcPort.open(m_name + "/rpc:i"))
356 {
357 yCError(JOYPADCONTROLSERVER) << "Unable to open rpc Port" << (m_name+"/rpc:o");
358 return false;
359 }
360 m_rpcPort.setReader(*this);
361
362 bool b=true;
363 b &= m_portAxes.open(m_name+"/axis:o");
364 b &= m_portButtons.open(m_name+"/buttons:o");
365 b &= m_portSticks.open(m_name+"/stick:o");
366 b &= m_portTouches.open(m_name+"/touch:o");
367 b &= m_portTrackballs.open(m_name+"/trackball:o");
368 b &= m_portHats.open(m_name+"/hat:o");
369 b &= m_allPort.open(m_name+"/allJoyData:o");
370 if (!b)
371 {
372 yCError(JOYPADCONTROLSERVER) << "Unable to streaming ports";
373 return false;
374 }
375
376 return true;
377}

379template <class T>
380bool VectorsDifferent(const std::vector<T>& a, const std::vector<T>& b)
381{
382 return a.size() != b.size() ||
383 !std::equal(a.begin(), a.end(), b.begin());
384}

387template <class T, class Compare>
388bool VectorsDifferentC(const std::vector<T>& a, const std::vector<T>& b, Compare comp)
389{
390 return a.size() != b.size() ||
391 !std::equal(a.begin(), a.end(), b.begin(), comp);
392}

407void JoypadControl_nws_yarp::run()
408{
409 std::string profile_message;
410
411 //1 Buttons
412 {
413 bool write;
414 write = m_portButtons.getOutputCount()>0;
415 auto& b = m_portButtons.prepare();
416 b.value.clear();
417 for(size_t i = 0; i < m_numberOfButtons; ++i)
418 {
419 if (!m_IJoypad->getButton(i, m_curr_buttons[i]))
420 {
421 write = false;
422 break;
423 }
424 }
425
426 if (write)
427 {
428 if (m_publish_on_event)
429 {
430 if (VectorsDifferent(m_curr_buttons,m_last_buttons))
431 {
432 b.value = m_curr_buttons;
433 m_portButtons.write();
434 m_last_buttons = m_curr_buttons;
435 }
436 }
437 else
438 {
439 b.value = m_curr_buttons;
440 m_portButtons.write();
441 m_last_buttons = m_curr_buttons;
442 }
443 }
444 }
445
446 //2 Hats
447 {
448 bool write;
449
450 write = m_portHats.getOutputCount()>0;
451 auto& b = m_portHats.prepare();
452 b.value.clear();
453 for(size_t i = 0; i < m_numberOfHats; ++i)
454 {
455 if (!m_IJoypad->getHat(i, m_curr_hats[i]))
456 {
457 write = false;
458 break;
459 }
460 }
461
462 if (write)
463 {
464 if (m_publish_on_event)
465 {
466 if (VectorsDifferent(m_curr_hats,m_last_hats))
467 {
468 b.value.resize(m_curr_hats.size());
469 std::transform(m_curr_hats.begin(), m_curr_hats.end(), b.value.begin(), [](unsigned char c) { return static_cast<char>(c); });
470 //b.value = m_curr_hats; //using std::transform instead
471 m_portHats.write();
472 m_last_hats = m_curr_hats;
473 }
474 }
475 else
476 {
477 b.value.resize(m_curr_hats.size());
478 std::transform(m_curr_hats.begin(), m_curr_hats.end(), b.value.begin(), [](unsigned char c) { return static_cast<char>(c); });
479 //b.value = m_curr_hats; //using std::transform instead
480 m_portHats.write();
481 m_last_hats = m_curr_hats;
482 }
483 }
484 }
485
486 //3 Axis
487 {
488 bool write;
489
490 write = m_portAxes.getOutputCount()>0;
491 auto& b = m_portAxes.prepare();
492 b.value.clear();
493 std::vector<double> tmp_axes;
494 if (!m_IJoypad->getAllAxes(tmp_axes) || tmp_axes.size() != m_numberOfAxes)
495 {
496 write = false;
497 }
498 else
499 {
500 m_curr_axes = tmp_axes;
501 }
502
503 if (write)
504 {
505 if (m_publish_on_event)
506 {
507 if (VectorsDifferent(m_curr_axes,m_last_axes))
508 {
509 b.value = m_curr_axes;
510 m_portAxes.write();
511 m_last_axes = m_curr_axes;
512 }
513 }
514 else
515 {
516 b.value = m_curr_axes;
517 m_portAxes.write();
518 m_last_axes = m_curr_axes;
519 }
520 }
521 }
522
523 //4 Trackball
524 {
525 bool write;
526
527 write = m_portTrackballs.getOutputCount()>0;
528 auto& b = m_portTrackballs.prepare();
529 b.value.clear();
530 for(size_t i = 0; i < m_numberOfTrackballs; ++i)
531 {
532 if (!m_IJoypad->getTrackball(i, m_curr_trackballs[i]))
533 {
534 write = false;
535 break;
536 }
537 }
538
539 if (write)
540 {
541 if (m_publish_on_event)
542 {
543 if (VectorsDifferentC(m_curr_trackballs,m_last_trackballs,
544 [](const yarp::dev::TrackballData& lhs,
545 const yarp::dev::TrackballData& rhs)
546 {
547 return lhs.x == rhs.x &&
548 lhs.y == rhs.y;
549 }
550 ))
551 {
552 b.value = m_curr_trackballs;
553 m_portTrackballs.write();
554 m_last_trackballs = m_curr_trackballs;
555 }
556 }
557 else
558 {
559 b.value = m_curr_trackballs;
560 m_portTrackballs.write();
561 m_last_trackballs = m_curr_trackballs;
562 }
563 }
564 }
565
566 //5 Stick
567 {
568 bool write;
569 write = m_portSticks.getOutputCount()>0;
570 auto& b = m_portSticks.prepare();
571 b.value.clear();
572 for(size_t i = 0; i < m_numberOfSticks; ++i)
573 {
574 if (!m_IJoypad->getStick(i, m_curr_sticks[i], m_coordsMode))
575 {
576 write = false;
577 break;
578 }
579 }
580
581 if (write)
582 {
583 if (m_publish_on_event)
584 {
585 if (VectorsDifferentC(m_curr_sticks,m_last_sticks,
586 [](const yarp::dev::StickData& lhs,
587 const yarp::dev::StickData& rhs)
588 {
589 return lhs.s1 == rhs.s1 &&
590 lhs.s2 == rhs.s2;
591 }
592 ))
593 {
594 b.value = m_curr_sticks;
595 m_portSticks.write();
596 m_last_sticks = m_curr_sticks;
597 }
598 }
599 else
600 {
601 b.value = m_curr_sticks;
602 m_portSticks.write();
603 m_last_sticks = m_curr_sticks;
604 }
605 }
606 }
607
608 //6 Touch
609 {
610 bool write;
611 write = m_portTouches.getOutputCount()>0;
612 auto& b = m_portTouches.prepare();
613 b.value.clear();
614
615 m_curr_touches.value.resize(m_numberOfTouchSurfaces);
616 for(unsigned int i = 0; i < m_numberOfTouchSurfaces; ++i)
617 {
618 std::vector<yarp::dev::TouchData> temp;
619 if (!m_IJoypad->getTouch(i, temp))
620 {
621 write = false;
622 break;
623 }
624 m_curr_touches.value[i].touches = temp;
625 }
626
627 if (write)
628 {
629 if (m_publish_on_event)
630 {
631 if (1)
632 //if (VectorsDifferent(m_curr_touches.value,m_last_touches.value))
633 {
634 b.value = m_curr_touches.value;
635 m_portTouches.write();
636 m_last_touches = m_curr_touches;
637 }
638 }
639 else
640 {
641 b.value = m_curr_touches.value;
642 m_portTouches.write();
643 m_last_touches = m_curr_touches;
644 }
645 }
646 }
647
648
649 //AllJoyDataPort
650 if (m_allPort.getOutputCount()>0)
651 {
652 yarp::dev::AllJoyData& message = m_allPort.prepare();
653 message.AxisDataVal = m_curr_axes;
654 message.ButtonDataVal = m_curr_buttons;
655 message.HatsDataVal = std::vector<std::int8_t>(m_curr_hats.begin(), m_curr_hats.end());
656 message.StickDataVal = m_curr_sticks;
657 message.TrackballDataVal = m_curr_trackballs;
658 message.TouchDataVal = m_curr_touches.value;
659 m_allPort.write();
660 }
661
662 //print data on screen
663 if(m_profile)
664 {
665 profile_message = " Axes: ";
666 profile_message += printVector(m_curr_axes);
667 profile_message += " Buttons: ";
668 profile_message += printVector(m_curr_buttons);
669 //profile_message += " Trackballs: ";
670 //profile_message += printVector(m_curr_trackballs);
671 //profile_message += " Sticks: ";
672 //profile_message += printVector(m_curr_sticks);
673 //profile_message += " Hats: ";
674 //profile_message += printVector(m_curr_hats);
675 //profile_message += " Touch Surface: ";
676 //profile_message += printVector(m_curr_touches);
677 yCInfo(JOYPADCONTROLSERVER) << profile_message;
678 }
679}

70void StateExtendedInputPort::onRead(yarp::dev::JointStateData &v)
71{
72 now=Time::now();
73 mutex.lock();
74
75 if (count>0)
76 {
77 double tmpDT=now-prev;
78 deltaT+=tmpDT;
79 if (tmpDT > deltaTMax) {
80 deltaTMax = tmpDT;
81 }
82 if (tmpDT < deltaTMin) {
83 deltaTMin = tmpDT;
84 }
85 }
86
87 prev=now;
88 count++;
89
90 valid=true;
91 last=v;
92 getEnvelope(lastStamp);
93 //check that timestamp are available
94 if (!lastStamp.isValid()) {
95 lastStamp.update(now);
96 }
97 mutex.unlock();
98}

104bool StateExtendedInputPort::getLastSingle(int j, int field, double *data, Stamp &stamp, double &localArrivalTime)
105{
106 mutex.lock();
107 bool ret = valid;
108 if (ret)
109 {
110 switch(field)
111 {
112 case VOCAB_ENCODER:
113 *data = last.jointPosition[j];
114 ret = last.jointPosition_isValid;
115 break;
116
117 case VOCAB_ENCODER_SPEED:
118 ret = last.jointVelocity_isValid;
119 *data = last.jointVelocity[j];
120 break;
121
122 case VOCAB_ENCODER_ACCELERATION:
123 ret = last.jointAcceleration_isValid;
124 *data = last.jointAcceleration[j];
125 break;
126
127 case VOCAB_MOTOR_ENCODER:
128 ret = last.motorPosition_isValid;
129 *data = last.motorPosition[j];
130 break;
131
132 case VOCAB_MOTOR_ENCODER_SPEED:
133 ret = last.motorVelocity_isValid;
134 *data = last.motorVelocity[j];
135 break;
136
137 case VOCAB_MOTOR_ENCODER_ACCELERATION:
138 ret = last.motorAcceleration_isValid;
139 *data = last.motorAcceleration[j];
140 break;
141
142 case VOCAB_TRQ:
143 ret = last.torque_isValid;
144 *data = last.torque[j];
145 break;
146
147 case VOCAB_PWMCONTROL_PWM_OUTPUT:
148 ret = last.pwmDutycycle_isValid;
149 *data = double(last.pwmDutycycle[j]);
150 break;
151
152 case VOCAB_AMP_CURRENT:
153 ret = last.current_isValid;
154 *data = last.current[j];
155 break;
156
157 case VOCAB_TEMPERATURE:
158 ret = last.temperature_isValid;
159 *data = double(last.temperature[j]);
160 break;
161
162
163 default:
164 yCError(CONTROLBOARD_NWC_YARP) << "Internal error while reading data. Cannot get 'single' data of type " << yarp::os::Vocab32::decode(field);
165 break;
166 }
167
168 localArrivalTime=now;
169 stamp = lastStamp;
170 double current_time = Time::now();
171 if (ret && ((current_time - localArrivalTime) > timeout)) {
172 ret = false;
173 }
174 }
175 mutex.unlock();
176
177 return ret;
178}

213bool StateExtendedInputPort::getLastVector(int field, double* data, Stamp& stamp, double& localArrivalTime)
214{
215 mutex.lock();
216 bool ret = valid;
217 if (ret)
218 {
219 switch(field)
220 {
221 case VOCAB_ENCODERS:
222 ret = last.jointPosition_isValid;
223 memcpy(data, last.jointPosition.data(), last.jointPosition.size() * last.jointPosition.getElementSize() );
224 break;
225
226 case VOCAB_ENCODER_SPEEDS:
227 ret = last.jointVelocity_isValid;
228 memcpy(data, last.jointVelocity.data(), last.jointVelocity.size() * last.jointVelocity.getElementSize() );
229 break;
230
231 case VOCAB_ENCODER_ACCELERATIONS:
232 ret = last.jointAcceleration_isValid;
233 memcpy(data, last.jointAcceleration.data(), last.jointAcceleration.size() * last.jointAcceleration.getElementSize() );
234 break;
235
236 case VOCAB_MOTOR_ENCODERS:
237 ret = last.motorPosition_isValid;
238 memcpy(data, last.motorPosition.data(), last.motorPosition.size() * last.motorPosition.getElementSize() );
239 break;
240
241 case VOCAB_MOTOR_ENCODER_SPEEDS:
242 ret = last.motorVelocity_isValid;
243 memcpy(data, last.motorVelocity.data(), last.motorVelocity.size() * last.motorVelocity.getElementSize() );
244 break;
245
246 case VOCAB_MOTOR_ENCODER_ACCELERATIONS:
247 ret = last.motorAcceleration_isValid;
248 memcpy(data, last.motorAcceleration.data(), last.motorAcceleration.size() * last.motorAcceleration.getElementSize() );
249 break;
250
251 case VOCAB_TRQS:
252 ret = last.torque_isValid;
253 memcpy(data, last.torque.data(), last.torque.size() * last.torque.getElementSize() );
254 break;
255
256 case VOCAB_PWMCONTROL_PWM_OUTPUTS:
257 ret = last.pwmDutycycle_isValid;
258
259 // In the interface the pwmDutycycle is a double, but in the jointData it is a float so we need to copy it manually
260 for (size_t i = 0; i < last.pwmDutycycle.size(); i++) {
261 data[i] = last.pwmDutycycle[i];
262 }
263 break;
264
265 case VOCAB_AMP_CURRENTS:
266 ret = last.current_isValid;
267 memcpy(data, last.current.data(), last.current.size() * last.current.getElementSize());
268 break;
269
270 case VOCAB_TEMPERATURES:
271 ret = last.temperature_isValid;
272
273 // In the interface the temperature is a double, but in the jointData it is a float so we need to copy it manually
274 for (size_t i = 0; i < last.temperature.size(); i++) {
275 data[i] = last.temperature[i];
276 }
277 break;
278
279 default:
280 yCError(CONTROLBOARD_NWC_YARP) << "Internal error while reading data. Cannot get 'vector' data of type " << yarp::os::Vocab32::decode(field);
281 break;
282 }
283
284 localArrivalTime=now;
285 stamp = lastStamp;
286 if (ret && ((Time::now() - localArrivalTime) > timeout)) {
287 ret = false;
288 }
289 }
290 mutex.unlock();
291
292 return ret;
293}

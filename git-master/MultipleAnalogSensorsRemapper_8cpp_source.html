98// Return an empty list if the key is not found, and an error (false) if the key was found but it is not a list of strings
99bool getVectorOfStringFromListInConfig(const std::string& key, const yarp::os::Searchable& config, std::vector<std::string> & vectorOfStrings)
100{
101 yarp::os::Property prop;
102 prop.fromString(config.toString());
103 bool keyExists = prop.check(key);
104
105 yarp::os::Bottle *propList=prop.find(key).asList();
106 if (!propList && keyExists)
107 {
108 yCError(MULTIPLEANALOGSENSORSREMAPPER) << "Error parsing parameters: if present " << key << " should be followed by a list of strings.\n";
109 return false;
110 }
111
112 if (!propList && !keyExists)
113 {
114 vectorOfStrings.resize(0);
115 return true;
116 }
117
118 vectorOfStrings.resize(propList->size());
119 for (size_t ax=0; ax < propList->size(); ax++)
120 {
121 vectorOfStrings[ax] = propList->get(ax).asString();
122 }
123
124 return true;
125}

127bool MultipleAnalogSensorsRemapper::parseOptions(const Property& prop)
128{
129 bool ok = true;
130
131 m_remappedSensors.resize(MAS_NrOfSensorTypes);
132
133 for (size_t i = 0; i < MAS_NrOfSensorTypes; i++)
134 {
135 auto sensType = static_cast<MAS_SensorType>(i);
136 std::string optionName = MAS_getTagFromEnum(sensType) +"Names";
137 ok = getVectorOfStringFromListInConfig(optionName , prop, m_remappedSensors[i]);
138 if (!ok)
139 {
140 yCError(MULTIPLEANALOGSENSORSREMAPPER) << optionName << "should be followed by a list of string.";
141 return false;
142 }
143 }
144
145 return ok;
146}

153template<typename Interface>
154bool MultipleAnalogSensorsRemapper::genericAttachAll(const MAS_SensorType sensorType,
155 std::vector<Interface *>& subDeviceVec,
156 const PolyDriverList &polylist,
157 ReturnValue (Interface::*getNameMethodPtr)(size_t, std::string&) const,
158 ReturnValue (Interface::*getNrOfSensorsMethodPtr)(size_t& ) const)
159{
160 std::map<std::string, SensorInSubDevice> sensorLocationMap;
161
162 subDeviceVec.resize(polylist.size());
163
164 for(int p=0; p<polylist.size(); p++)
165 {
166 // If this fails it is ok, this just means that this devices does not expose this kind of sensors
167 polylist[p]->poly->view(subDeviceVec[p]);
168
169 if (subDeviceVec[p])
170 {
171 size_t nrOfSensorsInSubDevice = 0;
172 ReturnValue retnum = MAS_CALL_MEMBER_FN(subDeviceVec[p], getNrOfSensorsMethodPtr)(nrOfSensorsInSubDevice);
173 if (!retnum)
174 {
175 yCError(MULTIPLEANALOGSENSORSREMAPPER) << "Failure in reading number of sensors of type " << MAS_getTagFromEnum(sensorType) << " in device " << polylist[p]->key;
176 return false;
177 }
178 for (size_t s=0; s < nrOfSensorsInSubDevice; s++)
179 {
180 std::string name;
181 bool ok = MAS_CALL_MEMBER_FN(subDeviceVec[p], getNameMethodPtr)(s,name);
182 if (!ok)
183 {
184 yCError(MULTIPLEANALOGSENSORSREMAPPER) << "Failure in getting a name in the device " << polylist[p]->key;
185 return false;
186 }
187
188 // If the name is already in the map, raise an error
189 if (sensorLocationMap.find(name) != sensorLocationMap.end())
190 {
191 SensorInSubDevice deviceWithSameName = sensorLocationMap.find(name)->second;
192 yCError(MULTIPLEANALOGSENSORSREMAPPER)
193 << "Sensor ambiguity: sensor with name"
194 << name
195 << "present on both device"
196 << polylist[p]->key
197 << polylist[deviceWithSameName.subDevice]->key;
198 return false;
199 }
200
201 sensorLocationMap[name] = SensorInSubDevice(p, s);
202 }
203 }
204 }
205
206 // Fill the indices map given the name of all the subdevices
207 std::vector<SensorInSubDevice>& sensIndicesMap = m_indicesMap[static_cast<size_t>(sensorType)];
208 sensIndicesMap.resize(m_remappedSensors[sensorType].size());
209 for(size_t i=0; i < m_remappedSensors[sensorType].size(); i++)
210 {
211 std::string name = m_remappedSensors[sensorType][i];
212 if (sensorLocationMap.find(name) == sensorLocationMap.end())
213 {
214 yCError(MULTIPLEANALOGSENSORSREMAPPER) << "Impossible to find sensor name" << name << ", exiting.";
215 yCError(MULTIPLEANALOGSENSORSREMAPPER) << " Names of available sensors are:";
216 for(auto& availableDevice: sensorLocationMap)
217 {
218 yCError(MULTIPLEANALOGSENSORSREMAPPER) << " * " << availableDevice.first;
219 }
220 return false;
221 }
222
223 sensIndicesMap[i] = sensorLocationMap.find(name)->second;
224 }
225
226 return true;
227}

231bool MultipleAnalogSensorsRemapper::attachAll(const PolyDriverList &polylist)
232{
233 bool ok = true;
234 m_indicesMap.resize(MAS_NrOfSensorTypes);
235 ok = ok && genericAttachAll(ThreeAxisGyroscopes, m_iThreeAxisGyroscopes, polylist,
236 &IThreeAxisGyroscopes::getThreeAxisGyroscopeName, &IThreeAxisGyroscopes::getNrOfThreeAxisGyroscopes);
237 ok = ok && genericAttachAll(ThreeAxisLinearAccelerometers, m_iThreeAxisLinearAccelerometers, polylist,
238 &IThreeAxisLinearAccelerometers::getThreeAxisLinearAccelerometerName, &IThreeAxisLinearAccelerometers::getNrOfThreeAxisLinearAccelerometers);
239 ok = ok && genericAttachAll(ThreeAxisAngularAccelerometers, m_iThreeAxisAngularAccelerometers, polylist,
240 &IThreeAxisAngularAccelerometers::getThreeAxisAngularAccelerometerName, &IThreeAxisAngularAccelerometers::getNrOfThreeAxisAngularAccelerometers);
241 ok = ok && genericAttachAll(ThreeAxisMagnetometers, m_iThreeAxisMagnetometers, polylist,
242 &IThreeAxisMagnetometers::getThreeAxisMagnetometerName, &IThreeAxisMagnetometers::getNrOfThreeAxisMagnetometers);
243 ok = ok && genericAttachAll(PositionSensors, m_iPositionSensors, polylist,
244 &IPositionSensors::getPositionSensorName, &IPositionSensors::getNrOfPositionSensors);
245 ok = ok && genericAttachAll(LinearVelocitySensors, m_iLinearVelocitySensors, polylist,
246 &ILinearVelocitySensors::getLinearVelocitySensorName, &ILinearVelocitySensors::getNrOfLinearVelocitySensors);
247 ok = ok && genericAttachAll(OrientationSensors, m_iOrientationSensors, polylist,
248 &IOrientationSensors::getOrientationSensorName, &IOrientationSensors::getNrOfOrientationSensors);
249 ok = ok && genericAttachAll(TemperatureSensors, m_iTemperatureSensors, polylist,
250 &ITemperatureSensors::getTemperatureSensorName, &ITemperatureSensors::getNrOfTemperatureSensors);
251 ok = ok && genericAttachAll(SixAxisForceTorqueSensors, m_iSixAxisForceTorqueSensors, polylist,
252 &ISixAxisForceTorqueSensors::getSixAxisForceTorqueSensorName, &ISixAxisForceTorqueSensors::getNrOfSixAxisForceTorqueSensors);
253 ok = ok && genericAttachAll(ContactLoadCellArrays, m_iContactLoadCellArrays, polylist,
254 &IContactLoadCellArrays::getContactLoadCellArrayName, &IContactLoadCellArrays::getNrOfContactLoadCellArrays);
255 ok = ok && genericAttachAll(EncoderArrays, m_iEncoderArrays, polylist,
256 &IEncoderArrays::getEncoderArrayName, &IEncoderArrays::getNrOfEncoderArrays);
257 ok = ok && genericAttachAll(SkinPatches, m_iSkinPatches, polylist,
258 &ISkinPatches::getSkinPatchName, &ISkinPatches::getNrOfSkinPatches);
259
260 return ok;

284 size_t& sens_index,
285 const std::vector<Interface *>& subDeviceVec,
286 MAS_status (Interface::*methodPtr)(size_t) const) const
287{

294 }
295 return MAS_ERROR;
296 }
297
298 SensorInSubDevice subDeviceSensor = m_indicesMap[sensorType][sens_index];
299 return MAS_CALL_MEMBER_FN(subDeviceVec[subDeviceSensor.subDevice], methodPtr)(subDeviceSensor.indexInSubDevice);
300}

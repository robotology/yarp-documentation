96 std::list<yarp::stats::ConnectionStats> stats;
97 m_engine.update(stats);

99 yCDebug(YARPCONNINFO) << "Processed data:" << stats.size();
100
101 // Compute total bytes per second for each source/destination IP pair
102 std::map<std::pair<std::string, std::string>, double> ip_pair_bytes;
103 for (const auto& stat : stats)
104 {
105 // Get IP addresses from port names
106 std::string source_ip;
107 std::string dest_ip;
108
109 auto source_it = m_portsmap.find(stat.source);
110 if (source_it != m_portsmap.end()) {
111 source_ip = source_it->second.ip;
112 }
113 auto dest_it = m_portsmap.find(stat.destination);
114 if (dest_it != m_portsmap.end()) {
115 dest_ip = dest_it->second.ip;
116 }
117
118 // Accumulate bytes per second for this IP pair
119 if (!source_ip.empty() && !dest_ip.empty()) {
120 auto ip_pair = std::make_pair(source_ip, dest_ip);
121 ip_pair_bytes[ip_pair] += stat.bytes_per_second;
122 }
123 }
124
125 // Sort the stats based on the selected sorting mode
126 if (display_by_source_name) {sortConnections(stats, SortMode::Source);}
127 else if (display_by_dest_name) {sortConnections(stats, SortMode::Destination);}
128 else if (display_by_bps) {sortConnections(stats, SortMode::BytesPerSecondDesc);}
129
130 //Print the stats on screen
131 yCInfo(YARPCONNINFO) << "=== Connection statistics ===";
132 size_t longest_source = 0;
133 size_t longest_destination = 0;
134 for (auto it=stats.begin(); it!=stats.end(); it++)
135 {
136 if (it->source.length() > longest_source) { longest_source = it->source.length();}
137 if (it->destination.length() > longest_destination) { longest_destination = it->destination.length(); }
138 }
139 for (auto it=stats.begin(); it!=stats.end(); it++)
140 {
141 std::ostringstream line;
142
143 line << std::left
144 << std::setw(longest_source+2) << it->source
145 << " -> "
146 << std::setw(longest_destination+2) << it->destination
147 << " | "
148 << std::right
149 << std::setw(12) << it->bytes_per_second << " B/s"
150 << " | "
151 << std::setw(8) << it->frequency << " pkt/s"
152 << " | "
153 // << (it->isForward ? "forward" : "backward")
154 // << " | "
155 << std::setw(8) << it->carrier
156 << " | "
157 << std::setw(8) << std::fixed << std::setprecision(3) << it->timestamp;
158
159 yInfo() << line.str();
160 }
161
162 // Print IP pair totals
163 yCInfo(YARPCONNINFO);
164 yCInfo(YARPCONNINFO) << "=== Bytes per second by IP address pairs ===";
165 for (const auto& [ip_pair, total_bytes] : ip_pair_bytes)
166 {
167 std::ostringstream line;
168 line << std::left
169 << std::setw(15) << ip_pair.first
170 << " -> "
171 << std::setw(15) << ip_pair.second
172 << " | "
173 << std::right
174 << std::setw(12) << total_bytes << " B/s";
175 yInfo() << line.str();
176 }
177
178 return true;
179 }

181 void sortConnections(std::list<yarp::stats::ConnectionStats>& connections, SortMode mode)
182 {
183 switch (mode)
184 {
185 case SortMode::Source:
186 connections.sort(
187 [](const auto& a, const auto& b)
188 {
189 return a.source < b.source;
190 });
191 break;
192
193 case SortMode::Destination:
194 connections.sort(
195 [](const auto& a, const auto& b)
196 {
197 return a.destination < b.destination;
198 });
199 break;
200
201 case SortMode::BytesPerSecondDesc:
202 connections.sort(
203 [](const auto& a, const auto& b)
204 {
205 return a.bytes_per_second > b.bytes_per_second;
206 });
207 break;
208 }
209 }

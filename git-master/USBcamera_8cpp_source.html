39bool USBCameraDriver::open(yarp::os::Searchable& config)
40{
41 // open OS dependant device
42 yCTrace(USBCAMERA) << "input params are " << config.toString();
43
44#if defined(_MSC_VER)
45 os_device = (DeviceDriver*)new WIN_camera;
46#elif defined __unix
47 os_device = (DeviceDriver*)new V4L_camera;
48#endif
49
50 yarp::os::Property prop;
51 prop.fromString(config.toString());
52 if (!prop.check("pixelType")) {
53 switch (pixelType) {
54 case VOCAB_PIXEL_MONO:
55 prop.put("pixelType", VOCAB_PIXEL_MONO);
56 break;
57
58 case VOCAB_PIXEL_RGB:
59 default:
60 prop.put("pixelType", VOCAB_PIXEL_RGB);
61 break;
62 }
63 }
64 if (!os_device->open(prop)) {
65 delete os_device;
66 return false;
67 }
68
69 os_device->view(frameGrabberImage);
70 os_device->view(frameGrabberImageRaw);
71 os_device->view(deviceControls);
72 os_device->view(deviceTimed);
73 os_device->view(deviceRgbVisualParam);
74
75 if (frameGrabberImage != nullptr) {
76 _width = frameGrabberImage->width();
77 _height = frameGrabberImage->height();
78 }
79
80 if (frameGrabberImageRaw != nullptr) {
81 _width = frameGrabberImageRaw->width();
82 _height = frameGrabberImageRaw->height();
83 }
84 return true;
85}

35 inline void exec_iRangefinder2D_test_1(IRangefinder2D* irf, const ValuestoCheck& v=ValuestoCheck())
36 {
37 REQUIRE(irf != nullptr);
38
39 bool b;
40
41 // wait for the device to be ready
42 yarp::dev::IRangefinder2D::Device_status status;
43 for (size_t counter = 0; counter<10; counter++)
44 {
45 b = irf->getDeviceStatus(status);
46 CHECK(b);
47 if (status == yarp::dev::IRangefinder2D::Device_status::DEVICE_OK_IN_USE) break;
48 yDebug() << "Device not ready yet, retrying..";
49 yarp::os::Time::delay(0.5);
50 }
51 CHECK(status == yarp::dev::IRangefinder2D::Device_status::DEVICE_OK_IN_USE);
52
53 std::string info;
54 b = irf->getDeviceInfo(info);
55 CHECK(b);
56 CHECK(!info.empty());
57
58 double scanrate;
59 b = irf->getScanRate(scanrate);
60 CHECK(b);
61 CHECK(scanrate==v.test_scanrate);
62
63 double min_angle, max_angle;
64 b = irf->getScanLimits(min_angle, max_angle);
65 CHECK(b);
66 CHECK(min_angle == v.min_angle);
67 CHECK(max_angle == v.max_angle);
68
69 double hstep;
70 b = irf->getHorizontalResolution(hstep);
71 CHECK(b);
72 CHECK(hstep == v.test_horizontal_res);
73
74 double min,max;
75 b = irf->getDistanceRange(min,max);
76 CHECK(b);
77 CHECK(min == v.test_min);
78 CHECK(max == v.test_max);
79
80 std::vector<yarp::sig::LaserMeasurementData> las;
81 double timestamp;
82 b = irf->getLaserMeasurement(las, &timestamp);
83 CHECK(b);
84 CHECK(timestamp != 0);
85 CHECK(las.size() == v.test_lsize);
86
87 //check the measurement values.
88 //REQUIRE is needed to prevent segfault if nothing is received.
89 {
90 REQUIRE(las.size() > 0);
91 double r,t;
92 las[0].get_polar(r, t);
93 CHECK(yarp::dev::tests::check_approximate(r, v.test_rho));
94 CHECK(yarp::dev::tests::check_approximate(t, v.test_theta));
95 }
96 {
97 REQUIRE(las.size() > 0);
98 double x,y;
99 las[0].get_cartesian(x, y);
100 CHECK(yarp::dev::tests::check_approximate(x, v.test_cartesian_x));
101 CHECK(yarp::dev::tests::check_approximate(y, v.test_cartesian_y));
102 }
103
104 }

46inline size_t MAS_getMeasureSizeFromEnum(const MAS_SensorTypeServer type)
47{
48 switch(type)
49 {
50 case ThreeAxisGyroscopes:

52 break;
53 case ThreeAxisLinearAccelerometers:
54 return 3;
55 break;
56 case ThreeAxisAngularAccelerometers:
57 return 3;
58 break;
59 case ThreeAxisMagnetometers:
60 return 3;
61 break;
62 case OrientationSensors:
63 return 3;
64 break;
65 case PositionSensors:
66 return 3;
67 break;
68 case LinearVelocitySensors:
69 return 3;
70 break;
71 case TemperatureSensors:
72 return 1;
73 break;
74 case SixAxisForceTorqueSensors:
75 return 6;
76 break;

78 assert(false);
79 // "MAS_getMeasureSizeFromEnum passed unexepcted enum";

162template<typename Interface>
163ReturnValue MultipleAnalogSensorsServer::populateSensorsMetadata(Interface * wrappedDeviceInterface,
164 std::vector<SensorMetadata>& metadataVector, const std::string& tag,
165 yarp::dev::ReturnValue (Interface::*getNrOfSensorsMethodPtr)(size_t&) const,
166 yarp::dev::ReturnValue (Interface::*getNameMethodPtr)(size_t, std::string&) const,
167 yarp::dev::ReturnValue (Interface::*getFrameNameMethodPtr)(size_t, std::string&) const)
168{
169 if (wrappedDeviceInterface)
170 {
171 size_t nrOfSensors = 0;
172 ReturnValue retnum = MAS_CALL_MEMBER_FN(wrappedDeviceInterface, getNrOfSensorsMethodPtr)(nrOfSensors);
173 if (!retnum)
174 {
175 yCError(MULTIPLEANALOGSENSORSSERVER,
176 "Failure in reading number of sensors of type %s.",
177 tag.c_str());
178 return retnum;
179 }
180 metadataVector.resize(nrOfSensors);
181 for (size_t i=0; i < nrOfSensors; i++)
182 {
183 std::string sensorName;
184 ReturnValue ret = MAS_CALL_MEMBER_FN(wrappedDeviceInterface, getNameMethodPtr)(i, sensorName);
185 if (!ret)
186 {
187 yCError(MULTIPLEANALOGSENSORSSERVER,
188 "Failure in reading name of sensor of type %s at index %zu.",
189 tag.c_str(),
190 i);
191 return ret;
192 }
193 std::string frameName;
194 ret = MAS_CALL_MEMBER_FN(wrappedDeviceInterface, getFrameNameMethodPtr)(i, frameName);
195 if (!ret)
196 {
197 yCError(MULTIPLEANALOGSENSORSSERVER,
198 "Failure in reading frame name of sensor of type %s at index %zu.",
199 tag.c_str(),
200 i);
201 return ret;
202 }
203
204 metadataVector[i].name = sensorName;
205 metadataVector[i].frameName = frameName;
206 metadataVector[i].additionalMetadata = "";
207 }
208
209 }
210 else
211 {
212 metadataVector.resize(0);
213 }
214 return ReturnValue_ok;
215}

317 const std::vector< SensorMetadata >& metadataVector,

387bool MultipleAnalogSensorsServer::attach(yarp::dev::PolyDriver* poly)
388{
389 if (!poly)
390 {
391 yCError(MULTIPLEANALOGSENSORSSERVER, "Null pointer passed to attach.");
392 close();
393 return false;
394 }
395
396 // View all the interfaces
397 poly->view(m_iThreeAxisGyroscopes);
398 poly->view(m_iThreeAxisLinearAccelerometers);
399 poly->view(m_iThreeAxisAngularAccelerometers);
400 poly->view(m_iThreeAxisMagnetometers);
401 poly->view(m_iPositionSensors);
402 poly->view(m_iLinearVelocitySensors);
403 poly->view(m_iOrientationSensors);
404 poly->view(m_iTemperatureSensors);
405 poly->view(m_iSixAxisForceTorqueSensors);
406 poly->view(m_iContactLoadCellArrays);
407 poly->view(m_iEncoderArrays);
408 poly->view(m_iSkinPatches);
409
410
411 // Populate the RPC data to be served on the RPC port
412 bool ok = populateAllSensorsMetadata();
413
414 if(!ok)
415 {
416 yCError(MULTIPLEANALOGSENSORSSERVER, "Failure in populateAllSensorsMetadata()");
417 close();
418 return false;
419 }
420
421 // Attach was successful, open the ports
422 ok = m_streamingPort.open(m_streamingPortName);
423 if (!ok)
424 {
425 yCError(MULTIPLEANALOGSENSORSSERVER, "Failure in opening port named %s.", m_streamingPortName.c_str());
426 close();
427 return false;
428 }
429
430 ok = this->yarp().attachAsServer(m_rpcPort);
431 if (!ok)
432 {
433 yCError(MULTIPLEANALOGSENSORSSERVER, "Failure in attaching RPC port to thrift RPC interface.");
434 close();
435 return false;
436 }
437
438 ok = m_rpcPort.open(m_RPCPortName);
439 if (!ok)
440 {
441 yCError(MULTIPLEANALOGSENSORSSERVER, "Failure in opening port named %s.", m_RPCPortName.c_str());
442 close();
443 return false;
444 }
445
446 // Set rate period
447 ok = this->setPeriod(m_periodInS);
448 ok = ok && this->start();
449 if (!ok)
450 {
451 yCError(MULTIPLEANALOGSENSORSSERVER, "Failure in starting thread.");
452 close();
453 return false;
454 }
455
456 yCDebug(MULTIPLEANALOGSENSORSSERVER, "Attach complete");
457 return true;
458}

480template<typename Interface>
481bool MultipleAnalogSensorsServer::genericStreamData(Interface* wrappedDeviceInterface,
482 const std::vector< SensorMetadata >& metadataVector,
483 std::vector< typename yarp::dev::SensorMeasurement >& streamingDataVector,
484 yarp::dev::MAS_status (Interface::*getStatusMethodPtr)(size_t) const,
485 ReturnValue (Interface::*getMeasureMethodPtr)(size_t, yarp::sig::Vector&, double&) const,
486 const char* sensorType)
487{
488 if (wrappedDeviceInterface)
489 {
490 size_t nrOfSensors = metadataVector.size();
491 for (size_t i=0; i < nrOfSensors; i++)
492 {
493 yarp::sig::Vector& outputBuffer = streamingDataVector[i].measurement;
494 double& outputTimestamp = streamingDataVector[i].timestamp;
495 yarp::dev::MAS_status status = MAS_CALL_MEMBER_FN(wrappedDeviceInterface, getStatusMethodPtr)(i);
496 if (status == yarp::dev::MAS_OK)
497 {
498 MAS_CALL_MEMBER_FN(wrappedDeviceInterface, getMeasureMethodPtr)(i, outputBuffer, outputTimestamp);
499 } else
500 {
501 yCError(MULTIPLEANALOGSENSORSSERVER,
502 "Failure in reading data from sensor %s of type %s with code %s, no data will be sent on the port.",
503 metadataVector[i].name.c_str(), sensorType, MAS_getStatusString(status).c_str());
504 return false;
505 }
506 }
507 }
508
509 return true;
510}

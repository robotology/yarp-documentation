52bool RemoteControlBoardRemapper::openAllRemoteControlBoards(const Property& remoteControlBoardsOptions)
53{
54 for(size_t ctrlBrd=0; ctrlBrd < m_remoteControlBoards.size(); ctrlBrd++ )
55 {
56 std::string remote = m_remoteControlBoards[ctrlBrd];
57 // Note: as local parameter we use localPortPrefix+remoteOfTheReportControlBoard
58 std::string local = m_localPortPrefix+remote;
59
60 Property options = remoteControlBoardsOptions;
61 options.put("device", "controlBoard_nwc_yarp");
62 options.put("local", local);
63 options.put("remote", remote);
64
65 //the pointer will be inserted in a PolyDriverList, that will take care of closing the PolyDriver when it is destroyed
66 PolyDriver* tmpDriver = new PolyDriver;
67 bool ok = tmpDriver->open(options);
68
69 if( !ok || !(tmpDriver->isValid()) )
70 {
71 yCError(REMOTECONTROLBOARDREMAPPER) << "Opening controlBoard_nwc_yarp with remote \"" << remote << "\", opening the device failed.";
72 closeAllRemoteControlBoards();
73 return false;
74 }
75
76 // We use the remote name of the controlBoard_nwc_yarp as the key for it, in absence of anything better
77 m_remoteControlBoardDevices.push(tmpDriver,remote.c_str());
78 }
79
80 return true;
81}

83bool RemoteControlBoardRemapper::open(Searchable& config)
84{
85 // From RemoteControlBoardRemapper_ParamsParser

89 // Load the REMOTE_CONTROLBOARD_OPTIONS, containing any additional option to pass to the remote control boards
90 Property prop;
91 prop.fromString(config.toString());
92 Property remoteControlBoardsOptions;
93 Bottle & optionsGroupBot = prop.findGroup("REMOTE_CONTROLBOARD_OPTIONS");
94 if( !(optionsGroupBot.isNull()) )
95 {
96 remoteControlBoardsOptions.fromString(optionsGroupBot.toString());
97 }
98
99 // Open all the remote control boards
100 b = openAllRemoteControlBoards(remoteControlBoardsOptions);
101 if (!b) return false;
102
103 // Device opened, now we open the ControlBoardRemapper and then we call attachAll
104 bool ok = true;
105/* ok = ControlBoardRemappingEngine::open(prop);
106 if( !ok )
107 {
108 yCError(REMOTECONTROLBOARDREMAPPER) << "Opening the controlboardremapper device, opening the device failed.";
109 ControlBoardRemappingEngine::detachAllDevices();
110 closeAllRemoteControlBoards();
111 return false;
112 }
113*/
114 this->setAxesNames(m_axesNames);
115 this->setNrOfControlledAxes(m_axesNames.size());
116
117 // If open went ok, we now call attachAll

119
120 if( !ok )
121 {
122 yCError(REMOTECONTROLBOARDREMAPPER) << "Calling attachAll in the controlboardremapper device, opening the device failed.";
123 ControlBoardRemappingEngine::detachAllDevices();
124 closeAllRemoteControlBoards();
125 return false;
126 }
127
128 // All went ok, return true
129 // TODO: close devices that are not actually used by the remapper

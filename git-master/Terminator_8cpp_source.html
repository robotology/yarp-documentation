46bool Terminator::terminateByName(const char* name)
47{
48 if (name == nullptr) {
49 return false;
50 }
51
52 std::string s(name);
53
54 if (name[0] != '/') {
55 s.clear();
56 s += "/";
57 s += name;
58 }
59
60 Contact c = NetworkBase::queryName(s);
61 if (!c.isValid()) {
62 yCError(TERMINATOR, "Terminator port not found");
63 return false;
64 }
65 c.setCarrier("tcp");
66
67 // Client side of the TerminatorMsgs thrift service: the commands are
68 // sent through a (not registered) port connected to the Terminee port.
69 Port port;
70 if (!port.openFake("terminator")) {
71 yCError(TERMINATOR, "Cannot open the terminator client port");
72 return false;
73 }
74 if (!port.addOutput(c)) {
75 yCError(TERMINATOR, "Cannot connect to %s", s.c_str());
76 return false;
77 }
78 TerminatorMsgs client;
79 client.yarp().attachAsClient(port);
80 if (!client.quit()) {
81 yCError(TERMINATOR, "The quit request has not been accepted by %s", s.c_str());
82 return false;
83 }
84
85 return true;
86}

91Terminee::Terminee(const char* name)
92{
93 ok = false;
94 implementation = nullptr;
95 if (name == nullptr) {
96 quit = true;
97 yCError(TERMINATOR, "Terminator: Please supply a proper port name");
98 return;
99 }
100
101 std::string s(name);
102 if (name[0] != '/') {
103 s.clear();
104 s += "/";
105 s += name;
106 }
107
108 implementation = new TermineeHelper();
109 yCAssert(TERMINATOR, implementation != nullptr);
110 TermineeHelper& helper = HELPER(implementation);
111 ok = helper.open(s);
112 if (!ok) {
113 quit = true;
114 yCError(TERMINATOR, "Kill port conflict: make sure you supply a distinct --name /PORTNAME");
115 } else {
116 quit = false;
117 start();
118 }
119}

143void Terminee::run()
144{

146 TerminatorServer server(quit);
147 while (!isStopping() && !quit) {
148 // The thrift service reads the command and writes the reply
149 // (including the errors, e.g. for unknown commands).

39bool FrameGrabberCropper::attach(yarp::dev::PolyDriver* poly)
40{
41 if (!poly->isValid()) {
42 yCError(FRAMEGRABBERCROPPER) << "The device is not valid";
43 return false;
44 }
45
46 poly->view(FrameGrabberCropperOf<yarp::sig::ImageOf<yarp::sig::PixelRgb>>::iFrameGrabberOf);
47 poly->view(FrameGrabberCropperOf<yarp::sig::ImageOf<yarp::sig::PixelMono>>::iFrameGrabberOf);
48 poly->view(iFrameGrabberControls);
49 poly->view(iFrameGrabberControlsDC1394);
50 poly->view(iRgbVisualParams);
51 poly->view(iPreciselyTimed);
52
53 if (!FrameGrabberCropperOf<yarp::sig::ImageOf<yarp::sig::PixelRgb>>::iFrameGrabberOf && !FrameGrabberCropperOf<yarp::sig::ImageOf<yarp::sig::PixelMono>>::iFrameGrabberOf) {
54 yCError(FRAMEGRABBERCROPPER) << "The device does not implement the required interfaces";
55 return false;
56 }
57
58 return true;
59}

23TEST_CASE("pm::depthimage_to_rgbTest", "[yarp::pm]")
24{
25 YARP_REQUIRE_PLUGIN("depthimage_to_rgb", "portmonitor")
26
27 yarp::os::Network yarp(yarp::os::YARP_CLOCK_SYSTEM);
28
29 yarp::os::NetworkBase::setLocalMode(true);
30
31 REQUIRE(yarp::os::Carriers::getCarrierTemplate("portmonitor") != nullptr);
32
33 yarp::os::Time::delay(1.0);
34
35 struct TestCase {
36 std::string carrier;
37 bool should_convert;
38 };
39
40 auto tc = GENERATE(
41 TestCase {"fast_tcp", false},
42 TestCase {"fast_tcp+recv.portmonitor+file.depthimage_to_rgb+type.dll", true}
43 );
44
45 SECTION("Test depth image to RGB heatmap conversion with portmonitor: " + tc.carrier)
46 {
47 yarp::os::BufferedPort<yarp::sig::ImageOf<yarp::sig::PixelFloat>> sender;
48 yarp::os::BufferedPort<yarp::sig::FlexImage> receiver;
49
50 sender.open("/send");
51 receiver.open("/recv");
52
53 bool b = yarp::os::Network::connect("/send", "/recv", tc.carrier);
54 REQUIRE(b);
55
56 // Create a test depth image (16x16 pixels)
57 constexpr int width = 16;
58 constexpr int height = 16;
59
60 yarp::sig::ImageOf<yarp::sig::PixelFloat>& depthImg = sender.prepare();
61 depthImg.resize(width, height);
62
63 // Fill with test data - values between 0.2 and 10.0 meters
64 for (int y = 0; y < height; y++)
65 {
66 for (int x = 0; x < width; x++)
67 {
68 float depth_value = 0.2f + (static_cast<float>(x + y * width) / (width * height - 1)) * 9.8f;
69 depthImg.pixel(x, y) = depth_value;
70 }
71 }
72
73 sender.write();
74
75 yarp::os::Time::delay(0.5);
76
77 yarp::sig::FlexImage* receivedImg = receiver.read();
78 REQUIRE(receivedImg != nullptr);
79
80 // Verify dimensions
81 CHECK(receivedImg->width() == width);
82 CHECK(receivedImg->height() == height);
83
84 if (tc.should_convert)
85 {
86 // Verify it's converted to RGB heatmap
87 CHECK(receivedImg->getPixelCode() == VOCAB_PIXEL_RGB);
88 CHECK(receivedImg->getPixelSize() == 3);
89
90 // Verify some pixels have valid RGB values
91 unsigned char* pixels = receivedImg->getRawImage();
92 bool hasNonZeroPixel = false;
93 for (int i = 0; i < width * height * 3; i++)
94 {
95 CHECK(pixels[i] <= 255);
96 if (pixels[i] > 0)
97 {
98 hasNonZeroPixel = true;
99 }
100 }
101 // At least some pixels should have color (not all black)
102 CHECK(hasNonZeroPixel);
103 }
104 else
105 {
106 // Without conversion, should still be float
107 CHECK(receivedImg->getPixelCode() == VOCAB_PIXEL_MONO_FLOAT);
108 }
109
110 receiver.close();
111 sender.close();
112 }
113
114 yarp::os::NetworkBase::setLocalMode(false);
115}

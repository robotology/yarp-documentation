78ReturnValue ImplementTorqueControl::setMotorTorqueParams(int j, const yarp::dev::MotorTorqueParameters params)
79{
80 std::lock_guard lock(m_imp_mutex);
81 JOINTIDCHECK(IMPLEMENT_LAYER_COMPONENT,j)
82
83 int k;
84
85 yarp::dev::MotorTorqueParameters params_raw;
86 castToMapper(m_helper)->bemf_user2raw(params.bemf, j, params_raw.bemf, k);

88 params_raw.bemf_scale = params.bemf_scale;
89 params_raw.ktau_scale = params.ktau_scale;
90
91 castToMapper(m_helper)->viscousPos_user2raw(params.viscousPos, j, params_raw.viscousPos, k);
92 castToMapper(m_helper)->viscousNeg_user2raw(params.viscousNeg, j, params_raw.viscousNeg, k);
93 castToMapper(m_helper)->coulombPos_user2raw(params.coulombPos, j, params_raw.coulombPos, k);
94 castToMapper(m_helper)->coulombNeg_user2raw(params.coulombNeg, j, params_raw.coulombNeg, k);
95 castToMapper(m_helper)->velocityThres_user2raw(params.velocityThres, j, params_raw.velocityThres, k);

100ReturnValue ImplementTorqueControl::getMotorTorqueParams(int j, yarp::dev::MotorTorqueParameters *params)
101{
102 std::lock_guard lock(m_imp_mutex);
103 JOINTIDCHECK(IMPLEMENT_LAYER_COMPONENT,j)
104 POINTERCHECK(IMPLEMENT_LAYER_COMPONENT,params);
105
106 int k=castToMapper(m_helper)->toHw(j);
107
108 yarp::dev::MotorTorqueParameters params_raw;
109 ReturnValue b = m_iraw->getMotorTorqueParamsRaw(k, &params_raw);
110 int tmp_j;
111
112 if (b)
113 {
114 *params = params_raw;
115 castToMapper(m_helper)->bemf_raw2user(params_raw.bemf, k, (*params).bemf, tmp_j);
116 castToMapper(m_helper)->ktau_raw2user(params_raw.ktau, k, (*params).ktau, tmp_j);
117 (*params).bemf_scale = params_raw.bemf_scale;
118 (*params).ktau_scale = params_raw.ktau_scale;
119 castToMapper(m_helper)->viscousPos_raw2user(params_raw.viscousPos, k, (*params).viscousPos, tmp_j);
120 castToMapper(m_helper)->viscousNeg_raw2user(params_raw.viscousNeg, k, (*params).viscousNeg, tmp_j);
121 castToMapper(m_helper)->coulombPos_raw2user(params_raw.coulombPos, k, (*params).coulombPos, tmp_j);
122 castToMapper(m_helper)->coulombNeg_raw2user(params_raw.coulombNeg, k, (*params).coulombNeg, tmp_j);
123 castToMapper(m_helper)->velocityThres_raw2user(params_raw.velocityThres, k, (*params).velocityThres, tmp_j);
124 }
125 return b;
126}

172ReturnValue ImplementTorqueControl::setRefTorques(const int n_joints, const int *joints, const double *t)
173{
174 std::lock_guard lock(m_imp_mutex);
175 POINTERCHECK(IMPLEMENT_LAYER_COMPONENT,joints);
176 POINTERCHECK(IMPLEMENT_LAYER_COMPONENT,t);
177 JOINTSIDCHECK(IMPLEMENT_LAYER_COMPONENT,n_joints,joints);
178
179 for(int idx=0; idx<n_joints; idx++)
180 {
181 m_buffer_doubles[idx] = castToMapper(m_helper)->trqN2S(t[idx], joints[idx]);
182 m_buffer_ints[idx] = castToMapper(m_helper)->toHw(joints[idx]);
183 }
184 ReturnValue ret = m_iraw->setRefTorquesRaw(n_joints, m_buffer_ints.data(), m_buffer_doubles.data());
185
186 return ret;

28// Monitor: create
29bool RpcMonitor::create(const yarp::os::Property& options)
30{
31 sender = options.find("sender_side").asBool();
32 source = options.find("source").asString();
33 destination = options.find("destination").asString();
34 const std::string source_port = (sender ? source : destination) + "/monitor";
35
36 const std::string monitor_port = options.check("monitor_name", yarp::os::Value("/monitor")).asString();
37
38 if (!sender) {
39 yCError(RPCMONITOR, "Attaching on receiver side is not supported yet.");
40 return false;
41 }
42 if (!port.open(source_port)) {
43 yCError(RPCMONITOR, "Could not open port %s.", source_port.c_str());
44 return false;
45 }
46 if (!yarp::os::Network::connect(source_port, monitor_port, "fast_tcp"))
47 {
48 yCError(RPCMONITOR, "Could not connect to port %s.", monitor_port.c_str());
49 return false;
50 }
51 return true;
52}

54// Monitor: update
55yarp::os::Things& RpcMonitor::update(yarp::os::Things& thing)
56{
57 yarp::os::Bottle msg;
58 msg.addFloat64(yarp::os::SystemClock::nowSystem());
59 msg.addString(source);
60 msg.addString(destination);
61 msg.addString("command");
62 msg.addString(sender ? "client" : "server");
63 msg.addString("RpcMonitor");
64 auto& bcmd = msg.addList();
65 auto& bargs [[maybe_unused]] = msg.addList();
66 auto& breply [[maybe_unused]] = msg.addList();
67
68 if (const auto* reply = thing.cast_as<yarp::os::CommandBottle>()) {
69 bcmd.addString(reply->cmd.get(0).toString());
70 yarp::os::Portable::copyPortable(reply->cmd, bargs);
71 } else if (const auto* cmd = thing.cast_as<yarp::os::Bottle>()) {
72 bcmd.addString(cmd->get(0).toString());
73 yarp::os::Portable::copyPortable(*cmd, bargs);
74 } else {
75 yCWarning(RPCMONITOR) << "Sending unknown command";
76 bcmd.addString("[unknown]");
77 }
78 yCDebug(RPCMONITOR, "Writing: %s", msg.toString().c_str());
79 if (port.getOutputCount() > 0) {
80 port.write(msg);
81 }
82 return thing;
83}

94 msg.addString("RpcMonitor");

98

100 bcmd.addString(reply->cmd.get(0).toString());
101 yarp::os::Portable::copyPortable(reply->cmd, bargs);
102 yarp::os::Portable::copyPortable(reply->reply, breply);

104 yarp::os::Portable::copyPortable(*reply, breply);
105 } else {

309 bool read(yarp::os::ConnectionReader& connection) override
310 {
311 // Read into a temporary object, so that a failed or malformed read
312 // leaves this matrix (and its row pointers) untouched.
313 Matrix tmp;
314 if (!tmp.MatrixData::read(connection)) {
315 return false;
316 }
317 if (tmp.nrows < 0 ||
318 tmp.ncols < 0 ||
319 tmp.storage.size() != static_cast<size_t>(tmp.nrows) * static_cast<size_t>(tmp.ncols))
320 {
321 return false;
322 }
323 nrows = tmp.nrows;
324 ncols = tmp.ncols;
325 storage = std::move(tmp.storage);
326 updatePointers();
327 return true;
328 }
329
330 bool write(yarp::os::ConnectionWriter& connection) const override
331 {
332 return MatrixData::write(connection);

30bool RGBDSensor_nwc_yarp::open(yarp::os::Searchable& config)
31{
32 if (!parseParams(config)) { return false; }
33
34 bool ret = false;
35
36 // Opening Streaming ports
37 ret = m_colorFrame_StreamingPort.open(m_localImagePort);
38 ret &= m_depthFrame_StreamingPort.open(m_localDepthPort);
39
40 if (!ret)
41 {
42 yCError(RGBDSENSOR_NWC_YARP) << " cannot open local streaming ports: " << m_localImagePort << " or " << m_localDepthPort;
43 m_colorFrame_StreamingPort.close();
44 m_depthFrame_StreamingPort.close();
45 }
46
47 if (!yarp::os::Network::connect(m_remoteImagePort, m_colorFrame_StreamingPort.getName(), m_ImageCarrier))
48 {
49 yCError(RGBDSENSOR_NWC_YARP) << m_colorFrame_StreamingPort.getName() << " cannot connect to remote port " << m_remoteImagePort << "with carrier " << m_ImageCarrier;
50 return false;
51 }
52
53 if (!yarp::os::Network::connect(m_remoteDepthPort, m_depthFrame_StreamingPort.getName(), m_DepthCarrier))
54 {
55 yCError(RGBDSENSOR_NWC_YARP) << m_depthFrame_StreamingPort.getName() << " cannot connect to remote port " << m_remoteDepthPort << "with carrier " << m_DepthCarrier;
56 return false;
57 }
58
59
60 // RPC port
61 ret = m_rpcPort.open(m_localRpcPort);
62 if (!ret)
63 {
64 yCError(RGBDSENSOR_NWC_YARP) << " cannot open local RPC port " << m_localRpcPort;
65 m_colorFrame_StreamingPort.close();
66 m_depthFrame_StreamingPort.close();
67 m_rpcPort.close();
68 }
69
70 if (!m_rpcPort.addOutput(m_remoteRpcPort))
71 {
72 yCError(RGBDSENSOR_NWC_YARP) << " cannot connect to port " << m_remoteRpcPort;
73 m_colorFrame_StreamingPort.close();
74 m_depthFrame_StreamingPort.close();
75 m_rpcPort.close();
76 return false;
77 }
78
79 if (!m_rgbdsensor_RPC.yarp().attachAsClient(m_rpcPort))
80 {
81 yCError(RGBDSENSOR_NWC_YARP, "Error! Cannot attach the port as a client");
82 m_colorFrame_StreamingPort.close();
83 m_depthFrame_StreamingPort.close();
84 m_rpcPort.close();
85 return false;
86 }
87
88 // Check the protocol version
89 if (!m_rgbdsensor_RPC.checkProtocolVersion()) {
90 return false;
91 }
92
93 streamingReader->attach(&m_colorFrame_StreamingPort, &m_depthFrame_StreamingPort);
94
95 return true;
96}

99{
100 m_colorFrame_StreamingPort.close();
101 m_depthFrame_StreamingPort.close();
102 m_rpcPort.close();
103 return true;
104}

17bool SoundFilter_resample::create(const yarp::os::Property &options)
18{
19 yCDebug(SOUNDFILTER_RESAMPLE, "created!\n");
20 yCDebug(SOUNDFILTER_RESAMPLE, "I am attached to the %s\n",
21 (options.find("sender_side").asBool()) ? "sender side" : "receiver side");
22
23 //set the user parameters
24 m_channel = (options.check("channel") ? options.find("channel").asInt8() : -1);
25 m_output_freq = (options.check("frequency") ? options.find("frequency").asInt32() : -1);
26 m_gain = (options.check("gain_percent") ? options.find("gain_percent").asFloat32() : -1)/100;
27
28 if (m_channel < 0)
29 {
30 yCInfo(SOUNDFILTER_RESAMPLE, "no channels selected for extraction, using all sound channels");
31 }
32 else
33 {
34 yCInfo(SOUNDFILTER_RESAMPLE, "channel selected: %d", m_channel);
35 }
36
37 if (m_gain < 0)
38 {
39 yCInfo(SOUNDFILTER_RESAMPLE, "No amplification requested");
40 }
41 else
42 {
43 yCInfo(SOUNDFILTER_RESAMPLE, "Amplification gain: %f (%d%%)", m_gain, int(m_gain*100));
44 }
45
46 if (m_output_freq < 0)
47 {
48 yCInfo(SOUNDFILTER_RESAMPLE, "No resampling requested");
49 }
50 else if (m_output_freq == 0)
51 {
52 yCError(SOUNDFILTER_RESAMPLE, "Invalid frequency value selected");
53 return false;
54 }
55 else
56 {
57 yCInfo(SOUNDFILTER_RESAMPLE, "output frequency : %d", m_output_freq);
58 }
59
60 return true;
61}

93yarp::os::Things & SoundFilter_resample::update(yarp::os::Things &thing)
94{
95 //get data to process
96 yarp::sig::Sound *s = thing.cast_as<yarp::sig::Sound>();
97 if (s == NULL)
98 {
99 yCWarning(SOUNDFILTER_RESAMPLE, "expected type Sound but got wrong data type!\n");
100 return thing;
101 }
102
103 //extract one channel
104 m_s2.clear();
105 if (m_channel >=0)
106 {
107 m_s2 = s->extractChannelAsSound(m_channel);
108 }
109 else
110 {
111 m_s2 = *s;
112 }
113
114 if (m_gain >= 0)
115 {
116 m_s2.amplify(m_gain);
117 }
118
119 if (m_output_freq>0)
120 {
121 bool b= yarp::sig::soundfilters::resample (m_s2,m_output_freq);
122 if (!b) yCError(SOUNDFILTER_RESAMPLE, "Resample failed!\n");
123 }
124
125 //send data
126 m_th.setPortWriter(&m_s2);
127 return m_th;
128}
